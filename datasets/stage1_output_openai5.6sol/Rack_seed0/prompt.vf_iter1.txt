def make_material(name, color, metallic=0.0, roughness=0.5):
    mat = bpy.data.materials.new(name)
    mat.diffuse_color = (*color, 1.0)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes.get("Principled BSDF")
    bsdf.inputs["Base Color"].default_value = (*color, 1.0)
    bsdf.inputs["Metallic"].default_value = metallic
    bsdf.inputs["Roughness"].default_value = roughness
    return mat

def add_bevel(obj, width=0.01, segments=2):
    modifier = obj.modifiers.new(name="Edge Bevel", type='BEVEL')
    modifier.width = width
    modifier.segments = segments
    modifier.limit_method = 'ANGLE'
    modifier.angle_limit = math.radians(25.0)

def create_box(name, location, dimensions, material, bevel=0.0):
    bpy.ops.mesh.primitive_cube_add(location=location)
    obj = bpy.context.object
    obj.name = name
    obj.dimensions = dimensions
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    obj.data.materials.append(material)
    if bevel > 0.0:
        add_bevel(obj, bevel, 2)
    return obj

def create_bar(name, p1, p2, thickness, width, material, bevel=0.006):
    start = Vector(p1)
    end = Vector(p2)
    direction = end - start
    length = direction.length
    midpoint = (start + end) * 0.5
    bpy.ops.mesh.primitive_cube_add(location=midpoint)
    obj = bpy.context.object
    obj.name = name
    obj.dimensions = (thickness, width, length)
    obj.rotation_mode = 'QUATERNION'
    obj.rotation_quaternion = direction.to_track_quat('Z', 'X')
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    obj.data.materials.append(material)
    if bevel > 0.0:
        add_bevel(obj, bevel, 2)
    return obj

def create_capsule_cutters(name, center_x, center_y, z_values,
                           slot_width=0.066, slot_height=0.14, depth=0.34,
                           arc_segments=6):
    vertices = []
    faces = []
    radius = slot_width * 0.5
    straight_half = max(0.0, (slot_height - slot_width) * 0.5)
    profile = []

    for i in range(arc_segments + 1):
        angle = math.pi * i / arc_segments
        profile.append((radius * math.cos(angle),
                        straight_half + radius * math.sin(angle)))
    for i in range(arc_segments + 1):
        angle = math.pi + math.pi * i / arc_segments
        profile.append((radius * math.cos(angle),
                        -straight_half + radius * math.sin(angle)))

    count = len(profile)
    half_depth = depth * 0.5

    for z_center in z_values:
        base = len(vertices)
        for y_offset in (-half_depth, half_depth):
            for px, pz in profile:
                vertices.append((center_x + px, center_y + y_offset, z_center + pz))

        faces.append(tuple(base + i for i in reversed(range(count))))
        faces.append(tuple(base + count + i for i in range(count)))

        for i in range(count):
            j = (i + 1) % count
            faces.append((base + i, base + j, base + count + j, base + count + i))

    mesh = bpy.data.meshes.new(name + "_Mesh")
    mesh.from_pydata(vertices, [], faces)
    mesh.update()
    cutter = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(cutter)
    return cutter

def create_cylinder(name, location, radius, depth, material, rotation=(0.0, 0.0, 0.0),
                    vertices=16, bevel=0.003):
    bpy.ops.mesh.primitive_cylinder_add(
        vertices=vertices,
        radius=radius,
        depth=depth,
        location=location,
        rotation=rotation
    )
    obj = bpy.context.object
    obj.name = name
    obj.data.materials.append(material)
    for polygon in obj.data.polygons:
        polygon.use_smooth = False
    if bevel > 0.0:
        add_bevel(obj, bevel, 2)
    return obj
