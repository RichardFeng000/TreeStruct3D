def add_beveled_box(name, location, dimensions, bevel_width, bevel_segments=5):
    bpy.ops.mesh.primitive_cube_add(location=location)
    obj = bpy.context.object
    obj.name = name
    obj.dimensions = dimensions
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)

    bevel = obj.modifiers.new("Soft edge bevel", 'BEVEL')
    bevel.width = bevel_width
    bevel.segments = bevel_segments
    bevel.limit_method = 'ANGLE'
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier=bevel.name)

    for polygon in obj.data.polygons:
        polygon.use_smooth = True
    obj.data.materials.append(material)
    parts.append(obj)
    return obj

def create_hanging_ring():
    segment_count = 64
    outer_radius = 0.49
    inner_radius = 0.19
    center_y = 4.68
    half_depth = 0.17

    verts = []
    for z in (-half_depth, half_depth):
        for radius in (outer_radius, inner_radius):
            for i in range(segment_count):
                angle = 2.0 * math.pi * i / segment_count
                verts.append((
                    radius * math.cos(angle),
                    center_y + radius * math.sin(angle),
                    z
                ))

    outer_bottom = 0
    inner_bottom = segment_count
    outer_top = segment_count * 2
    inner_top = segment_count * 3

    faces = []
    for i in range(segment_count):
        j = (i + 1) % segment_count

        # Bottom annular face.
        faces.append((
            outer_bottom + j,
            outer_bottom + i,
            inner_bottom + i,
            inner_bottom + j
        ))
        # Top annular face.
        faces.append((
            outer_top + i,
            outer_top + j,
            inner_top + j,
            inner_top + i
        ))
        # Outer cylindrical wall.
        faces.append((
            outer_bottom + i,
            outer_bottom + j,
            outer_top + j,
            outer_top + i
        ))
        # Inner wall of the hanging hole.
        faces.append((
            inner_bottom + j,
            inner_bottom + i,
            inner_top + i,
            inner_top + j
        ))

    mesh = bpy.data.meshes.new("Circular Hanging Hole Mesh")
    mesh.from_pydata(verts, [], faces)
    mesh.update()
    ring = bpy.data.objects.new("Circular Hanging Hole End", mesh)
    bpy.context.collection.objects.link(ring)

    bevel = ring.modifiers.new("Rounded ring edges", 'BEVEL')
    bevel.width = 0.045
    bevel.segments = 4
    bevel.limit_method = 'ANGLE'
    bpy.context.view_layer.objects.active = ring
    bpy.ops.object.modifier_apply(modifier=bevel.name)

    for polygon in ring.data.polygons:
        polygon.use_smooth = True
    ring.data.materials.append(material)
    parts.append(ring)

create_hanging_ring()
