def make_material(name, color, roughness=0.38):
    mat = bpy.data.materials.new(name)
    mat.diffuse_color = (*color, 1.0)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs["Base Color"].default_value = (*color, 1.0)
        bsdf.inputs["Roughness"].default_value = roughness
        bsdf.inputs["Metallic"].default_value = 0.0
    return mat

def stripe_index(phi, latitude, extra=0.0):
    phase = (
        7.0 * phi
        + 1.15 * math.sin(2.3 * latitude + 0.4)
        + 0.55 * math.sin(5.1 * latitude - 0.7)
        + 0.20 * math.sin(3.0 * phi + 4.0 * latitude)
        + extra
    )
    wave = math.sin(phase)
    fine = math.sin(2.0 * phase + 0.8 * math.sin(7.0 * latitude))
    if wave > 0.76:
        return 3
    if wave > 0.25:
        return 2
    if wave > -0.05 or fine > 0.91:
        return 1
    return 0

def add_body():
    radial_segments = 144
    vertical_segments = 72
    start = len(vertices)

    for j in range(vertical_segments + 1):
        t = j / vertical_segments
        z = -1.20 + 2.18 * t
        q = (z + 0.10) / 1.30

        base_radius = 1.04 * math.sqrt(max(0.0, 1.0 - q * q))
        lower_fill = 0.16 * math.exp(-((z + 0.90) / 0.34) ** 2)
        shoulder = 0.24 * math.exp(-((z - 0.63) / 0.38) ** 2)
        radius = base_radius + lower_fill + shoulder

        if j == 0 or j == vertical_segments:
            radius = 0.018

        for i in range(radial_segments):
            phi = 2.0 * math.pi * i / radial_segments
            asymmetry = 1.0 + 0.055 * math.cos(phi - 0.3)
            ripple = 1.0 + 0.009 * math.sin(18.0 * phi + 5.0 * z)
            r = radius * asymmetry * ripple

            x = 1.02 * r * math.cos(phi)
            y = 0.82 * r * math.sin(phi)
            x += 0.08 * (1.0 - t)
            vertices.append((x, y, z))

    for j in range(vertical_segments):
        row_a = start + j * radial_segments
        row_b = row_a + radial_segments
        z_mid = -1.20 + 2.18 * (j + 0.5) / vertical_segments
        latitude = math.asin(max(-1.0, min(1.0, (z_mid + 0.10) / 1.30)))

        for i in range(radial_segments):
            ni = (i + 1) % radial_segments
            faces.append((row_a + i, row_a + ni, row_b + ni, row_b + i))
            phi = 2.0 * math.pi * (i + 0.5) / radial_segments
            face_materials.append(stripe_index(phi, latitude))

def add_squashed_whorl(z, radius, height, phase_offset):
    rings = 24
    segments = 128
    start = len(vertices)

    for j in range(rings + 1):
        theta = math.pi * j / rings
        ring_radius = radius * math.sin(theta)
        ring_z = z + 0.5 * height * math.cos(theta)

        for i in range(segments):
            phi = 2.0 * math.pi * i / segments
            waviness = 1.0 + 0.012 * math.sin(11.0 * phi + phase_offset)
            x = ring_radius * waviness * math.cos(phi)
            y = 0.82 * ring_radius * waviness * math.sin(phi)
            vertices.append((x, y, ring_z))

    for j in range(rings):
        row_a = start + j * segments
        row_b = row_a + segments
        latitude = math.pi * (j + 0.5) / rings - math.pi * 0.5

def aperture_center(angle):
    cx = -0.27
    cz = -0.16
    a = 0.34
    b = 0.88
    taper = 0.84 + 0.16 * (1.0 - math.sin(angle))
    x = cx + a * taper * math.cos(angle)
    z = cz + b * math.sin(angle)
    y = -0.825 - 0.025 * math.cos(angle)
    return Vector((x, y, z))

def add_aperture_disk():
    segments = 128
    radial = 22
    start = len(vertices)
    center = Vector((-0.27, -0.838, -0.16))

    for j in range(radial + 1):
        r = j / radial
        for i in range(segments):
            angle = 2.0 * math.pi * i / segments
            edge = aperture_center(angle)
            point = center.lerp(edge, r)
            point.y -= 0.018 * (1.0 - r * r)
            vertices.append(tuple(point))

    for j in range(radial):
        row_a = start + j * segments
        row_b = row_a + segments
        for i in range(segments):
            ni = (i + 1) % segments
            faces.append((row_a + i, row_b + i, row_b + ni, row_a + ni))
            face_materials.append(5)

def add_aperture_lip():
    segments = 144
    tube_segments = 14
    start = len(vertices)
    tube_radius = 0.065

    for i in range(segments):
        angle = 2.0 * math.pi * i / segments
        center = aperture_center(angle)

        eps = 0.001
        before = aperture_center(angle - eps)
        after = aperture_center(angle + eps)
        tangent = (after - before).normalized()

        outward = Vector((
            center.x + 0.27,
            0.0,
            (center.z + 0.16) * 0.39
        ))
        if outward.length < 0.001:
            outward = Vector((1.0, 0.0, 0.0))
        outward.normalize()

        front = tangent.cross(outward)
        if front.length < 0.001:
            front = Vector((0.0, -1.0, 0.0))
        front.normalize()
        if front.y > 0.0:
            front.negate()

        for k in range(tube_segments):
            beta = 2.0 * math.pi * k / tube_segments
            width = tube_radius * (1.0 + 0.12 * math.sin(3.0 * angle))
            point = center + outward * (width * math.cos(beta)) + front * (width * math.sin(beta))
            vertices.append(tuple(point))

    for i in range(segments):
        ni = (i + 1) % segments
        row_a = start + i * tube_segments
        row_b = start + ni * tube_segments
        for k in range(tube_segments):
            nk = (k + 1) % tube_segments
            faces.append((row_a + k, row_b + k, row_b + nk, row_a + nk))
            if math.sin(7.0 * angle + 0.5 * math.sin(3.0 * angle)) > 0.55:
                face_materials.append(2)
            else:
                face_materials.append(4)
