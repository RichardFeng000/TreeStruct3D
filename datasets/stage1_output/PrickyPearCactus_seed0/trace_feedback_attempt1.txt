import bpy
import bmesh
import math
import random
from mathutils import Vector, Matrix

def clear_scene():
    """Clear the default Blender scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_oval_pad(bm, center, scale, rotation):
    """
    Creates a flattened oval pad segment within the provided bmesh.
    The pad is basically a scaled UV sphere.
    """
    # Create a temporary mesh to get the UV sphere geometry, then merge into bm
    tmp_mesh = bpy.data.meshes.new("tmp")
    bm_tmp = bmesh.new()
    bmesh.ops.create_uvsphere(bm_tmp, u_segments=16, v_segments=12)
    
    # Scale to make it a flat oval (Prickly Pear style)
    # X: width, Y: thickness, Z: height
    for v in bm_tmp.verts:
        v.co.x *= scale[0]
        v.co.y *= scale[1]
        v.co.z *= scale[2]
        
        # Apply rotation and translation
        rot_mat = Matrix.Rotation(rotation[0], 4, 'X') @ \
                  Matrix.Rotation(rotation[1], 4, 'Y') @ \
                  Matrix.Rotation(rotation[2], 4, 'Z')
        v.co = (rot_mat @ v.co) + center

    # Transfer to main bmesh
    bm.from_mesh(tmp_mesh)
    bm_tmp.free()
    tmp_mesh.user_clear()
    return bm

def create_spine(bm, start_pos, normal, length):
    """Creates a thin cone-like spine at the given position."""
    # Create a small cylinder/cone for the spine
    # To keep poly count manageable, we use a very simple 4-sided prism
    radius = 0.01
    segments = 4
    
    # Create base vertices
    base_verts = []
    for i in range(segments):
        angle = (2 * math.pi / segments) * i
        # Construct a perpendicular vector for the radius
        perp = Vector((0, 0, 1)) if abs(normal.z) < 0.9 else Vector((0, 1, 0))
        side_a = normal.cross(perp).normalized()
        side_b = normal.cross(side_a).normalized()
        
        offset = (side_a * math.cos(angle) + side_b * math.sin(angle)) * radius
        base_verts.append(bm.verts.new(start_pos + offset))

    # Create tip vertex
    tip = bm.verts.new(start_pos + normal * length)

    # Create faces for the spine
    for i in range(segments):
        bm.faces.new((base_verts[i], base_verts[(i + 1) % segments], tip))

def generate_cactus():
    clear_scene()
    
    # Main BMesh container
    main_mesh = bpy.data.meshes.new("PricklyPear")
    main_obj = bpy.data.objects.new("PricklyPear", main_mesh)
    bpy.context.collection.objects.link(main_obj)
    bm = bmesh.new()

    # Parameters for the cactus growth
    stack_depth = 5 # How many layers of pads
    base_scale = Vector((1.0, 0.3, 1.2)) # Width, Thickness, Height
    
    # To keep track of where new pads can grow from
    # Each entry: (position, rotation, current_scale)
    growth_points = [(Vector((0, 0, 0)), Vector((0, 0, 0)), base_scale)]
    
    # Store all pad surfaces to add spines later
    all_pads_verts = []

    for depth in range(stack_depth):
        next_growth_points = []
        current_points = growth_points[:] # Copy list
        
        # Tapering scale factor for higher pads
        taper = 0.75 ** depth
        layer_scale = Vector((base_scale.x * taper, base_scale.y * taper, base_scale.z * taper))
        
        for pos, rot, scale in current_points:
            # Create a pad
            # Randomize rotation slightly for organic look
            rand_rot = Vector((
                rot.x + random.uniform(-0.4, 0.4),
                rot.y + random.uniform(-0.4, 0.4),
                rot.z + random.uniform(-0.6, 0.6)
            ))
            
            # We use a helper to add the oval geometry
            create_oval_pad(bm, pos, layer_scale, rand_rot)
            
            # Determine where children pads emerge (top and sides)
            # The "top" of the current pad is roughly in the +Z direction relative to its rotation
            local_up = Vector((0, 0, 1))
            rot_mat = Matrix.Rotation(rand_rot.x, 4, 'X') @ \
                      Matrix.Rotation(rand_rot.y, 4, 'Y') @ \
                      Matrix.Rotation(rand_rot.z, 4, 'Z')
            world_up = (rot_mat @ local_up).to_3d()
            
            # Offset for the next pad's center
            # The current pad height is layer_scale.z * 2 (since it's centered at origin)
            # But since we placed it at 'pos', its top is pos + world_up * layer_scale.z
            top_center = pos + world_up * layer_scale.z
            
            # Branching factor: fewer branches as we go up
            num_branches = random.randint(1, 3) if depth < 2 else random.randint(0, 2)
            if depth == stack_depth - 1: num_branches = 0 # End the growth
                
            for i in range(num_branches):
                # Slight offset from center for organic placement
                offset_dir = Vector((random.uniform(-1, 1), random.uniform(-1, 1), 0)).normalized()
                if abs(world_up.z) < 0.9: # Avoid aligning too perfectly with world Z
                    # Project offset onto the pad's plane
                    pass 
                
                child_pos = top_center + (offset_dir * (layer_scale.x * 0.3))
                next_growth_points.append((child_pos, rand_rot, layer_scale))

        growth_points = next_growth_points

    # Now that geometry is generated, we add spines based on the vertices
    # To avoid excessive poly count, we don't put a spine on every single vertex
    # We sample vertices and place clusters of spines
    bm.verts.ensure_lookup_table()
    
    # Group vertices into approximate "areoles" (clusters)
    num_spines = 1200 # Total spines across the whole cactus
    for _ in range(num_spines):
        # Pick a random vertex
        v = random.choice(bm.verts)
        
        # Calculate normal at this vertex for spine direction
        # Since it's a scaled sphere, the vector from center is approx the normal
        # We find the closest "center" logic or just use a simple average of faces
        normal = Vector((0, 0, 0))
        for face in v.link_faces:
            normal += face.normal
        normal.normalize()
        
        # Vary spine length (some short bristles, some longer needles)
        if random.random() > 0.8:
            length = random.uniform(0.15, 0.3) # Long needle
        else:
            length = random.uniform(0.02, 0.08) # Short bristle
            
        create_spine(bm, v.co, normal, length)

    # Finalize mesh
    bm.to_mesh(main_mesh)
    bm.free()
    
    # Apply a simple smoothing if needed (optional for this style)
    # For the prickly pear look, we want it slightly rounded but distinct
    
    # Clean up: move everything to be centered at origin base
    # Calculate bounding box bottom
    bbox = main_obj.bound_box
    min_z = min([v[2] for v in bbox])
    main_obj.location.z = -min_z
