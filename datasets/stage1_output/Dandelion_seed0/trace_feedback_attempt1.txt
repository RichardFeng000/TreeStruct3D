import bpy
import bmesh
import math
import random
from mathutils import Vector

def clear_scene():
    """Clears the default Blender scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Creates a simple principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
    return mat

def create_stem(start, end, curvature=0.5):
    """Creates a slender, gently curving stem."""
    # Create a curve to define the path
    curve_data = bpy.data.curves.new('StemCurve', type='CURVE')
    curve_data.dimensions = '3D'
    curve_data.fill_mode = 'FULL'
    curve_data.bevel_depth = 0.025
    curve_data.bevel_resolution = 4

    polyline = curve_data.splines.new('BEZIER')
    
    # Midpoint for curvature
    mid = (start + end) / 2 + Vector((curvature, curvature * 0.5, 0))
    
    points = [start, mid, end]
    polyline.bezier_points.add(len(points) - 1)
    
    for i, p in enumerate(points):
        bp = polyline.bezier_points[i]
        bp.co = p
        # Set handles for smooth curve
        if i == 0:
            bp.handle_right = start + Vector((0, 0, 0.5))
        elif i == len(points) - 1:
            bp.handle_left = end + Vector((0, 0, -0.5))
        else:
            bp.handle_left = mid + Vector((-0.2, -0.2, -0.5))
            bp.handle_right = mid + Vector((0.2, 0.2, 0.5))

    stem_obj = bpy.data.objects.new('Stem', curve_data)
    bpy.context.collection.objects.link(stem_obj)
    
    # Convert to mesh for material consistency and potential modifiers
    bpy.context.view_layer.objects.active = stem_obj
    bpy.ops.object.convert(target='MESH')
    
    green_mat = create_material('StemGreen', (0.1, 0.3, 0.05, 1.0))
    stem_obj.data.materials.append(green_mat)
    return stem_obj

def create_puffball(center, radius, seed_count=180, scale=1.0, flattened=False):
    """Creates a spherical cluster of dandelion seeds."""
    mesh = bpy.data.meshes.new('PuffballMesh')
    obj = bpy.data.objects.new('Puffball', mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()
    
    # Material for the seeds
    white_mat = create_material('SeedWhite', (0.9, 0.9, 0.9, 1.0))

    for _ in range(seed_count):
        # Uniformly distribute points on a sphere
        phi = random.uniform(0, 2 * math.pi)
        cos_theta = random.uniform(-1, 1)
        sin_theta = math.sqrt(1 - cos_theta**2)
        
        dir_vec = Vector((
            sin_theta * math.cos(phi),
            sin_theta * math.sin(phi),
            cos_theta
        ))
        
        # Start point at center, end point at radius
        start_p = center
        end_p = center + dir_vec * (radius * scale)
        
        if flattened:
            # Squash the seed position slightly if it's fallen
            end_p.z *= 0.6
            
        # 1. Create the central stalk of the seed (the style/beak)
        # We use a thin cylinder/line
        segments = 4
        prev_v = bm.verts.new(start_p)
        for i in range(1, segments + 1):
            curr_v = bm.verts.new(start_p + (end_p - start_p) * (i / segments))
            bm.edges.new((prev_v, curr_v))
            prev_v = curr_v
        
        # 2. Create the "parachute" hairs (the pappus) at the end
        seed_tip = prev_v
        hair_count = 8
        hair_length = radius * 0.25 * scale
        
        for j in range(hair_count):
            # Hairs radiate outwards from the tip, generally following the dir_vec
            # but spreading out like an umbrella
            angle = (2 * math.pi / hair_count) * j
            # Create a local coordinate system for the tip to spread hairs
            ortho_v1 = Vector((0, 1, 0)).cross(dir_vec).normalized() if abs(dir_vec.x) < 0.9 else Vector((1, 0, 0)).cross(dir_vec).normalized()
            ortho_v2 = dir_vec.cross(ortho_v1).normalized()
            
            hair_dir = (dir_vec + ortho_v1 * math.cos(angle) * 0.5 + ortho_v2 * math.sin(angle) * 0.5).normalized()
            hair_end = end_p + hair_dir * hair_length
            
            # Add the hair edge
            hair_v = bm.verts.new(hair_end)
            bm.edges.new((seed_tip, hair_v))

    bm.to_mesh(mesh)
    bm.free()
    
    obj.data.materials.append(white_mat)
    
    # To make the "wispy" look better without massive polycounts, 
    # we can apply a skin modifier or just leave them as thin lines (edges).
    # However, since these are edges, they won't render unless converted to volumes.
    # We'll use a simple trick: convert the puffball to curves and then bevel them.
    return obj

def finalize_puffball(obj):
    """Converts edge-based mesh to beveled curves for rendering thickness."""
    bpy.context.view_layer.objects.active = obj
    # Convert Mesh -> Curve
    bpy.ops.object.convert(target='CURVE')
    
    curve_obj = bpy.context.active_object
    curve_data = curve_obj.data
    curve_data.bevel_depth = 0.005
    curve_data.bevel_resolution = 2
    curve_data.fill_mode = 'FULL'

def main():
    clear_scene()
    
    # Parameters
    stem_start = Vector((0, 0, 0))
    stem_end = Vector((0.1, -0.1, 4.5)) # Slightly elevated/curved end point
    puffball_radius = 0.8
    
    # Create the main stem
    stem = create_stem(stem_start, stem_end, curvature=0.6)
    
    # Create the main head
    head = create_puffball(stem_end, puffball_radius, seed_count=250)
    finalize_puffball(head)
    
    # Create a fallen puffball at the base
    # Shifted slightly from center, scaled down, and flattened
    fallen_center = Vector((0.4, 0.3, 0))
    fallen_radius = puffball_radius * 0.7
    fallen = create_puffball(fallen_center, fallen_radius, seed_count=120, scale=0.8, flattened=True)
    finalize_puffball(fallen)
    
    # Final adjustment to rotation of the fallen one so it looks like it landed naturally
    fallen.rotation_euler = (random.uniform(0, 0.5), random.uniform(0, 0.5), random.uniform(0, 6.28))

if __name__ == "__main__":
    main()
