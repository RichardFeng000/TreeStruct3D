def create_material(name, color):
    """Creates a simple principled BSDF material with a given color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    node_principled = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    node_principled.inputs['Base Color'].default_value = color
    mat.node_tree.links.new(node_principled.outputs['BSDF'], node_output.inputs['Surface'])
    return mat

def create_branch_segment(bm, start, end, radius_start, radius_end):
    """Creates a tapered cylinder segment between two points."""
    # Calculate direction and length
    direction = (end - start).normalized()
    length = (end - start).length
    
    # Create a cone/cylinder centered at the midpoint
    midpoint = (start + end) / 2
    
    # Setup rotation matrix to align Z-axis with direction
    up = Vector((0, 0, 1))
    if abs(direction.dot(up)) > 0.99:
        up = Vector((0, 1, 0))
    
    rot_quat = direction.to_track_quat('Z', 'Y')
    mat_rot = Vector((0,0,0)) # Translation handled by midpoint
    
    # We use create_cone for tapered cylinders
    # Note: bmesh.ops.create_cone creates it along Z axis
    res = bmesh.ops.create_cone(
        bm, 
        cap_ends=True, 
        segments=8, 
        radius1=radius_start, 
        radius2=radius_end, 
        depth=length
    )
    
    # Move and rotate the newly created vertices
    verts = res['verts']
    for v in verts:
        v.co = (rot_quat @ v.co) + midpoint

def generate_dead_tree():

    # Create a new mesh and object
    mesh = bpy.data.meshes.new("DeadTree")
    obj = bpy.data.objects.new("DeadTree", mesh)

    bm = bmesh.new()
    
    # Root parameters
    root_start = Vector((0, 0, 0))
    root_dir = Vector((0, 0, 1))

    trunk_radius = 0.6

    # Create the main trunk first as a few segments for better irregularity
    current_pos = root_start
    current_dir = root_dir
    current_rad = trunk_radius
    segments = 3
    for i in range(segments):

        # Jitter the trunk a bit
        jitter_dir = current_dir + Vector((random.uniform(-0.1, 0.1), random.uniform(-0.1, 0.1), 0))
