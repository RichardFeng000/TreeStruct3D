import bpy
import bmesh
import math

def clear_scene():
    """Removes all default objects from the scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_book():
    # Parameters (in meters)
    width = 0.16  # X
    height = 0.24 # Y
    thickness = 0.035 # Z (Page block thickness)
    cover_thick = 0.003
    overhang = 0.002 # How much the cover extends past pages
    spine_width = cover_thick * 3.0
    
    # 1. Create Page Block
    # The page block is a slightly smaller rectangle than the overall book
    page_w = width - spine_width
    page_h = height - (overhang * 2)
    page_t = thickness
    
    bpy.ops.mesh.primitive_cube_add(size=1.0, location=(page_w/2, page_h/2, page_t/2 + cover_thick))
    pages = bpy.context.active_object
    pages.name = "PageBlock"
    pages.scale = (page_w, page_h, page_t)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    
    # Give pages a slight bevel to avoid razor edges
    bev_pages = pages.modifiers.new(name="Bevel", type='BEVEL')
    bev_pages.width = 0.001
    bev_pages.segments = 2

    # 2. Create Hardcover
    # We'll construct the cover as one contiguous piece (Back -> Spine -> Front)
    bm = bmesh.new()
    
    # Define corners for the back cover
    # Back cover sits at Z=0
    v1 = bm.verts.new((0, 0, 0))
    v2 = bm.verts.new((width + spine_width, 0, 0))
    v3 = bm.verts.new((width + spine_width, height + (overhang*2), 0))
    v4 = bm.verts.new((0, height + (overhang*2), 0))
    
    bm.faces.new((v1, v2, v3, v4))
    
    # Extrude back cover thickness
    res = bmesh.ops.extrude_face_region(bm, geom=[bm.faces[0]])
    verts_top = [v for v in res['geom'] if isinstance(v, bmesh.types.BMVert)]
    # Move the extruded verts up (though we want them to be the back cover thickness)
    # Since it's lying flat, the "bottom" is Z=0, and thickness goes 'up'.
    for v in verts_top:
        v.co.z += cover_thick

    # Now let's build a more robust shell for the cover
    # I will reset the BM and create a solid volume to avoid manifold issues
    bm.free()
    bm = bmesh.new()
    
    # Dimensions of the total outer hull (including covers)
    outer_w = width + spine_width
    outer_h = height + (overhang * 2)
    outer_t = thickness + (cover_thick * 2)
    
    # Create a base cube for the hardcover shell
    bmesh.ops.create_cube(bm, size=1.0)
    # Scale and move to match book dimensions
    for v in bm.verts:
        v.co.x = (v.co.x + 0.5) * outer_w
        v.co.y = (v.co.y + 0.5) * outer_h
        v.co.z = (v.co.z + 0.5) * outer_t
        
    # Now we "hollow out" the page area. 
    # We want to remove the volume where pages go, but keep the spine and covers.
    # Page block relative coords: x in [spine_width, outer_w], y in [overhang, outer_h - overhang]
    # Z in [cover_thick, outer_t - cover_thick]
    
    # Instead of a boolean, let's manually define the "shell" for higher fidelity.
    bm.free()
    bm = bmesh.new()
    
    # Create vertices for 3 main slabs: Back, Spine, Front
    def create_slab(ox, oy, oz, sx, sy, sz):
        v0 = bm.verts.new((ox, oy, oz))
        v1 = bm.verts.new((ox + sx, oy, oz))
        v2 = bm.verts.new((ox + sx, oy + sy, oz))
        v3 = bm.verts.new((ox, oy + sy, oz))
        v4 = bm.verts.new((ox, oy, oz + sz))
        v5 = bm.verts.new((ox + sx, oy, oz + sz))
        v6 = bm.verts.new((ox + sx, oy + sy, oz + sz))
        v7 = bm.verts.new((ox, oy + sy, oz + sz))
        bm.faces.new((v0, v1, v2, v3)) # bottom
        bm.faces.new((v4, v7, v6, v5)) # top
        bm.faces.new((v0, v1, v5, v4)) # front
        bm.faces.new((v1, v2, v6, v5)) # right
        bm.faces.new((v2, v3, v7, v6)) # back
        bm.faces.new((v3, v0, v4, v7)) # left

    # Back Cover (bottom)
    create_slab(0, overhang, 0, outer_w, outer_h - (overhang*2), cover_thick)
    # Front Cover (top)
    create_slab(0, overhang, outer_t - cover_thick, outer_w, outer_h - (overhang*2), cover_thick)
    # Spine (left edge)
    create_slab(0, overhang, cover_thick, spine_width, outer_h - (overhang*2), thickness)

    # Clean up duplicate vertices and merge components
    bmesh.ops.remove_doubles(bm, verts=set(bm.verts), dist=0.001)
    
    cover_obj = bpy.data.objects.new("HardCover", bm)
    bpy.context.collection.objects.link(cover_obj)
    bm.free()

    # Apply Bevel to Cover for realistic rounded edges
    bev_cover = cover_obj.modifiers.new(name="Bevel", type='BEVEL')
    bev_cover.width = 0.002
    bev_cover.segments = 3
    bev_cover.limit_method = 'ANGLE'
    bev_cover.angle_limit = math.radians(30)

    # Add spine details (ribs/bands)
    # We create small cylinders or boxes along the spine to represent traditional binding ribs
    rib_count = 4
    rib_width = 0.005
    for i in range(rib_count):
        y_pos = overhang + (outer_h - (overhang*2)) * (i + 1) / (rib_count + 1)
        bpy.ops.mesh.primitive_cube_add(size=1.0, location=(spine_width/2, y_pos, outer_t/2))
        rib = bpy.context.active_object
        rib.name = f"SpineRib_{i}"
        # Scale rib to be a thin band wrapping the spine
        # X: slightly wider than spine, Y: rib_width, Z: height of cover shell
        rib.scale = (spine_width * 1.05, rib_width, outer_t)
        bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
        
        # Join rib to cover
        rib.select_set(True)
        cover_obj.select_set(True)
        bpy.context.view_layer.objects.active = cover_obj
        bpy.ops.object.join()
        cover_obj.select_set(False)

    # Final touch: a subtle "dip" at the hinges
    # For this procedural script, we'll use a bevel on the interior edges of the spine
    # but since we joined everything, we'll just ensure the outer bevel is clean.
    
    # Move the whole assembly slightly so it sits centered if needed 
    # (currently it starts at origin and extends positive X, Y, Z)
    cover_obj.location = (-outer_w/2, -outer_h/2, 0)
    pages.location = (page_w/2 - outer_w/2, page_h/2 - outer_h/2, cover_thick)

if __name__ == "__main__":
    clear_scene()
    create_book()
