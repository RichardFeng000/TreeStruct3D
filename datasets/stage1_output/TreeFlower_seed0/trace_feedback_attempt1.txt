import bpy
import bmesh
import math
import random
from mathutils import Vector

def clear_scene():
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
    return mat

def create_petal(name, material):
    # Create a single petal mesh
    mesh = bpy.data.meshes.new(name)
    obj = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()
    
    # Petal dimensions: elongated and flat
    # Length 3.0, Width 0.4, Thickness 0.02
    length = 3.0
    width = 0.4
    thickness = 0.02
    segments = 10
    
    # Create vertices for a petal-like shape (tapered at ends)
    verts = []
    for i in range(segments + 1):
        t = i / segments
        # x: distance from center
        x = t * length
        # y: width varies based on a sine curve for an organic look
        w = math.sin(t * math.pi) * width
        
        verts.append(bm.verts.new((x, -w/2, 0)))
        verts.append(bm.verts.new((x, w/2, 0)))

    # Create faces
    for i in range(segments):
        v1 = verts[i*2]
        v2 = verts[i*2 + 1]
        v3 = verts[(i+1)*2 + 1]
        v4 = verts[(i+1)*2]
        bm.faces.new((v1, v2, v3, v4))

    # Add a tiny bit of thickness (extrude)
    # We can just use a solidify modifier later or extrude manually
    # For high fidelity without too many vertices, we'll do manual extrusion
    bm.select_all(True)
    res = bmesh.ops.extrude_face_region(bm, geom=bm.faces[:])
    for v in res['geom'].values():
        if isinstance(v, bmesh.types.BMVert):
            v.co.z += thickness

    bm.to_mesh(mesh)
    bm.free()
    
    obj.data.materials.append(material)
    return obj

def create_central_disk():
    # Dome shape central disk
    bpy.ops.mesh.primitive_uv_sphere_add(radius=1.0, segments=32, ring_count=16)
    disk = bpy.context.active_object
    disk.name = "CentralDisk"
    
    # Cut the sphere in half to make a dome
    bm = bmesh.new()
    bm.from_mesh(disk.data)
    
    # Delete vertices below z=0
    verts_to_del = [v for v in bm.verts if v.co.z < -0.01]
    bmesh.ops.delete(bm, geom=verts_to_del, context='VERTS')
    
    bm.to_mesh(disk.data)
    bm.free()
    
    # Scale to be a flatter dome
    disk.scale = (1.0, 1.0, 0.4)
    bpy.ops.object.transform_apply(scale=True)
    
    return disk

def populate_florets(disk_obj, material):
    # Create many small cylinders on the surface of the dome
    num_florets = 400
    floret_radius = 0.03
    floret_height = 0.12
    
    # Use a Fibonacci spiral distribution for "densely packed" organic look
    # We distribute them on the surface of the hemisphere
    phi = math.pi * (3. - 1.) # golden angle in radians
    
    for i in range(num_florets):
        y = 1 - (i / float(num_florets - 1)) * 2  # y goes from 1 to -1
        radius_at_y = math.sqrt(1 - y * y) # radius of circle at height y
        
        theta = phi * i
        x = math.cos(theta) * radius_at_y
        z = math.sin(theta) * radius_at_y
        # Since our dome is oriented on Z, we map (x, z) to X and Y and y to Z
        pos = Vector((x, z, y)) # this gives a unit sphere coord
        
        # The disk was scaled by 0.4 in Z axis
        pos.z *= 0.4
        
        # Create floret
        bpy.ops.mesh.primitive_cylinder_add(radius=floret_radius, depth=floret_height, location=pos)
        floret = bpy.context.active_object
        
        # Align cylinder to normal of the dome
        # Normal at pos is just the unit vector from origin (roughly)
        normal = Vector((x, z, y * 2.5)).normalized() # adjusted for flatten
        rotation_quat = Vector((0, 0, 1)).rotation_difference(normal)
        floret.rotation_mode = 'QUATERNION'
        floret.rotation_quaternion = rotation_quat
        
        # Move floret so the base is on the surface
        floret.location += normal * (floret_height / 2)
        
        floret.data.materials.append(material)

def main():
    clear_scene()
    
    # Materials
    mat_white = create_material("WhitePetal", (1.0, 1.0, 1.0, 1.0))
    mat_tan = create_material("TanFloret", (0.8, 0.6, 0.4, 1.0))
    
    # Central Disk
    disk = create_central_disk()
    disk.data.materials.append(mat_tan)
    
    # Florets on the disk
    populate_florets(disk, mat_tan)
    
    # Petals radiating outward
    num_petals = 36
    petal_proto = create_petal("PetalProto", mat_white)
    
    # Hide original prototype
    petal_proto.hide_viewport = True
    petal_proto.hide_render = True
    
    for i in range(num_petals):
        angle = (2 * math.pi / num_petals) * i
        
        # Create an instance/copy of the petal
        new_petal = petal_proto.copy()
        new_petal.data = petal_proto.data.copy()
        bpy.context.collection.objects.link(new_petal)
        
        # Rotation around Z axis
        new_petal.rotation_euler[2] = angle
        
        # Offset the petal so it starts from the edge of the disk (radius 1.0)
        # The petal's geometry is centered at x=0 to length, so we shift by 1.0
        cos_a = math.cos(angle)
        sin_a = math.sin(angle)
        new_petal.location = (cos_a * 1.0, sin_a * 1.0, 0)
        
        # To make them radiate from center, we rotate the object around world origin
        # But since we already set location based on angle and rotation relative to its own axis:
        # We need to ensure the petal points "outward"
        new_petal.rotation_euler[2] = angle
    
    # Final polish: remove prototype from scene if necessary (already hidden)
    bpy.data.objects.remove(petal_proto, do_unlink=True)

if __name__ == "__main__":
    main()
