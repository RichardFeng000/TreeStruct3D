import bpy
import bmesh
import math

def clear_scene():
    """Clears the default scene objects."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Creates a simple diffuse material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    # Get the Principled BSDF node
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
    return mat

def create_box(name, w, h, d, location, material=None):
    """Helper to create a scaled cube as a box."""
    bpy.ops.mesh.primitive_cube_add(size=1.0, location=location)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (w, h, d)
    bpy.ops.object.apply(type='SCALE')
    if material:
        obj.data.materials.append(material)
    return obj

def main():
    clear_scene()

    # Dimensions (in meters)
    WIDTH = 0.6  # External width
    HEIGHT = 1.2 # External height
    DEPTH = 0.3  # External depth
    THICKNESS = 0.02  # Panel thickness
    SHELF_COUNT = 2

    # Material
    white_mat = create_material("BookcaseWhite", (0.9, 0.9, 0.9, 1.0))

    # --- Construction Logic ---
    # We'll use the Y axis as Height, X as Width, Z as Depth for a standard view
    
    # Side Panels (Left and Right) - Full height
    left_side = create_box("Side_L", THICKNESS, HEIGHT, DEPTH, 
                           (-WIDTH/2 + THICKNESS/2, 0, 0), white_mat)
    right_side = create_box("Side_R", THICKNESS, HEIGHT, DEPTH, 
                            (WIDTH/2 - THICKNESS/2, 0, 0), white_mat)

    # Top and Bottom Panels - Fit between the sides
    inner_width = WIDTH - (2 * THICKNESS)
    top_panel = create_box("Top", inner_width, THICKNESS, DEPTH, 
                           (0, HEIGHT/2 - THICKNESS/2, 0), white_mat)
    bottom_panel = create_box("Bottom", inner_width, THICKNESS, DEPTH, 
                              (0, -HEIGHT/2 + THICKNESS/2, 0), white_mat)

    # Back Panel - Thin sheet fitting inside the outer frame
    back_panel = create_box("Back", WIDTH - THICKNESS, HEIGHT - THICKNESS, THICKNESS/2, 
                            (0, 0, -DEPTH/2 + THICKNESS/4), white_mat)

    # Internal Shelves
    shelf_spacing = (HEIGHT - (2 * THICKNESS)) / (SHELF_COUNT + 1)
    for i in range(SHELF_COUNT):
        y_pos = -HEIGHT/2 + THICKNESS + (i + 1) * shelf_spacing
        create_box(f"Shelf_{i+1}", inner_width, THICKNESS, DEPTH - THICKNESS, 
                   (0, y_pos, THICKNESS/4), white_mat)

    # --- Mounting Holes ---
    # We'll create small cylinders to boolean subtract from the sides
    hole_radius = 0.008  # 8mm radius
    hole_depth = 0.015   # Slightly shallower than thickness or just enough to be visible
    
    # Hole positions relative to the side panel center (X, Y, Z)
    # We place holes near corners of the sides on the outer faces
    hole_offsets = [
        (0, HEIGHT/2 - 0.05, DEPTH/2 - 0.05), # Top Rear
        (0, HEIGHT/2 - 0.05, -DEPTH/2 + 0.05), # Top Front (though usually just rear)
        (0, -HEIGHT/2 + 0.05, DEPTH/2 - 0.05), # Bottom Rear
        (0, -HEIGHT/2 + 0.05, -DEPTH/2 + 0.05), # Bottom Front
    ]

    def add_holes_to_panel(panel, side_multiplier):
        for offset in hole_offsets:
            # Calculate world position for the cutter
            loc = (
                (panel.location[0]), 
                offset[1], 
                offset[2]
            )
            
            # Create a cylinder to act as the boolean cutter
            bpy.ops.mesh.primitive_cylinder_add(
                radius=hole_radius, 
                depth=THICKNESS * 2, 
                location=loc, 
                rotation=(0, math.radians(90), 0) # Rotate to face the panel side
            )
            cutter = bpy.context.active_object
            
            # Add Boolean Modifier
            bool_mod = panel.modifiers.new(name="MountHole", type='BOOLEAN')
            bool_mod.operation = 'DIFFERENCE'
            bool_mod.object = cutter
            
            # Apply modifier immediately to keep it clean
            bpy.context.view_layer.objects.active = panel
            bpy.ops.object.modifier_apply(modifier=bool_mod.name)
            
            # Remove the cutter object
            bpy.data.objects.remove(cutter, do_unlink=True)

    add_holes_to_panel(left_side, -1)
    add_holes_to_panel(right_side, 1)

    # --- Final Cleanup ---
    # Join all parts into a single object for the final output
    bpy.ops.object.select_all(action='DESELECT')
    for obj in bpy.data.objects:
        if "Side" in obj.name or "Top" in obj.name or "Bottom" in obj.name or "Back" in obj.name or "Shelf" in obj.name:
            obj.select_set(True)
    
    bpy.context.view_layer.objects.active = left_side
    bpy.ops.object.join()
    
    # Rename the final assembly
    final_obj = bpy.context.active_object
    final_obj.name = "CompactBookcase"

if __name__ == "__main__":
    main()
