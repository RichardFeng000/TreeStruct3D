import bpy
import bmesh
import math
import random
from mathutils import Vector

def clear_scene():
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    bsdf.inputs['Base Color'].default_value = color
    bsdf.inputs['Roughness'].default_value = 0.9  # Matte appearance
    return mat

def create_leather_coral():
    # Parameters for the coral base
    radius = 2.5
    lobes_count = 7
    base_thickness = 0.6
    polyps_density = 0.15 # Scale of high-freq noise
    
    mesh = bpy.data.meshes.new("LeatherCoral")
    obj = bpy.data.objects.new("LeatherCoral", mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()

    # 1. Generate the Base Shape (Broad fleshy lobes)
    radial_res = 80 # Higher resolution for better texture
    ring_res = 40
    verts_map = []
    
    for r_idx in range(ring_res + 1):
        r_factor = r_idx / ring_res
        current_ring = []
        for s_idx in range(radial_res):
            theta = (s_idx / radial_res) * 2 * math.pi
            
            # Organic lobe variation: broader and more irregular than a simple sine wave
            lobe_variation = (
                math.sin(lobes_count * theta) * 0.4 + 
                math.sin(lobes_count * 1.5 * theta) * 0.2 + 
                math.cos(3 * theta) * 0.1
            )
            # Variation increases toward the edge, but remains broad
            current_radius = radius * r_factor * (1 + lobe_variation * (r_factor**1.5))
            
            x = math.cos(theta) * current_radius
            y = math.sin(theta) * current_radius
            z = 0
            
            v = bm.verts.new(Vector((x, y, z)))
            current_ring.append(v)
        verts_map.append(current_ring)

    for r in range(ring_res):
        for s in range(radial_res):
            s_next = (s + 1) % radial_res
            bm.faces.new((verts_map[r][s], verts_map[r][s_next], verts_map[r+1][s_next], verts_map[r+1][s]))

    # 2. Extrude for thickness (Fleshy volume)
    bm.faces.ensure_lookup_table()
    all_faces = bm.faces[:]
    res = bmesh.ops.extrude_face_region(bm, geom=all_faces)
    extruded_verts = [v for v in res['geom'] if isinstance(v, bmesh.types.BMVert)]
    for v in extruded_verts:
        dist = math.sqrt(v.co.x**2 + v.co.y**2)
        # Thicker center, thinner edges
        thickness = base_thickness * (1 - (dist / radius) * 0.5)
        v.co.z -= max(0.1, thickness)

    # 3. Apply "Granular" surface and Fleshy curvature
    bm.verts.ensure_lookup_table()
    for v in bm.verts:
        dist = math.sqrt(v.co.x**2 + v.co.y**2)
        is_top = v.co.z >= -0.1
        
        if is_top:
            # Macro dome shape
            dome = (1 - (dist/radius)**2) * 0.5 if dist <= radius else 0
            
            # High-frequency granular noise for polyps
            # Sum of multiple sine waves to simulate random grit
            grain = (
                math.sin(v.co.x * 40) * math.cos(v.co.y * 40) * 0.04 +
                math.sin(v.co.x * 100) * math.sin(v.co.y * 100) * 0.02 +
                random.uniform(-0.01, 0.01)
            )
            v.co.z += dome + grain
        else:
            # Bottom surface slightly concave and rougher
            grain_bottom = math.sin(v.co.x * 20) * math.cos(v.co.y * 20) * 0.03
            v.co.z += grain_bottom

    # 4. Central Spire
    spire_segments = 20
    spire_height = 1.4
    spire_radius_base = 0.35
    spire_radius_top = 0.18
    
    spire_rings = []
    for h_idx in range(15):
        h_factor = h_idx / 14.0
        z = h_factor * spire_height
        r = spire_radius_base + (spire_radius_top - spire_radius_base) * h_factor
        # Slight lean for organic feel
        off_x = math.sin(h_factor * 1.5) * 0.15
        off_y = math.cos(h_factor * 1.2) * 0.15
        
        ring = []
        for s_idx in range(spire_segments):
            theta = (s_idx / spire_segments) * 2 * math.pi
            v = bm.verts.new(Vector((math.cos(theta)*r + off_x, math.sin(theta)*r + off_y, z)))
            ring.append(v)
        spire_rings.append(ring)

    for h in range(14):
        for s in range(spire_segments):
            s_next = (s + 1) % spire_segments
            bm.faces.new((spire_rings[h][s], spire_rings[h][s_next], spire_rings[h+1][s_next], spire_rings[h+1][s]))

    # Top of spire closure
    top_center = bm.verts.new(Vector((off_x, off_y, spire_height)))
    last_ring = spire_rings[-1]
    for s in range(spire_segments):
        s_next = (s + 1) % spire_segments
        bm.faces.new((last_ring[s], last_ring[s_next], top_center))

    # Texture the spire
    for v in bm.verts:
        if v.co.z > 0.4: # Affect only spire height
            v.co += Vector((random.uniform(-0.02, 0.02), random.uniform(-0.02, 0.02), random.uniform(-0.02, 0.02)))

    bm.to_mesh(mesh)
    bm.free()

    # Modifiers for organic look
    subsurf = obj.modifiers.new(name="Subdiv", type='SUBSURF')
    subsurf.levels = 1
    
    # Adding an extra Displace modifier for the "extremely rough" feel
    tex = bpy.data.textures.new("PolypsTex", type='NOISE')
    tex.noise_scale = 0.02
    disp = obj.modifiers.new(name="Roughness", type='DISPLACE')
    disp.texture = tex
    disp.strength = 0.04

    # Material: Sandy Brown with Olive Green tones
    # We create a mix by using vertex colors or just a neutral warm tone as base
    coral_mat = create_material("CoralMat", (0.6, 0.5, 0.3, 1.0)) # Warm sandy brown
    obj.data.materials.append(coral_mat)

    return obj

def main():
    clear_scene()
    coral = create_leather_coral()
    coral.location = (0, 0, 0)
    bpy.context.view_layer.objects.active = coral
    bpy.ops.object.shade_smooth()

if __name__ == "__main__":
    main()
