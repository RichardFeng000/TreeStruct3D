def clear_scene():
    """Clears the default scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color, metallic=0.0, roughness=0.5):
    """Creates a basic material with given parameters."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
        bsdf.inputs['Metallic'].default_value = metallic
        bsdf.inputs['Roughness'].default_value = roughness
    return mat

def create_square_tin():
    # Dimensions
    width = 2.0
    height = 2.5
    depth = 2.0
    corner_radius = 0.2
    wall_thickness = 0.1
    recess_depth = 0.15
    lid_offset = 0.02 # Small gap for realism

    # Create the main body mesh
    bm = bmesh.new()
    
    # Initial cube
    bmesh.ops.create_cube(bm, size=1.0)
    
    # Scale to desired dimensions
    # Default cube is size 1 centered at origin (-0.5 to 0.5)
    for v in bm.verts:
        v.co.x *= width
        v.co.y *= depth
        v.co.z *= height

    # Bevel vertical edges (edges parallel to Z axis)
    vertical_edges = [e for e in bm.edges if abs(e.verts[0].co.z - e.verts[1].co.z) > 0.9 * height]
    bmesh.ops.bevel(bm, geom=vertical_edges, offset=corner_radius, segments=8, affect='EDGES')

    # Create the top recess
    # Find the top face (highest center Z)
    top_face = None
    max_z = -float('inf')
    for f in bm.faces:
        center_z = sum(v.co.z for v in f.verts) / len(f.verts)
        if center_z > max_z:
            max_z = center_z
            top_face = f

    # Inset the top face to create the rim
    # bmesh.ops.inset creates new faces; we need to handle the result
    res = bmesh.ops.inset_individual(bm, faces=[top_face], thickness=0.1, use_boundary=True)
    inner_face = res['faces'][0]

    # Extrude the inner face downwards into the can for the recess
    bmesh.ops.translate(bm, verts=inner_face.verts, vec=(0, 0, -recess_depth))
    
    # To make it look like a hollow tin with a thickness:
    # We should technically extrude the whole thing or leave as is for "can shell"
    # For this model, we'll treat the body as a solid block with a recessed top hole.

    bm.to_mesh(bpy.data.meshes.new("TinBody"))
    body_obj = bpy.data.objects.new("TinBody", bpy.data.meshes.new("TinBody"))
    bpy.context.collection.objects.link(body_obj)
    bm.free()

    # Create the Lid
    # The lid is a slightly smaller box that fits into the recess
    lid_bm = bmesh.new()
    bmesh.ops.create_cube(lid_bm, size=1.0)
    
    # Scale lid to fit inside (accounting for inset and rounded corners)
    # The body's top is width 2, but we inset by 0.1 on each side
    l_width = width - 0.2
    l_depth = depth - 0.2
    l_height = recess_depth * 0.8 # Slightly thinner than the hole

    for v in lid_bm.verts:
        v.co.x *= l_width
        v.co.y *= l_depth
        v.co.z *= l_height

    # Bevel vertical edges of the lid to match body curvature
    l_vertical_edges = [e for e in lid_bm.edges if abs(e.verts[0].co.z - e.verts[1].co.z) > 0.5]
    bmesh.ops.bevel(lid_bm, geom=l_vertical_edges, offset=corner_radius - 0.1, segments=8, affect='EDGES')

    lid_mesh = bpy.data.meshes.new("TinLid")
    lid_bm.to_mesh(lid_mesh)

    # Body height is 2.5, so top surface is at 1.25. Recess goes down to 1.25 - recess_depth.
    # Lid center Z = 1.25 - (recess_depth / 2)
    lid_obj.location.z = (height / 2) - (recess_depth / 2)

    return body_obj, lid_obj
