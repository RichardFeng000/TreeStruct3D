import bpy
import bmesh
import math
import random

def clear_scene():
    """Clears all objects from the current scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_oval_cylinder(name, rx, ry, height, z_offset):
    """Creates a cylinder and scales it to form an oval disc."""
    # Create a base cylinder with radius 1
    bpy.ops.mesh.primitive_cylinder_add(
        vertices=128, 
        radius=1.0, 
        depth=height, 
        location=(0, 0, z_offset + height / 2)
    )
    obj = bpy.context.active_object
    obj.name = name
    
    # Scale to create oval shape
    obj.scale = (rx, ry, 1.0)
    
    # Apply scale so that subsequent modifiers or operations use the real dimensions
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    return obj

def create_ridge_ring(radius_factor, rx, ry, thickness, z_pos):
    """Creates a ring that follows the oval proportions of the discs."""
    # Use a torus to create a smooth rounded ridge
    # Major radius is the distance from center to the middle of the tube
    # Minor radius is the thickness of the ridge itself
    major_r = radius_factor 
    minor_r = thickness / 2.0
    
    bpy.ops.mesh.primitive_torus_add(
        align='WORLD', 
        location=(0, 0, z_pos), 
        major_radius=major_r, 
        minor_radius=minor_r, 
        absegments=64, 
        psegments=16
    )
    ring = bpy.context.active_object
    
    # Scale the ring to match the oval proportions of the main disc
    ring.scale = (rx, ry, 1.0)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    return ring

def main():
    clear_scene()

    # Parameters for the trinket
    # Base disc dimensions
    base_rx = 1.0
    base_ry = 0.8
    base_h = 0.2
    base_z = 0.0
    
    # Upper disc dimensions
    top_rx = 1.2
    top_ry = 1.0
    top_h = 0.3
    top_z = base_h # Sits exactly on top of the base
    
    # Create the charcoal base disc
    base_disc = create_oval_cylinder("BaseDisc", base_rx, base_ry, base_h, base_z)
    
    # Create the cream upper disc
    top_disc = create_oval_cylinder("TopDisc", top_rx, top_ry, top_h, top_z)
    
    # Generate the concentric circular ridge patterns on the top surface of the upper disc
    # The top face is at Z = top_z + top_h
    surface_z = top_z + top_h
    ridge_thickness = 0.04
    num_ridges = 12
    
    ridges = []
    for i in range(1, num_ridges + 1):
        # Vary the radius slightly to feel "nature-inspired" (like tree rings)
        jitter = random.uniform(-0.03, 0.03)
        # Distribute radii from center outwards up to just inside the edge of top_rx
        radius_factor = (i / num_ridges) * (top_rx * 0.9) + jitter
        
        # Create a ridge ring on the surface
        # We offset the Z by half thickness so it sits on top rather than embedded
        ring = create_ridge_ring(
            radius_factor, 
            top_rx, 
            top_ry, 
            ridge_thickness, 
            surface_z + (ridge_thickness / 2)
        )
        ridges.append(ring)

    # Join all ridges to the top disc for a single coherent object
    bpy.context.view_layer.objects.active = top_disc
    for ridge in ridges:
        ridge.select_set(True)
    
    bpy.ops.object.join()
    
    # Final polishing: ensure objects are centered and named clearly
    top_disc.name = "NatureTrinket_Top"
    base_disc.name = "NatureTrinket_Base"

if __name__ == "__main__":
    main()
