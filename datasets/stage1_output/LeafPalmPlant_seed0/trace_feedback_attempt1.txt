import bpy
import bmesh
import math
import random
from mathutils import Vector

def clear_scene():
    """Clears the default Blender scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_fan_palm():
    # Parameters
    stem_height = 8.0
    stem_radius_base = 0.15
    stem_radius_top = 0.1
    num_segments = 42
    segment_length_min = 3.0
    segment_length_max = 5.0
    segment_width_base = 0.12
    segment_width_tip = 0.02
    fan_spread_angle = 160 * (math.pi / 180) # radians
    curvature_factor = 0.5 # how much segments droop

    # Create a BMesh for the entire leaf assembly
    bm = bmesh.new()

    # --- Stem Construction ---
    # Simple tapered cylinder for the petiole
    segments_stem = 12
    rings_stem = 16
    
    # Base ring
    verts_base = []
    for i in range(segments_stem):
        angle = (2 * math.pi * i) / segments_stem
        x = math.cos(angle) * stem_radius_base
        y = math.sin(angle) * stem_radius_base
        verts_base.append(bm.verts.new(Vector((x, y, 0))))

    # Top ring
    verts_top = []
    for i in range(segments_stem):
        angle = (2 * math.pi * i) / segments_stem
        x = math.cos(angle) * stem_radius_top
        y = math.sin(angle) * stem_radius_top
        verts_top.append(bm.verts.new(Vector((x, y, stem_height))))

    # Connect rings
    for i in range(segments_stem):
        v1 = verts_base[i]
        v2 = verts_base[(i + 1) % segments_stem]
        v3 = verts_top[(i + 1) % segments_stem]
        v4 = verts_top[i]
        bm.faces.new((v1, v2, v3, v4))

    # --- Blade Construction (The Fan) ---
    center_point = Vector((0, 0, stem_height))
    
    # We create segments radiating from the center
    start_angle = -fan_spread_angle / 2
    end_angle = fan_spread_angle / 2
    
    for i in range(num_segments):
        # Angle for this specific segment
        angle = start_angle + (i / (num_segments - 1)) * (end_angle - start_angle)
        
        # Randomize length slightly for organic look
        length = random.uniform(segment_length_min, segment_length_max)
        if i == 0 or i == num_segments - 1:
            length *= 0.7 # shorter at the edges of the fan

        # Create a strip of vertices for one side of the leaf segment
        # We'll use a simple parametric curve to simulate the droop/bend
        num_points = 8
        seg_verts_left = []
        seg_verts_right = []
        
        for p in range(num_points + 1):
            t = p / num_points # normalized distance along segment [0, 1]
            
            # Basic radial direction
            dir_vec = Vector((math.cos(angle), math.sin(angle), 0))
            
            # Width interpolation (tapers towards tip)
            current_width = segment_width_base * (1.0 - t * 0.8)
            perp_vec = Vector((-dir_vec.y, dir_vec.x, 0)) # perpendicular to direction
            
            # Curvature / Droop logic
            # Z drops as t increases, but follows a parabolic curve
            z_offset = -curvature_factor * (t**2) * length * 0.5
            # X, Y extension
            pos_base = center_point + dir_vec * (t * length)
            pos_base.z += z_offset
            
            # Add slight noise/organic jitter to position
            jitter = Vector((random.uniform(-0.05, 0.05), random.uniform(-0.05, 0.05), random.uniform(-0.05, 0.05)))
            if t > 0: pos_base += jitter * t

            # Create the two edges of the strip
            v_l = bm.verts.new(pos_base + perp_vec * (current_width / 2))
            v_r = bm.verts.new(pos_base - perp_vec * (current_width / 2))
            seg_verts_left.append(v_l)
            seg_verts_right.append(v_r)

        # Create faces for the segment strip
        for p in range(num_points):
            bm.faces.new((
                seg_verts_left[p], 
                seg_verts_left[p+1], 
                seg_verts_right[p+1], 
                seg_verts_right[p]
            ))

    # Finalize BMesh to Object
    bm.to_mesh("FanPalmLeaf")
    bm.free()

    obj = bpy.data.objects.new("FanPalmLeaf", bpy.data.meshes["FanPalmLeaf"])
    bpy.context.collection.objects.link(obj)
    
    # Rotate the whole object slightly for a "natural" look (elevated perspective feel)
    obj.rotation_euler[0] = math.radians(-15) 
    
    return obj

def apply_modifiers(obj):
    """Adds smoothness to the organic geometry."""
    # Subdivision Surface for smoother edges
    subsurf = obj.modifiers.new(name="Subdiv", type='SUBSURF')
    subsurf.levels = 1
    subsurf.render_levels = 2

    # Smooth shading
    for poly in obj.data.polygons:
        poly.use_smooth = True

if __name__ == "__main__":
    clear_scene()
    palm_obj = create_fan_palm()
    apply_modifiers(palm_obj)
