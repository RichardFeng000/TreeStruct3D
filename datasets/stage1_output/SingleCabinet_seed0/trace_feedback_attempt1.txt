import bpy
import bmesh
import math

def clear_scene():
    """Removes default objects from the scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Creates a simple diffuse material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    node_principled = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    node_principled.inputs['Base Color'].default_value = color
    # Set roughness high for wood-like matte finish
    node_principled.inputs['Roughness'].default_value = 0.8
    mat.node_tree.links.new(node_principled.outputs['BSDF'], node_output.inputs['Surface'])
    return mat

def create_box(name, width, depth, height, location=(0, 0, 0), material=None):
    """Creates a beveled box mesh."""
    mesh = bpy.data.meshes.new(name)
    obj = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()
    # Create a cube with the desired dimensions
    bmesh.ops.create_cube(bm, size=1.0)
    
    # Scale to dimensions
    # Note: create_cube is centered at 0,0,0 and has size 1 (meaning -0.5 to 0.5)
    # We scale it by the target dims.
    for v in bm.verts:
        v.co.x *= width
        v.co.y *= depth
        v.co.z *= height
        
    bm.to_mesh(mesh)
    bm.free()
    
    obj.location = location
    
    # Add a bevel modifier for realism
    bevel = obj.modifiers.new(name="Bevel", type='BEVEL')
    bevel.width = 0.005
    bevel.segments = 3
    
    if material:
        obj.data.materials.append(material)
        
    return obj

def create_handle(parent_obj, door_width, door_height, thickness):
    """Creates a small handle for the cabinet door."""
    # Handle placement: right side of door, mid-height
    handle_w = 0.02
    handle_h = 0.08
    handle_d = 0.03 # protrudes out
    
    mesh = bpy.data.meshes.new("Handle")
    handle_obj = bpy.data.objects.new("Handle", mesh)
    bpy.context.collection.objects.link(handle_obj)
    
    bm = bmesh.new()
    bmesh.ops.create_cube(bm, size=1.0)
    for v in bm.verts:
        v.co.x *= handle_w
        v.co.y *= handle_d
        v.co.z *= handle_h
        
    bm.to_mesh(mesh)
    bm.free()
    
    # Positioning relative to the door (which is at origin or offset)
    # Door width is along X, height along Z, thickness along Y
    # The door panel we will create will be centered.
    # Position handle on the right side of the door face.
    handle_obj.location = (door_width/2 - 0.04, thickness/2 + handle_d/2, 0)
    
    # Material for handle (slightly different dark wood or metal)
    mat = create_material("HandleMat", (0.1, 0.08, 0.05, 1.0))
    handle_obj.data.materials.append(mat)
    
    # Parent to door for cohesion
    handle_obj.parent = parent_obj
    
    # Bevel the handle too
    bevel = handle_obj.modifiers.new(name="Bevel", type='BEVEL')
    bevel.width = 0.003
    bevel.segments = 2

def build_cabinet():
    clear_scene()
    
    # Dimensions
    cab_w = 0.6  # Width
    cab_d = 0.4  # Depth
    cab_h = 2.1  # Height
    wall_thickness = 0.02
    
    # Wood color (Dark wood tone)
    dark_wood_mat = create_material("DarkWood", (0.15, 0.08, 0.04, 1.0))
    
    # 1. Main Carcass
    # Instead of one cube, we build the frame for more detail if needed, 
    # but a slightly inset body works well here.
    body = create_box("CabinetBody", cab_w, cab_d, cab_h, location=(0, 0, cab_h/2), material=dark_wood_mat)
    
    # Use bmesh to "hollow out" the front for a cleaner look or just place door on top.
    # For a high-fidelity single-door cabinet, let's make the body as a shell.
    bm = bmesh.new()
    bm.from_object(body.data)
    
    # Find the face pointing towards -Y (the front) and inset it slightly 
    # or just rely on the door panel overlapping. To be cleaner, let's create 
    # a cutout for the door.
    # But since we are doing procedural primitives for efficiency, we will use pieces.
    bm.free()

    # Redefining Body as assembly of panels for better geometry
    bpy.data.objects.remove(body)
    
    # Sides
    side_l = create_box("SideL", wall_thickness, cab_d, cab_h, location=(-cab_w/2 + wall_thickness/2, 0, cab_h/2), material=dark_wood_mat)
    side_r = create_box("SideR", wall_thickness, cab_d, cab_h, location=(cab_w/2 - wall_thickness/2, 0, cab_h/2), material=dark_wood_mat)
    # Top
    top = create_box("Top", cab_w, cab_d, wall_thickness, location=(0, 0, cab_h - wall_thickness/2), material=dark_wood_mat)
    # Bottom / Base
    bottom = create_box("Bottom", cab_w, cab_d, wall_thickness, location=(0, 0, wall_thickness/2), material=dark_wood_mat)
    # Back panel
    back = create_box("Back", cab_w, wall_thickness, cab_h - (wall_thickness*2), location=(0, -cab_d/2 + wall_thickness/2, cab_h/2), material=dark_wood_mat)
    # Kickplate
    kick = create_box("Kickplate", cab_w, 0.05, 0.1, location=(0, -cab_d/2 + 0.025, 0.05), material=dark_wood_mat)

    # 2. Door Panel
    door_w = cab_w - (wall_thickness * 2) # Fit between sides
    door_h = cab_h - (wall_thickness * 2 + 0.1) # Fit between top/bottom and above kickplate
    door_thick = 0.015
    
    # Door position: Centered in X, at the front face (-Y), height offset for kickplate
    door_pos_y = -cab_d/2 + door_thick/2
    door_pos_z = (cab_h - door_h)/2 + 0.05 # Small gap from top and above kickplate
    
    door = create_box("Door", door_w, door_thick, door_h, location=(0, door_pos_y, door_pos_z), material=dark_wood_mat)
    
    # 3. Door Handle
    create_handle(door, door_w, door_h, door_thick)

    # Group into one object (optional but clean)
    bpy.ops.object.select_all(action='DESELECT')
    for obj in [side_l, side_r, top, bottom, back, kick, door]:
        obj.select_set(True)
    
    # To avoid the handle being lost, let's ensure it is selected too
    handle = bpy.data.objects.get("Handle")
    if handle:
        handle.select_set(True)
        
    # Join all parts into one coherent assembly
    bpy.context.view_layer.objects.active = door
    bpy.ops.object.join()
    
    # Rename final object
    cabinet = bpy.context.active_object
    cabinet.name = "TallCabinet"

if __name__ == "__main__":
    build_cabinet()
