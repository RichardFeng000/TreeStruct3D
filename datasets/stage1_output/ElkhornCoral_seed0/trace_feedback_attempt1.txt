import bpy
import bmesh
import math
import random
from mathutils import Vector, Matrix

def clear_scene():
    """Clears all objects from the current scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_coral_material():
    """Creates a warm sandy beige material for the coral."""
    mat = bpy.data.materials.new(name="ElkhornCoralMat")
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    
    # Principled BSDF
    node_bsdf = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_bsdf.inputs['Base Color'].default_value = (0.85, 0.78, 0.65, 1.0) # Sandy beige
    node_bsdf.inputs['Roughness'].default_value = 0.9
    
    # Noise for subtle variation
    node_noise = nodes.new(type='ShaderNodeTexNoise')
    node_noise.inputs['Scale'].default_value = 20.0
    node_noise.inputs['Detail'].default_value = 15.0
    
    node_color_ramp = nodes.new(type='ShaderNodeValToRGB')
    node_color_ramp.color_ramp.elements[0].color = (0.7, 0.6, 0.45, 1.0) # Darker tan
    node_color_ramp.elements[1].color = (0.9, 0.85, 0.75, 1.0) # Lighter beige
    
    # Links
    links = mat.node_tree.links
    links.new(node_noise.outputs['Fac'], node_color_ramp.inputs['Fac'])
    links.new(node_color_ramp.outputs['Color'], node_bsdf.inputs['Base Color'])
    
    # Output
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    links.new(node_bsdf.outputs['BSDF'], node_output.inputs['Surface'])
    
    return mat

def create_blade_segment(bm, start_vert, direction, length, width, thickness, segments=4):
    """Extrudes a flattened blade segment from a starting vertex."""
    # Create the cross-section (an ellipse/flattened circle)
    # We define a perpendicular plane to the direction
    up = Vector((0, 0, 1))
    if abs(direction.dot(up)) > 0.9:
        up = Vector((0, 1, 0))
        
    right = direction.cross(up).normalized()
    actual_up = right.cross(direction).normalized()
    
    # Initial face around the start_vert
    res = 8
    verts = []
    for i in range(res):
        angle = (2 * math.pi * i) / res
        off = (right * math.cos(angle) * width + actual_up * math.sin(angle) * thickness) * 0.5
        v = bm.verts.new((start_vert.co + off))
        verts.append(v)
    
    bm.faces.new(verts)
    
    current_face = bm.faces[0] # Simplified for a single segment, but we need to track it
    # In bmesh, we can't just take the first face; let's be more precise
    target_face = None
    for f in bm.faces:
        if any(v == verts[0] for v in f.verts):
            target_face = f
            break

    # Extrude segments
    last_face = target_face
    for s in range(segments):
        seg_len = length / segments
        # Scale width as it grows (palmate expansion)
        growth_factor = 1.0 + (s * 0.2)
        
        # Extrude
        res_ext = bmesh.ops.extrude_face_region(bm, geom=[last_face])
        verts_new = [v for v in res_ext['geom'] if isinstance(v, bmesh.types.BMVert)]
        
        # Move and scale the new verts
        for v in verts_new:
            v.co += direction * seg_len
            # Offset from center of segment to apply width scaling
            center = Vector((0, 0, 0))
            for vn in verts_new: center += vn.co
            center /= len(verts_new)
            
            local_pos = v.co - center
            # Project local pos onto the plane and scale
            v.co = center + (local_pos * growth_factor)
            
        # Update last_face for next iteration
        for f in res_ext['geom']:
            if isinstance(f, bmesh.types.BMFace):
                last_face = f
                break
    
    return last_face

def build_elkhorn():
    clear_scene()
    mat = create_coral_material()
    
    # Create a mesh and bmesh for the coral
    mesh = bpy.data.meshes.new("ElkhornCoral")
    obj = bpy.data.objects.new("ElkhornCoral", mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()
    
    # 1. Shared Base (a rough mound)
    base_radius = 0.4
    base_height = 0.3
    # Create a half-sphere base
    bmesh.ops.create_uvsphere(bm, u_segments=16, v_segments=8, radius=base_radius)
    # Flatten the bottom
    for v in bm.verts:
        if v.co.z < 0:
            v.co.z = 0
            
    # Find vertices on top of the base to start branches
    top_verts = [v for v in bm.verts if v.co.z > 0.2]
    
    # 2. Procedural Branching Logic
    num_main_branches = 6
    for i in range(num_main_branches):
        # Random starting point on top of the base
        start_v = random.choice(top_verts) if top_verts else bm.verts[0]
        
        # Branch direction: generally upwards and outwards
        angle = (2 * math.pi * i / num_main_branches) + random.uniform(-0.2, 0.2)
        dir_vec = Vector((math.cos(angle), math.sin(angle), 1.0)).normalized()
        
        # Create main stem
        last_face = create_blade_segment(bm, start_v, dir_vec, length=1.5, width=0.2, thickness=0.1)
        
        # Create splits (palmate fanning)
        num_splits = random.randint(2, 4)
        for j in range(num_splits):
            split_angle_x = random.uniform(-0.5, 0.5)
            split_angle_y = random.uniform(-0.5, 0.5)
            
            # Rotate the direction vector for the split
            rot_mat = Matrix.Rotation(split_angle_x, 4, 'X') @ Matrix.Rotation(split_angle_y, 4, 'Y')
            split_dir = (rot_mat @ dir_vec).normalized()
            
            # Find a vertex on the last face to start from
            start_v_split = random.choice(last_face.verts)
            create_blade_segment(bm, start_v_split, split_dir, length=1.2, width=0.3, thickness=0.08)

    # Finalize mesh
    bm.to_mesh(mesh)
    bm.free()
    
    # Assign Material
    obj.data.materials.append(mat)
    
    # 3. Adding "Granular" Detail via Modifiers
    # Subdivision for smoothness
    subsurf = obj.modifiers.new(name="Subdiv", type='SUBSURF')
    subsurf.levels = 2
    subsurf.render_levels = 2
    
    # Displace modifier for the "rough, granular surface texture"
    displace = obj.modifiers.new(name="GranularTexture", type='DISPLACE')
    tex = bpy.data.textures.new("CoralGrain", type='CLOUDS')
    tex.noise_scale = 0.1
    displace.texture = tex
    displace.strength = 0.05
    
    # Smooth shading
    for poly in mesh.polygons:
        poly.use_smooth = True

if __name__ == "__main__":
    build_elkhorn()
