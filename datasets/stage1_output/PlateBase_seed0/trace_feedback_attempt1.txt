def create_material(name, color):
    """Create a simple principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
    return mat

def create_plate_final():
    # Dimensions
    r_outer = 0.15
    r_inner = 0.12
    h_total = 0.02
    t_bottom = 0.006
    
    bm = bmesh.new()
    
    # Bottom Disk
    bmesh.ops.create_circle(bm, cap_ends=True, radius=r_outer, segments=64)
    
    # Extrude thickness of bottom
    faces = bm.faces[:]
    res = bmesh.ops.extrude_face_region(bm, geom=faces)
    verts = res['geom'].vertices
    bmesh.ops.translate(bm, verts=verts, vec=(0, 0, t_bottom))
    
    # Identify the top face (the bottom of the "well")
    top_face = [f for f in bm.faces if abs(f.calc_center_median().z - t_bottom) < 0.001][0]
    
    # Inset to create the rim's inner edge
    res = bmesh.ops.inset_individual(bm, faces=[top_face], width=r_outer - r_inner)
    well_face = res['faces'][0]
    
    # Extrude well_face (the center part) slightly down to create a subtle concave effect
    res = bmesh.ops.extrude_face_region(bm, geom=[well_face])
    well_verts = res['geom'].vertices
    bmesh.ops.translate(bm, verts=well_verts, vec=(0, 0, -0.002))
    
    # Now we need the outer rim walls
    # The inset created a ring face around the well_face. Let's find that ring face.
    rim_face = [f for f in bm.faces if f != well_face and abs(f.calc_center_median().z - t_bottom) < 0.01][0]
    
    # Extrude the rim face upwards to create the lip
    res = bmesh.ops.extrude_face_region(bm, geom=[rim_face])
    rim_verts = res['geom'].vertices
    bmesh.ops.translate(bm, verts=rim_verts, vec=(0, 0, h_total - t_bottom))
    
    # Bevel the edges for smoothness
    bm.edges.index_set('select', 0)
    for e in bm.edges:
        if abs(e.verts[0].co.z - h_total) < 0.001 or abs(e.verts[1].co.z - h_total) < 0.001:
            e.select = True
    
    bmesh.ops.bevel(bm, geom=bm.edges, offset=0.002, segments=3, affect='EDGES')

    # Create Mesh and Object
    mesh = bpy.data.meshes.new("PlateMesh")
    bm.to_mesh(mesh)
    bm.free()
    
    obj = bpy.data.objects.new("DinnerPlate", mesh)
    bpy.context.collection.objects.link(obj)
    return obj

def create_base_ring():
    # Dimensions
    r_outer = 0.08
    r_inner = 0.075
    h = 0.003
    
    bm = bmesh.new()
    bmesh.ops.create_circle(bm, cap_ends=True, radius=r_outer, segments=64)
    # Inset to make a ring
    res = bmesh.ops.inset_individual(bm, faces=[bm.faces[0]], width=r_outer - r_inner)
    # Remove the center face to leave just the ring
    center_face = res['faces'][0]
    bm.faces.remove(center_face)
    
    # Extrude for height
    edges = bm.edges[:]
    res = bmesh.ops.extrude_edge_only(bm, edges=edges)
    verts = res['geom'].vertices
    bmesh.ops.translate(bm, verts=verts, vec=(0, 0, -h)) # Move down from origin
    
    # Close the ring
    bmesh.ops.contextual_create(bm, geom=res['geom'])

    mesh = bpy.data.meshes.new("BaseRingMesh")
    bm.to_mesh(mesh)
    bm.free()
    
    obj = bpy.data.objects.new("BaseRing", mesh)
    bpy.context.collection.objects.link(obj)
    return obj
