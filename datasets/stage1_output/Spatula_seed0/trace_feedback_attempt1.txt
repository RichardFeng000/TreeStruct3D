import bpy
import bmesh
import math

def clear_scene():
    """Clears default Blender scene objects."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_spatula_material():
    """Creates a dark blue-gray material."""
    mat = bpy.data.materials.new(name="SpatulaMaterial")
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    # Dark blue-gray color: roughly (0.1, 0.15, 0.2)
    bsdf.inputs['Base Color'].default_value = (0.1, 0.15, 0.2, 1.0)
    bsdf.inputs['Roughness'].default_value = 0.4
    bsdf.inputs['Metallic'].default_value = 0.1
    return mat

def build_spatula():
    # Dimensions
    handle_radius = 0.5
    handle_length = 28.0
    head_width = 8.0
    head_depth = 12.0
    head_thickness = 0.3
    transition_length = 3.0
    
    # Create mesh and object
    mesh = bpy.data.meshes.new("SpatulaMesh")
    obj = bpy.data.objects.new("Spatula", mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()
    
    # 1. Create the handle (cylinder-like but transitionable)
    # We'll start from the bottom of the handle
    bmesh.ops.create_circle(bm, segments=32, radius=handle_radius, location=(0, 0, 0))
    
    # Extrude handle up to the transition point
    # Need a face first
    face = bm.faces[0]
    res = bmesh.ops.extrude_face(bm, vec=(0, 0, handle_length))
    verts_top = [v for v in res['geom'] if isinstance(v, bmesh.types.BMVert)]
    
    # 2. Transition from circular handle to rectangular head
    # We'll use a series of extrusions and scaling/shaping
    # To make it smooth, we'll split the circle into 4 quadrants and stretch them
    current_verts = verts_top
    steps = 5
    step_height = transition_length / steps
    
    for i in range(steps):
        res = bmesh.ops.extrude_face(bm, vec=(0, 0, step_height))
        new_verts = [v for v in res['geom'] if isinstance(v, bmesh.types.BMVert)]
        
        # Calculate target width and depth based on progress (i)
        t = (i + 1) / steps
        curr_w = handle_radius * 2 + (head_width - handle_radius * 2) * t
        curr_d = handle_radius * 2 + (head_depth - handle_radius * 2) * t
        
        # Reshape the vertices of the newly extruded face
        # We assume a circle, so we map them to a rounded rectangle
        for v in new_verts:
            x, y, z = v.co
            # Normalize x and y relative to current radius to find angle
            angle = math.atan2(y, x)
            # Map circular position to a box shape (approximately)
            # Using a simple smooth-step or sin/cos mapping for roundedness
            s_x = math.sin(angle)
            s_y = math.cos(angle)
            v.co.x = s_x * (curr_w / 2)
            v.co.y = s_y * (curr_d / 2)
            
        current_verts = new_verts

    # 3. Create the head thickness
    res = bmesh.ops.extrude_face(bm, vec=(0, 0, head_thickness))
    head_top_verts = [v for v in res['geom'] if isinstance(v, bmesh.types.BMVert)]
    
    # Create the top face of the head (already exists from extrusion)
    # Now we add a slight curve/bow to the working surface
    for v in head_top_verts:
        # The "working surface" is slightly curved along the depth (Y axis)
        # parabola shape: z = z0 + k * x^2 
        # Let's make it a gentle arc across the width
        dist_from_center = abs(v.co.x) / (head_width / 2)
        v.co.z += 0.15 * (1 - dist_from_center**2) # Slight crown

    # Finalize BMesh
    bm.to_mesh(mesh)
    bm.free()

    # Add Bevel Modifier for clean squared-off edges
    bevel = obj.modifiers.new(name="Bevel", type='BEVEL')
    bevel.width = 0.1
    bevel.segments = 3
    
    # Add Subdivision Surface for smoothness
    subsurf = obj.modifiers.new(name="Subsurf", type='SUBSURF')
    subsurf.levels = 2

    # Smooth shading
    for poly in mesh.polygons:
        poly.use_smooth = True

    # Assign material
    mat = create_spatula_material()
    obj.data.materials.append(mat)
    
    return obj

def main():
    clear_scene()
    spatula_obj = build_spatula()
    
    # Position object so it's centered around the origin but stands vertically
    # Since we built from z=0 upwards, let's offset it slightly to center it.
    total_height = 28.0 + 3.0 + 0.3
    spatula_obj.location.z = -total_height / 2

if __name__ == "__main__":
    main()
