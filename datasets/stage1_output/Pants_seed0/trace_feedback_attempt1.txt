import bpy
import bmesh
from mathutils import Vector

def clear_scene():
    """Clears all default objects from the scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Creates a simple principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
        # Make it look like fabric (less specular, slightly rough)
        bsdf.inputs['Roughness'].default_value = 0.8
        bsdf.inputs['Specular IOR Level'].default_value = 0.2
    return mat

def create_pants():
    """Constructs a pair of pants laid flat using BMesh."""
    # Dimensions and coordinates for the profile (X, Y)
    # Waist top: -15 to 15
    # Hips: -20 to 20
    # Leg length: down to -45
    # Crotch center: at Y = -20
    
    verts_coords = [
        (-15, 0),   # 0: Top Left Waist
        (15, 0),    # 1: Top Right Waist
        (20, -8),   # 2: Right Hip
        (12, -45),  # 3: Right Ankle Outer
        (5, -45),   # 4: Right Ankle Inner
        (0, -20),   # 5: Crotch Point (The center of the inverted-V)
        (-5, -45),  # 6: Left Ankle Inner
        (-12, -45), # 7: Left Ankle Outer
        (-20, -8),  # 8: Left Hip
    ]

    bm = bmesh.new()

    # Create the perimeter vertices
    verts = [bm.verts.new(Vector((x, y, 0))) for x, y in verts_coords]
    
    # Close the loop to create the main face
    for i in range(len(verts)):
        bm.edges.new((verts[i], verts[(i + 1) % len(verts)]))

    # Create the main surface face
    try:
        bm.faces.new(verts)
    except ValueError:
        # In case of overlapping or degenerate edges, though coords are clean here
        pass

    # Give it some geometric detail (subdivide for softness/modifiers)
    bmesh.ops.subdivide_edges(bm, edges=bm.edges, cuts=4, method='LINEAR')
    
    # To avoid a perfectly flat plane, we can slightly perturb vertices 
    # to simulate fabric laying on a surface or just rely on the solidify/subsurf
    
    # Convert BMesh to mesh
    mesh = bpy.data.meshes.new("PantsMesh")
    bm.to_mesh(mesh)
    bm.free()

    obj = bpy.data.objects.new("BlushPants", mesh)
    bpy.context.collection.objects.link(obj)

    # --- Modifiers for Fabric Look ---
    
    # 1. Solidify: Give the fabric thickness
    solidify = obj.modifiers.new(name="Solidify", type='SOLIDIFY')
    solidify.thickness = 0.15  # thin fabric
    solidify.offset = 0

    # 2. Subdivision Surface: Make edges round and soft
    subdiv = obj.modifiers.new(name="Subdivision", type='SUBSURF')
    subdiv.levels = 2
    subdiv.render_levels = 3

    # 3. Bevel: Smooth out the outer edges before solidify/subsurf if needed, 
    # but Subsurf usually handles this for simple planes. 
    # We'll apply a subtle displacement or just leave as is for "flat" look.

    return obj

def main():
    clear_scene()

    # Color: Soft light pink / blush (R=1, G=0.8, B=0.75)
    blush_color = (1.0, 0.82, 0.78, 1.0)
    pink_mat = create_material("BlushFabric", blush_color)

    # Create the pants object
    pants_obj = create_pants()

    # Assign material
    if pants_obj.data.materials:
        pants_obj.data.materials[0] = pink_mat
    else:
        pants_obj.data.materials.append(pink_mat)

    # Set shading to smooth
    bpy.context.view_layer.objects.active = pants_obj
    bpy.ops.object.shade_smooth()
    
    # Move it slightly so the thickness doesn't clip with Z=0 if rendered, 
    # though prompt says sitting at origin.
    pants_obj.location = (0, 0, 0)

if __name__ == "__main__":
    main()
