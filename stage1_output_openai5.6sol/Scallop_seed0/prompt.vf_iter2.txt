    material.use_nodes = True
    bsdf = material.node_tree.nodes.get("Principled BSDF")

    return material

# Shell dimensions.
HALF_ANGLE = math.radians(63.0)
R_INNER = 0.24
R_OUTER = 3.45
Y_SCALE = 1.10
RIB_COUNT = 14
NT = 101
NR = 69

UPPER_ANGLE = math.radians(31.0)
LOWER_ANGLE = math.radians(-18.0)

def margin_radius(theta):
    q = (theta + HALF_ANGLE) / (2.0 * HALF_ANGLE)
    side_rounding = 1.0 - 0.045 * (abs(theta) / HALF_ANGLE) ** 2
    scallops = 1.0 + 0.030 * math.cos(2.0 * math.pi * RIB_COUNT * q)
    return R_OUTER * side_rounding * scallops

def shell_point(theta, s, sign, outer_surface):
    q = (theta + HALF_ANGLE) / (2.0 * HALF_ANGLE)
    edge_radius = margin_radius(theta)
    radius = R_INNER + s * (edge_radius - R_INNER)

    x = radius * math.sin(theta)
    y = Y_SCALE * (radius * math.cos(theta) - R_INNER)

    dome = math.sin(math.pi * s) ** 0.72
    side_falloff = max(0.34, math.cos(theta * 0.72) ** 0.72)
    rib_envelope = math.sin(math.pi * min(1.0, s * 1.06)) ** 0.58
    rib_wave = math.cos(2.0 * math.pi * RIB_COUNT * q)
    growth_wave = math.sin(
        2.0 * math.pi * (10.5 * s + 0.025 * math.sin(theta * 2.0))
    )
    fine_growth = math.sin(2.0 * math.pi * 21.0 * s)

    if outer_surface:
        height = (
            0.055
            + 0.43 * dome * side_falloff
            + 0.074 * rib_wave * rib_envelope
            + 0.026 * growth_wave * (0.30 + 0.70 * s)
            + 0.007 * fine_growth * s
        )
    else:
        height = (
            0.018
            + 0.245 * dome * side_falloff
            + 0.018 * rib_wave * rib_envelope
            + 0.008 * growth_wave * s
        )

    return Vector((x, y, sign * height))

def rotate_about_hinge(point, angle):
    ca = math.cos(angle)
    sa = math.sin(angle)
    return Vector((
        point.x,
        point.y * ca - point.z * sa,
        point.y * sa + point.z * ca,
    ))

def outer_material_index(s):
    phase = math.sin(2.0 * math.pi * (10.5 * s + 0.015 * math.sin(5.0 * s)))
    fine = math.sin(2.0 * math.pi * 21.0 * s)

    if phase > 0.72:
        return 3
    if phase < -0.58:
        return 1
    if fine > 0.92:
        return 3
    if phase > 0.15:
        return 2
    return 0

def create_valve(name, sign, opening_angle):
    vertices = []
    faces = []
    material_ids = []

    # Outer and inner layers.
    for outer_surface in (True, False):
        for ir in range(NR):
            s = ir / (NR - 1)
            for it in range(NT):
                theta = -HALF_ANGLE + 2.0 * HALF_ANGLE * it / (NT - 1)
                point = shell_point(theta, s, sign, outer_surface)
                vertices.append(tuple(rotate_about_hinge(point, opening_angle)))

    layer_size = NR * NT

    # Exterior.
    for ir in range(NR - 1):
        s_mid = (ir + 0.5) / (NR - 1)
        for it in range(NT - 1):
            a = ir * NT + it
            b = a + 1
            c = a + NT + 1
            d = a + NT

            if sign > 0.0:
                faces.append((a, b, c, d))
            else:
                faces.append((a, d, c, b))
            material_ids.append(outer_material_index(s_mid))

    # Interior.
    for ir in range(NR - 1):
        s_mid = (ir + 0.5) / (NR - 1)
        for it in range(NT - 1):
            a = layer_size + ir * NT + it
            b = a + 1
            c = a + NT + 1
            d = a + NT

            if sign > 0.0:
                faces.append((a, d, c, b))
            else:
                faces.append((a, b, c, d))

            theta_mid = -HALF_ANGLE + 2.0 * HALF_ANGLE * (it + 0.5) / (NT - 1)
            q = (theta_mid + HALF_ANGLE) / (2.0 * HALF_ANGLE)
            rib_signal = math.cos(2.0 * math.pi * RIB_COUNT * q)
            if rib_signal > 0.72 and s_mid > 0.12:
                material_ids.append(6)
            else:
                material_ids.append(5)

    # Scalloped outer lip.
    outer_ir = NR - 1
    for it in range(NT - 1):
        oa = outer_ir * NT + it
        ob = oa + 1
        ia = layer_size + outer_ir * NT + it
        ib = ia + 1
        faces.append((oa, ia, ib, ob))
        material_ids.append(4 if (it // 4) % 2 == 0 else 3)

    # Closed hinge edge.
    for it in range(NT - 1):
        oa = it
        ob = it + 1
        ia = layer_size + it
        ib = ia + 1
        faces.append((oa, ob, ib, ia))
        material_ids.append(4)

    # Closed fan sides.
    for side_it in (0, NT - 1):
        for ir in range(NR - 1):
            oa = ir * NT + side_it
            ob = (ir + 1) * NT + side_it
            ia = layer_size + ir * NT + side_it
            ib = layer_size + (ir + 1) * NT + side_it

            if side_it == 0:
                faces.append((oa, ia, ib, ob))
            else:
                faces.append((oa, ob, ib, ia))

            material_ids.append(3 if (ir // 4) % 3 == 0 else 2)

    mesh = bpy.data.meshes.new(name + " Mesh")
    mesh.from_pydata(vertices, [], faces)
    mesh.materials.clear()
    for material in materials:
        mesh.materials.append(material)

    for polygon, material_index in zip(mesh.polygons, material_ids):
        polygon.material_index = material_index
        polygon.use_smooth = True

    mesh.validate()
    mesh.update()

    obj = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(obj)
    return obj
