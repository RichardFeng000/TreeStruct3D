def smoothstep(value):
    return value * value * (3.0 - 2.0 * value)


def envelope(z):
    if z <= profile[0][0]:
        return profile[0][1]
    if z >= profile[-1][0]:
        return profile[-1][1]

    for index in range(len(profile) - 1):
        z0, r0 = profile[index]
        z1, r1 = profile[index + 1]
        if z0 <= z <= z1:
            q = smoothstep((z - z0) / (z1 - z0))
            return r0 + (r1 - r0) * q
    return 0.02


def shell_radius(z, theta):
    t = max(0.0, min(1.0, (z - z_min) / (z_max - z_min)))
    base = envelope(z)
    growth = whorl_count * (t ** 0.91)
    phase = 2.0 * math.pi * growth - theta

    rounded = (0.5 + 0.5 * math.cos(phase)) ** 2.2
    relief = 0.958 + 0.072 * rounded

    soft_ribs = 0.006 * math.cos(18.0 * theta + 0.32 * math.sin(phase))
    fine_growth = 0.004 * math.sin(3.0 * phase + 0.7)
    fade = min(1.0, base / 0.28)

    return max(0.012, base * (relief + fade * (soft_ribs + fine_growth)))

def front_surface_point(x, z, outward=0.0):
    base = envelope(z)
    x_limit = max(0.03, base * 0.92)
    x = max(-x_limit, min(x_limit, x))

    y = -math.sqrt(max(0.002, base * base - x * x))
    theta = math.atan2(y, x)

    for _ in range(3):
        radius = shell_radius(z, theta)
        y = -math.sqrt(max(0.001, radius * radius - x * x))
        theta = math.atan2(y, x)

    return (x, y - outward, z)

def make_curve_object(name, points, bevel_depth, material, cyclic=False):
    curve = bpy.data.curves.new(name + " Curve", type='CURVE')
    curve.dimensions = '3D'
    curve.resolution_u = 2
    curve.bevel_depth = bevel_depth
    curve.bevel_resolution = 3
    curve.resolution_v = 2

    spline = curve.splines.new('NURBS')
    spline.points.add(len(points) - 1)

    for index, point in enumerate(points):
        spline.points[index].co = (*point, 1.0)

    spline.order_u = min(3, len(points))
    spline.use_endpoint_u = not cyclic
    spline.use_cyclic_u = cyclic

    obj = bpy.data.objects.new(name, curve)
    bpy.context.collection.objects.link(obj)
    obj.data.materials.append(material)
    return obj
