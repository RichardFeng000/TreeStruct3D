import bpy
import bmesh
import math

def clear_scene():
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
        # Metallic look for a utensil
        bsdf.inputs['Metallic'].default_value = 0.8
        bsdf.inputs['Roughness'].default_value = 0.3
    return mat

def create_fork():
    # Dimensions
    handle_length = 16.0
    handle_width = 0.4
    handle_depth = 0.2
    neck_curve_radius = 2.0
    head_width = 2.8
    tine_length = 5.0
    tine_width = 0.3
    gap_width = 0.2

    bm = bmesh.new()

    # Create handle base profile (rounded rectangle)
    # Vertices for the bottom end of the handle
    res = 8
    for i in range(res):
        angle = (math.pi * 2 * i) / res
        x = math.cos(angle) * handle_width * 0.5
        y = math.sin(angle) * handle_depth * 0.5
        bm.verts.new((x, 0, y))

    # Create faces for the bottom cap
    bottom_verts = bm.verts[:]
    bm.faces.new(bottom_verts)
    
    # Extrude handle along Y axis (the length of the fork)
    # We'll do this in segments to allow tapering and curving
    segments = 20
    prev_face = bm.faces[0]
    
    for i in range(1, segments + 1):
        t = i / segments
        # Slight taper: wider at the neck, narrower at the end
        scale = 0.8 + (t * 0.4) 
        
        # Offset for curvature in the "neck" area (last few segments)
        offset_x = 0
        if t > 0.7:
            # Create a gentle curve in the neck
            curve_t = (t - 0.7) / 0.3
            offset_x = math.sin(curve_t * math.pi / 2) * 0.5
        
        y_pos = t * handle_length
        
        # Extrude the face
        res_verts = bmesh.ops.extrude_face_region(bm, geom=[prev_face])['geom']
        new_verts = [v for v in res_verts if isinstance(v, bmesh.types.BMVert)]
        
        # Apply scale and offset
        for v in new_verts:
            v.co.x *= scale
            v.co.z *= scale
            v.co.x += offset_x
            v.co.y = y_pos
            
        prev_face = [f for f in bm.faces if any(v in new_verts for v in f.verts)][0]

    # Transition to the head (the shoulder)
    shoulder_face = prev_face
    shoulder_verts = shoulder_face.verts[:]
    
    # Flatten and widen into a paddle shape
    # We manually create the head base
    head_center_y = handle_length
    head_center_x = shoulder_verts[0].co.x
    
    # Create a flat wide rectangle for the head base
    v1 = bm.verts.new((head_center_x - head_width/2, head_center_y, 0))
    v2 = bm.verts.new((head_center_x + head_width/2, head_center_y, 0))
    v3 = bm.verts.new((head_center_x + head_width/2, head_center_y + 0.5, 0))
    v4 = bm.verts.new((head_center_x - head_width/2, head_center_y + 0.5, 0))
    bm.faces.new((v1, v2, v3, v4))

    # Connect the handle to the head base (filling gaps)
    # To simplify, we will just extrude a wide block for the tines' root
    root_face = bm.faces[-1]
    root_verts = root_face.verts[:]
    
    # Create 4 Tines
    tine_count = 4
    start_x = head_center_x - head_width/2 + (tine_width/2 + gap_width/2)
    
    for i in range(tine_count):
        tx = start_x + i * (tine_width + gap_width)
        # Create the box for each tine
        tv1 = bm.verts.new((tx - tine_width/2, head_center_y + 0.5, 0))
        tv2 = bm.verts.new((tx + tine_width/2, head_center_y + 0.5, 0))
        tv3 = bm.verts.new((tx + tine_width/2, head_center_y + 0.5 + tine_length, 0))
        tv4 = bm.verts.new((tx - tine_width/2, head_center_y + 0.5 + tine_length, 0))
        bm.faces.new((tv1, tv2, tv3, tv4))
        
        # Give tines some thickness (depth)
        # Extrude the face along Z
        tine_face = bm.faces[-1]
        res_ext = bmesh.ops.extrude_face_region(bm, geom=[tine_face])['geom']
        new_verts = [v for v in res_ext if isinstance(v, bmesh.types.BMVert)]
        for v in new_verts:
            v.co.z += handle_depth * 0.8
            # Taper the tines to a point at the end
            if v.co.y > head_center_y + tine_length * 0.7:
                taper = (head_center_y + tine_length) - v.co.y
                v.co.x -= (v.co.x - tx) * (taper / (tine_length * 0.3))

    # Finalize BMesh
    bm.to_mesh(bpy.data.meshes.new("ForkMesh"))
    obj = bpy.data.objects.new("Fork", bpy.data.meshes["ForkMesh"])
    bpy.context.collection.objects.link(obj)
    bm.free()

    # Add a Subdivision Surface modifier for smoothness
    subsurf = obj.modifiers.new(name="Subdiv", type='SUBSURF')
    subsurf.levels = 2
    subsurf.render_levels = 2

    return obj
