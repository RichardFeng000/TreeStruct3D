import bpy
import bmesh
import random
import math
from mathutils import Vector

def clear_scene():
    """Clears the default Blender scene."""
    if "Cube" in bpy.data.objects:
        bpy.data.objects.remove(bpy.data.objects["Cube"], do_unlink=True)
    if "Camera" in bpy.data.objects:
        bpy.data.objects.remove(bpy.data.objects["Camera"], do_unlink=True)
    if "Light" in bpy.data.objects:
        bpy.data.objects.remove(bpy.data.objects["Light"], do_unlink=True)

def create_material(name, color):
    """Creates a simple material with the given color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
    return mat

def generate_stump():
    clear_scene()

    # Parameters
    radius = 1.2
    height = 3.0
    segments = 64
    rings = 40  # Vertical resolution for bark
    
    # Materials
    bark_mat = create_material("BarkMat", (0.4, 0.3, 0.2, 1.0)) # Sandy tan-brown
    splinter_mat = create_material("SplinterMat", (0.9, 0.85, 0.75, 1.0)) # Pale cream

    # Mesh setup
    mesh = bpy.data.meshes.new("TreeStumpMesh")
    obj = bpy.data.objects.new("TreeStump", mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()

    # 1. Create the base cylinder structure
    verts_per_ring = segments
    all_rings_verts = []
    for i in range(rings + 1):
        z = (i / rings) * height
        ring = []
        for j in range(verts_per_ring):
            angle = (j / verts_per_ring) * 2 * math.pi
            x = radius * math.cos(angle)
            y = radius * math.sin(angle)
            v = bm.verts.new(Vector((x, y, z)))
            ring.append(v)
        all_rings_verts.append(ring)

    # Create faces for the cylinder sides
    for i in range(rings):
        for j in range(verts_per_ring):
            v1 = all_rings_verts[i][j]
            v2 = all_rings_verts[i][(j + 1) % verts_per_ring]
            v3 = all_rings_verts[i+1][(j + 1) % verts_per_ring]
            v4 = all_rings_verts[i+1][j]
            bm.faces.new((v1, v2, v3, v4))

    # Create the bottom cap
    bottom_ring = all_rings_verts[0]
    bm.faces.new(reversed(bottom_ring))

    # 2. Bark Geometry: Displacement on the lower half
    # We avoid modifying vertices that are part of the top break too much to keep it "wood-like"
    for i in range(rings + 1):
        z_val = (i / rings) * height
        weight = 0.0
        if z_val < height * 0.7:
            # Bark is strongest at the bottom and fades as it reaches the break point
            weight = 1.0 - (z_val / (height * 0.7))
            for v in all_rings_verts[i]:
                angle = math.atan2(v.co.y, v.co.x)
                # Create coarse vertical ridges and bumpy noise
                ridge = 0.25 * math.sin(angle * 6 + random.uniform(-0.2, 0.2))
                noise = (random.random() - 0.5) * 0.3
                normal = (v.co - Vector((0, 0, v.co.z))).normalized()
                v.co += normal * (ridge + noise) * weight

    # 3. The "Break" and Splinters
    top_ring = all_rings_verts[-1]
    
    # First, make the break point jagged
    for v in top_ring:
        v.co.z += random.uniform(-0.8, 0.4)
        v.co.x += random.uniform(-0.2, 0.2)
        v.co.y += random.uniform(-0.2, 0.2)

    # Create a jagged top face (the cross-section)
    try:
        bm.faces.new(top_ring)
    except ValueError:
        pass # If indices are too messy for a single face, we can skip it or triangulate

    # Generate splinter shards projecting upward
    num_splinters = 60
    for _ in range(num_splinters):
        # Pick a random starting point on the top break
        base_v = random.choice(top_ring)
        
        # To avoid "face already exists", we create new vertices for each shard's tip
        # A shard is represented as a thin, tapered tetrahedron or spike
        
        # Tip of the splinter
        extrude_h = random.uniform(0.5, 2.8)
        jitter_x = random.uniform(-0.4, 0.4)
        jitter_y = random.uniform(-0.4, 0.4)
        tip_v = bm.verts.new(base_v.co + Vector((jitter_x, jitter_y, extrude_h)))
        
        # To give the splinter some thickness, we create two more vertices near the base
        # We'll find neighbors of base_v in the top ring to form a small wedge
        idx = top_ring.index(base_v)
        v_prev = top_ring[(idx - 1) % verts_per_ring]
        v_next = top_ring[(idx + 1) % verts_per_ring]
        
        # Create faces for the shard (triangulating the spike)
        # These are unlikely to exist because tip_v is unique per iteration
        try:
            bm.faces.new((base_v, v_next, tip_v))
            bm.faces.new((base_v, v_prev, tip_v))
            bm.faces.new((v_next, v_prev, tip_v))
        except ValueError:
            pass # Skip if a face already exists

    # Finalize mesh
    bm.to_mesh(mesh)
    bm.free()

    # Assign materials based on height
    obj.data.materials.append(bark_mat)
    obj.data.materials.append(splinter_mat)
    
    for poly in obj.data.polygons:
        center_z = sum([v.co.z for v in poly.vertices]) / len(poly.vertices)
        # The top section (including the splinters) gets the cream material
        if center_z > height * 0.5:
            poly.material_index = 1
        else:
            poly.material_index = 0

    # Center object geometry at origin
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.origin_set(type='GEOMETRY_ORIGIN', center='BOUNDS')
    obj.location = (0, 0, 0)

if __name__ == "__main__":
    generate_stump()
