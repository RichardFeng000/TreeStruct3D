import bpy
import bmesh
import random
import math
from mathutils import Vector

def clear_scene():
    """Clears the default scene objects."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_sage_green_material():
    """Creates a muted, distinct sage green material."""
    mat = bpy.data.materials.new(name="LichenMaterial")
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        # Muted Sage Green: Slightly darker and more saturated than previous version to avoid washout
        bsdf.inputs['Base Color'].default_value = (0.35, 0.42, 0.28, 1.0)
        bsdf.inputs['Roughness'].default_value = 0.9
        bsdf.inputs['Specular IOR Level'].default_value = 0.2
    return mat

def create_organic_lobe(bm, angle, length, width_factor=1.0):
    """
    Generates a single organic lobe radiating from the center.
    """
    segments = 12
    rings = 8
    
    # Create vertices for the lobe
    verts = []
    for r in range(rings):
        t = r / (rings - 1)
        ring_verts = []
        
        # Radial distance from center
        dist = t * length
        
        for s in range(segments + 1):
            # Calculate angle offset for the lobe width
            # Lobe is wider at the end than the start
            half_width = (t * 0.3) * width_factor
            theta = angle + (s / segments - 0.5) * half_width
            
            # Add organic noise to position
            jitter = 0.05 * t
            x = dist * math.cos(theta) + random.uniform(-jitter, jitter)
            y = dist * math.sin(theta) + random.uniform(-jitter, jitter)
            
            # Z height: low rosette means overall flat, but with local ruffles
            # Base is at 0, edges curl slightly up or down
            z = (t**2 * 0.1) + math.sin(theta * 4 + t * 5) * 0.08 * t
            
            v = bm.verts.new(Vector((x, y, z)))
            ring_verts.append(v)
        verts.append(ring_verts)

    # Create faces
    for r in range(rings - 1):
        for s in range(segments):
            bm.faces.new((verts[r][s], verts[r][s+1], verts[r+1][s+1], verts[r+1][s]))

def generate_lichen():
    # Initialize BMesh
    bm = bmesh.new()

    # Create a base disk for the "central base" stability
    # This prevents it from looking like floating sheets
    base_res = 16
    base_verts = []
    for s in range(base_res):
        theta = (s / base_res) * 2 * math.pi
        v = bm.verts.new(Vector((0.1 * math.cos(theta), 0.1 * math.sin(theta), 0)))
        base_verts.append(v)
    for s in range(base_res):
        bm.faces.new((base_verts[s], base_verts[(s+1)%base_res], Vector((0,0,0)))) # simplified center

    # Radiating lobes: Low rosette shape
    num_lobes = 45
    for i in range(num_lobes):
        angle = (i / num_lobes) * 2 * math.pi + random.uniform(-0.1, 0.1)
        length = random.uniform(0.6, 1.3)
        width = random.uniform(0.7, 1.4)
        create_organic_lobe(bm, angle, length, width)

    # Mesh conversion
    mesh = bpy.data.meshes.new("LichenMesh")
    bm.to_mesh(mesh)
    bm.free()
    
    obj = bpy.data.objects.new("LichenGrowth", mesh)
    bpy.context.collection.objects.link(obj)
    
    # Material assignment
    mat = create_sage_green_material()
    obj.data.materials.append(mat)
    
    # Modifiers to achieve "ruffled" organic look
    # 1. Solidify for leaf-like thickness
    solid = obj.modifiers.new(name="Solidify", type='SOLIDIFY')
    solid.thickness = 0.015
    
    # 2. Subsurf for smoothness
    subsurf = obj.modifiers.new(name="Subdivision", type='SUBSURF')
    subsurf.levels = 2
    
    # 3. Displace for the organic "lichen" texture (micro-ruffles)
    tex = bpy.data.textures.new("LichenNoise", type='CLOUDS')
    tex.noise_scale = 0.15
    disp = obj.modifiers.new(name="Displace", type='DISPLACE')
    disp.texture = tex
    disp.strength = 0.04

    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.shade_smooth()

def main():
    clear_scene()
    generate_lichen()

if __name__ == "__main__":
    main()
