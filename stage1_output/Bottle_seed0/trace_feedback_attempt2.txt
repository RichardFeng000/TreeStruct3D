import bpy
import bmesh
import math
import random
from mathutils import Vector

def clear_scene():
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    node_principled = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    node_principled.inputs['Base Color'].default_value = color
    mat.node_tree.links.new(node_principled.outputs['BSDF'], node_output.inputs['Surface'])
    return mat

def create_bottle():
    # Parameters
    base_radius = 0.6
    shoulder_radius = 0.45
    neck_radius = 0.18
    body_height = 2.0
    neck_height = 0.7
    cap_height = 0.25
    cap_radius = 0.2
    label_height = 0.6
    label_z_start = 0.6
    label_offset = 0.015

    # --- Bottle Body and Neck ---
    bm = bmesh.new()
    
    # Create base disk (caps=True ensures there is a face)
    bmesh.ops.create_circle(bm, radius=base_radius, segments=32, caps=True)
    
    # Extrude and taper to shoulder
    face = bm.faces[0]
    res = bmesh.ops.extrude_face_region(bm, geom=[face])
    verts = [v for v in res['geom'] if isinstance(v, bmesh.types.Vert)]
    # The first set of verts are the base; we need to move and scale the extruded ones
    # In BMesh extrude, the original vertices remain, new ones are created.
    # We find the newly created vertices by filtering those not in the original circle.
    # Actually, a simpler way is to iterate all verts and check their Z height after extrusion
    for v in bm.verts:
        if v.co.z == 0: continue # skip base
        # Move to body_height and scale x,y
        v.co.z = body_height
        scale = shoulder_radius / base_radius
        v.co.x *= scale
        v.co.y *= scale

    # Shoulder transition to neck (smooth fillet-like taper)
    face = bm.faces[-1]
    res = bmesh.ops.extrude_face_region(bm, geom=[face])
    verts = [v for v in res['geom'] if isinstance(v, bmesh.types.Vert)]
    for v in verts:
        v.co.z += 0.3 # Shoulder height
        # Interpolate radius from shoulder_radius to neck_radius
        dist = Vector((v.co.x, v.co.y, 0)).length
        if dist > 0:
            scale = neck_radius / dist
            v.co.x *= scale
            v.co.y *= scale

    # Neck extrusion (cylinder)
    face = bm.faces[-1]
    res = bmesh.ops.extrude_face_region(bm, geom=[face])
    verts = [v for v in res['geom'] if isinstance(v, bmesh.types.Vert)]
    for v in verts:
        v.co.z += neck_height

    # Final closure (already handled by create_circle caps=True and extrude)
    bm.to_mesh("bottle_body")
    obj_body = bpy.data.objects.new("BottleBody", bpy.data.meshes["bottle_body"])
    bpy.context.collection.objects.link(obj_body)
    bm.free()

    # --- Cap ---
    cap_bm = bmesh.new()
    bmesh.ops.create_circle(cap_bm, radius=cap_radius, segments=32, caps=True)
    face = cap_bm.faces[0]
    res = bmesh.ops.extrude_face_region(cap_bm, geom=[face])
    verts = [v for v in res['geom'] if isinstance(v, bmesh.types.Vert)]
    for v in verts:
        v.co.z += cap_height
    
    # Round the top of the cap using bevel on the top face's edges
    top_face = cap_bm.faces[-1]
    edges = top_face.edges
    bmesh.ops.bevel(cap_bm, geom=edges, width=0.1, segments=8, affect='EDGES')
    
    cap_bm.to_mesh("bottle_cap")
    obj_cap = bpy.data.objects.new("BottleCap", bpy.data.meshes["bottle_cap"])
    bpy.context.collection.objects.link(obj_cap)
    # Position cap at top of bottle
    obj_cap.location.z = body_height + 0.3 + neck_height
    cap_bm.free()

    # --- Label Band ---
    # The label should follow the taper of the bottle body.
    label_bm = bmesh.new()
    # Start with a circle at the base of the label position
    z1 = label_z_start
    r1 = base_radius - (base_radius - shoulder_radius) * (z1 / body_height) + label_offset
    bmesh.ops.create_circle(label_bm, radius=r1, segments=32, caps=False)
    
    # Create the band by extruding and tapering
    verts_bottom = [v for v in label_bm.verts]
    edges_bottom = [e for e in label_bm.edges]
    res = bmesh.ops.extrude_edge_region(label_bm, geom=edges_bottom)
    verts_top = [v for v in res['geom'] if isinstance(v, bmesh.types.Vert)]
    
    z2 = z1 + label_height
    r2 = base_radius - (base_radius - shoulder_radius) * (z2 / body_height) + label_offset
    
    for v in verts_top:
        v.co.z = label_height # local coordinate since we will move the object later
        dist = Vector((v.co.x, v.co.y, 0)).length
        if dist > 0:
            scale = r2 / dist
            v.co.x *= scale
            v.co.y *= scale

    label_bm.to_mesh("bottle_label")
    obj_label = bpy.data.objects.new("BottleLabel", bpy.data.meshes["bottle_label"])
    bpy.context.collection.objects.link(obj_label)
    obj_label.location.z = label_z_start
    label_bm.free()

    # --- Materials ---
    glass_mat = create_material("Glass", (0.1, 0.5, 0.8, 1.0))
    cap_mat = create_material("CapPlastic", (0.9, 0.2, 0.2, 1.0))
    
    obj_body.data.materials.append(glass_mat)
    obj_cap.data.materials.append(cap_mat)

    # Colorful Label Pattern: assign different materials to face strips
    num_strips = 8
    label_mats = []
    for i in range(num_strips):
        color = (random.random(), random.random(), random.random(), 1.0)
        mat = create_material(f"LabelColor_{i}", color)
        label_mats.append(mat)
        obj_label.data.materials.append(mat)

    faces = obj_label.data.polygons
    if len(faces) > 0:
        for idx, face in enumerate(faces):
            # Map faces around the circumference to different materials
            # Assuming standard BMesh order for extruded circle edges
            strip_idx = (idx // (len(faces)//num_strips)) % num_strips
            face.material_index = strip_idx

    # Final Smoothing
    for obj in [obj_body, obj_cap]:
        bpy.context.view_layer.objects.active = obj
        bpy.ops.object.shade_smooth()

def main():
    clear_scene()
    create_bottle()

if __name__ == "__main__":
    main()
