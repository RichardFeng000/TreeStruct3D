import bpy
import bmesh

def clear_scene():
    """Clear the default scene of all objects."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Create a simple principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    bsdf.inputs['Base Color'].default_value = color
    return mat

def create_mini_fridge():
    # Dimensions
    width = 0.5  # x
    depth = 0.5  # y
    height = 0.6 # z
    corner_radius = 0.02
    door_gap = 0.005

    # Materials
    mat_body = create_material("BodyBrown", (0.15, 0.08, 0.05, 1.0)) # Dark brown
    mat_glass = create_material("GlassTop", (0.02, 0.02, 0.02, 1.0)) # Dark glass
    mat_metal = create_material("HandleMetal", (0.7, 0.7, 0.7, 1.0)) # Metallic silver

    # --- Main Chassis (The Shell) ---
    bpy.ops.mesh.primitive_cube_add(size=1.0, location=(0, 0, height/2))
    body = bpy.context.active_object
    body.name = "Fridge_Body"
    body.scale = (width, depth, height)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)

    # Create the hollow interior and separate the door face
    bm = bmesh.new()
    bm.from_mesh(body.data)
    
    # Find the front face (facing -Y usually in Blender if we consider Y as depth/width)
    # Let's assume Front is facing +Y or -Y. We'll use -Y for the door.
    front_face = None
    for f in bm.faces:
        if f.normal.y < -0.9:
            front_face = f
            break
    
    # Separate the front face to create the door later
    if front_face:
        # We will detach this face and move it slightly forward
        bm.faces.discard(front_face)

    bm.to_mesh(body.data)
    bm.free()

    # Bevel the main chassis edges for a smooth look
    bevel = body.modifiers.new(name="Bevel", type='BEVEL')
    bevel.width = corner_radius
    bevel.segments = 5

    # --- The Top Glass Surface ---
    bpy.ops.mesh.primitive_cube_add(size=1.0, location=(0, 0, height + 0.005))
    glass_top = bpy.context.active_object
    glass_top.name = "GlassTop"
    # Make it slightly smaller than the body and very thin
    glass_top.scale = (width * 0.98, depth * 0.98, 0.01)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    glass_top.data.materials.append(mat_glass)

    # --- The Door ---
    # Create a separate cube for the door
    bpy.ops.mesh.primitive_cube_add(size=1.0, location=(0, -depth/2 - door_gap, height/2))
    door = bpy.context.active_object
    door.name = "Fridge_Door"
    # Door dimensions (slightly smaller than body to fit)
    door.scale = (width * 0.98, 0.02, height * 0.95)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)

    # Bevel the door edges
    door_bev = door.modifiers.new(name="Bevel", type='BEVEL')
    door_bev.width = 0.01
    door_bev.segments = 3
    door.data.materials.append(mat_body)

    # --- The Handle ---
    handle_w = 0.02
    handle_h = 0.25
    handle_d = 0.03
    bpy.ops.mesh.primitive_cube_add(size=1.0, location=(width/2 - 0.05, -depth/2 - (door_gap + 0.02), height/2))
    handle = bpy.context.active_object
    handle.name = "DoorHandle"
    handle.scale = (handle_w, handle_d, handle_h)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    
    # Bevel the handle for a rounded metallic feel
    han_bev = handle.modifiers.new(name="Bevel", type='BEVEL')
    han_bev.width = 0.005
    han_bev.segments = 3
    handle.data.materials.append(mat_metal)

    # --- Feet ---
    foot_radius = 0.015
    foot_height = 0.03
    foot_positions = [
        (width/2 - 0.04, depth/2 - 0.04),
        (-width/2 + 0.04, depth/2 - 0.04),
        (width/2 - 0.04, -depth/2 + 0.04),
        (-width/2 + 0.04, -depth/2 + 0.04)
    ]

    for i, pos in enumerate(foot_positions):
        bpy.ops.mesh.primitive_cylinder_add(radius=foot_radius, depth=foot_height, location=(pos[0], pos[1], foot_height/2))
        foot = bpy.context.active_object
        foot.name = f"Foot_{i}"
        foot.data.materials.append(mat_body)

    # Assign body material to main chassis
    body.data.materials.append(mat_body)

    # Final adjustments: move the whole assembly so it sits on origin if needed, 
    # but we already calculated heights relative to z=0.
    
    # Parent parts for organization
    door.parent = body
    handle.parent = door
    glass_top.parent = body

if __name__ == "__main__":
    clear_scene()
    create_mini_fridge()
