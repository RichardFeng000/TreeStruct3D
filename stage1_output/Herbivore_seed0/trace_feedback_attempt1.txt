import bpy
import bmesh
import math
from mathutils import Vector

def clear_scene():
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material():
    # Create a material for the fluffy beast
    mat = bpy.data.materials.new(name="BeastFurMaterial")
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    links = mat.node_tree.links

    # Clear default nodes
    for node in nodes:
        nodes.remove(node)

    # Create a Principled BSDF
    bsdf = nodes.new('ShaderNodeBsdfPrincipled')
    output = nodes.new('ShaderNodeOutputMaterial')
    
    # Color logic for sandy beige-pink with brownish-gray stripes
    # Use a noise texture stretched on one axis to simulate flank stripes
    tex_coord = nodes.new('ShaderNodeTexCoord')
    mapping = nodes.new('ShaderNodeMapping')
    noise = nodes.new('ShaderNodeTexNoise')
    color_ramp = nodes.new('ShaderNodeValToRGB')
    
    # Set mapping to stretch noise for stripes (Scale Y=1, X=0.2, Z=1)
    mapping.inputs['Scale'].default_value = (0.5, 3.0, 1.0)
    
    # Colors: Sandy beige-pink vs Brownish-gray
    color_ramp.color_ramp.elements[0].color = (0.9, 0.75, 0.65, 1.0) # Beige-pink
    color_ramp.color_ramp.elements[1].color = (0.4, 0.35, 0.3, 1.0)   # Brownish-gray
    
    # Connect nodes
    links.new(tex_coord.outputs['Generated'], mapping.inputs['Vector'])
    links.new(mapping.outputs['Vector'], noise.inputs['Vector'])
    links.new(noise.outputs['Fac'], color_ramp.inputs['Fac'])
    links.new(color_ramp.outputs['Color'], bsdf.inputs['Base Color'])
    links.new(bsdf.outputs['BSDF'], output.inputs['Surface'])
    
    return mat

def create_beast():
    # 1. Body - Stocky, rounded
    bpy.ops.mesh.primitive_uv_sphere_add(radius=1, location=(0, 0, 1))
    body = bpy.context.active_object
    body.name = "BeastBody"
    body.scale = (1.2, 1.8, 1.0)
    bpy.ops.object.transform_apply(scale=True)

    # 2. Neck and Head
    # Short neck
    bpy.ops.mesh.primitive_cylinder_add(radius=0.4, depth=0.6, location=(0, 1.7, 1.3))
    neck = bpy.context.active_object
    neck.rotation_euler[0] = math.radians(-20)

    # Small head
    bpy.ops.mesh.primitive_uv_sphere_add(radius=0.45, location=(0, 2.1, 1.6))
    head = bpy.context.active_object
    head.scale = (0.8, 1.1, 0.9)

    # 3. Legs - Four thick legs
    leg_positions = [
        (0.7, 1.2, 0.5), (-0.7, 1.2, 0.5), # Front
        (0.7, -1.2, 0.5), (-0.7, -1.2, 0.5) # Back
    ]

    legs = []
    for pos in leg_positions:
        bpy.ops.mesh.primitive_cylinder_add(radius=0.3, depth=1.0, location=pos)
        leg = bpy.context.active_object
        # Taper the legs slightly using bmesh for rounded paws
        bm = bmesh.new()
        bm.from_mesh(leg.data)
        for v in bm.verts:
            if v.co.z < 0: # Bottom of leg
                v.co.x *= 1.3
                v.co.y *= 1.3
        bm.to_mesh(leg.data)
        bm.free()
        legs.append(leg)

    # Join all parts into one mesh
    bpy.ops.object.select_all(action='DESELECT')
    body.select_set(True)
    neck.select_set(True)
    head.select_set(True)
    for leg in legs:
        leg.select_set(True)
    
    bpy.context.view_layer.objects.active = body
    bpy.ops.object.join()

    # Smooth everything and add Subsurf for high fidelity roundedness
    bpy.ops.object.modifier_add(type='SUBSURF')
    body.modifiers["Subdivision"].levels = 2
    bpy.ops.object.shade_smooth()

    return body

def apply_fur(obj):
    # Add particle system for dense, fluffy fur
    part = obj.modifiers.new(name="Fur", type='PARTICLE_SYSTEM')
    psys = obj.particle_systems[0]
    settings = psys.settings

    settings.type = 'HAIR'
    settings.render_type = 'PATH'
    settings.count = 50000 # Dense fur
    settings.hair_length = 0.25
    
    # Shaggy appearance
    settings.roughness_1 = 0.1
    settings.roughness_2 = 0.05
    settings.phase_factor = 2.0
    settings.clumping = 0.1

def main():
    clear_scene()
    
    # Create material
    fur_mat = create_material()
    
    # Create the creature geometry
    beast = create_beast()
    
    # Assign material
    beast.data.materials.append(fur_mat)
    
    # Apply hair system
    apply_fur(beast)

if __name__ == "__main__":
    main()
