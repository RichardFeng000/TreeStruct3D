import bpy
import bmesh
import math
from mathutils import Vector

def clear_scene():
    """Removes all default objects from the scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_central_post(radius, height):
    """Creates the central support pillar."""
    bpy.ops.mesh.primitive_cylinder_add(
        vertices=32, 
        radius=radius, 
        depth=height, 
        location=(0, 0, height / 2)
    )
    post = bpy.context.active_object
    post.name = "CentralPost"
    return post

def create_tread(inner_r, outer_r, thickness, angle_deg):
    """Creates a single wedge-shaped tread using bmesh."""
    bm = bmesh.new()
    
    angle_rad = math.radians(angle_deg)
    
    # Vertices for the top face
    v1 = bm.verts.new(Vector((inner_r, 0, 0)))
    v2 = bm.verts.new(Vector((outer_r, 0, 0)))
    v3 = bm.verts.new(Vector((outer_r * math.cos(angle_rad), outer_r * math.sin(angle_rad), 0)))
    v4 = bm.verts.new(Vector((inner_r * math.cos(angle_rad), inner_r * math.sin(angle_rad), 0)))
    
    # Top face
    bm.faces.new((v1, v2, v3, v4))
    
    # Extrude downwards to create thickness
    edges = bm.edges
    res = bmesh.ops.extrude_face_region(bm, geom=[bm.faces[0]])
    
    # Move the extruded vertices down
    for v in res['geom']:
        if isinstance(v, bmesh.types.BMVert):
            v.co.z -= thickness

    bm.to_mesh("Tread")
    obj = bpy.data.objects.new("Tread", bm.to_mesh())
    bpy.context.collection.objects.link(obj)
    bm.free()
    return obj

def create_spiral_staircase():
    # Parameters
    post_radius = 0.2
    tread_width = 1.2
    outer_radius = post_radius + tread_width
    tread_thickness = 0.05
    step_height = 0.18
    angle_per_step = 22.5 # Degrees
    num_steps = 32        # Total steps (720 degrees total rotation)
    total_height = num_steps * step_height
    
    clear_scene()
    
    # 1. Central Post
    create_central_post(post_radius, total_height)
    
    # 2. Treads
    tread_proto = create_tread(post_radius, outer_radius, tread_thickness, angle_per_step)
    
    # We use the prototype to duplicate for efficiency or just loop construction
    # To keep it clean and procedural, we'll iterate and place them.
    for i in range(num_steps):
        # Create a new tread based on parameters since we need unique transforms
        t = create_tread(post_radius, outer_radius, tread_thickness, angle_per_step)
        
        angle_rad = math.radians(i * angle_per_step)
        z_pos = i * step_height
        
        # Rotation and Position
        t.location = (0, 0, z_pos)
        t.rotation_euler[2] = angle_rad
    
    # Cleanup the prototype if it exists in data but not scene
    if tread_proto.name == "Tread" and tread_proto not in bpy.context.scene.objects:
        bpy.data.objects.remove(tread_proto, do_unlink=True)

    # 3. Handrail
    # Create a curve that follows the outer edge of the treads
    curve_data = bpy.data.curves.new('HandrailCurve', type='CURVE')
    curve_data.dimensions = '3D'
    curve_data.fill_mode = 'FULL'
    curve_data.bevel_depth = 0.03
    curve_data.bevel_resolution = 4

    polyline = curve_data.splines.new('BEZIER')
    # Number of points: num_steps + 1 for a smooth end
    polyline.bezier_points.add(num_steps)
    
    rail_height_offset = 0.9 # height above the step
    
    for i in range(num_steps + 1):
        angle_rad = math.radians(i * angle_per_step)
        z_pos = (i * step_height) + rail_height_offset
        x = outer_radius * math.cos(angle_rad)
        y = outer_radius * math.sin(angle_rad)
        
        p = polyline.bezier_points[i]
        p.co = Vector((x, y, z_pos))
        # Set handle types to AUTOMATIC for a smooth spiral
        p.handle_left_type = 'AUTOMATIC'
        p.handle_right_type = 'AUTOMATIC'

    rail_obj = bpy.data.objects.new('Handrail', curve_data)
    bpy.context.collection.objects.link(rail_obj)

if __name__ == "__main__":
    create_spiral_staircase()
