import bpy
import bmesh
import math
import random
from mathutils import Vector, Matrix

def clear_scene():
    """Removes all objects from the current scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Creates a simple principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    
    # Create Principled BSDF and Output nodes
    node_bsdf = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    
    node_bsdf.inputs['Base Color'].default_value = color
    node_bsdf.inputs['Roughness'].default_value = 0.6
    
    links = mat.node_tree.links
    links.new(node_bsdf.outputs['BSDF'], node_output.inputs['Surface'])
    return mat

def create_agave():
    # Parameters for the agave plant
    num_leaves = 80
    golden_angle = 137.5 * (math.pi / 180) # Phyllotaxis angle in radians
    base_radius = 0.1
    max_leaf_length = 2.5
    max_leaf_width = 0.2
    leaf_thickness = 0.04
    
    # Create a container object for all leaves to keep things organized
    container = bpy.data.collections.new("AgavePlant")
    bpy.context.scene.collection.children.link(container)

    # Generate leaf materials with subtle color variation
    green_base = Vector((0.1, 0.3, 0.1)) # Dark green
    green_light = Vector((0.4, 0.6, 0.2)) # Pale greenish-yellow
    
    def get_random_green():
        # Interpolate between base and light green
        t = random.uniform(0, 1)
        return (
            green_base[0] + t * (green_light[0] - green_base[0]),
            green_base[1] + t * (green_light[1] - green_base[1]),
            green_base[2] + t * (green_light[2] - green_base[2]),
            1.0
        )

    # Create the central base/root mass
    bm_base = bmesh.new()
    bmesh.ops.create_uvsphere(bm_base, u_segments=16, v_segments=8, radius=0.3)
    # Squash it slightly to be more plant-like
    for v in bm_base.verts:
        v.co.z *= 0.5
    
    base_obj = bpy.data.objects.new("AgaveBase", bm_base)
    bpy.context.collection.objects.link(base_obj)
    bm_base.to_mesh(base_obj.data)
    bm_base.free()

    # Create leaves in a spiral pattern
    for i in range(num_leaves):
        # Phyllotaxis positioning
        angle = i * golden_angle
        radius = base_radius + (i * 0.015)
        x = math.cos(angle) * radius
        y = math.sin(angle) * radius
        z = i * 0.02  # Leaves emerge slightly higher as they go in
        
        # Variation in length and width based on growth order (inner leaves shorter/younger)
        # Outer leaves are indices 0 to N, inner are N down to 0? 
        # Let's treat i=0 as the oldest (outermost) and i=num_leaves-1 as newest (center).
        progress = i / num_leaves # 0 is old, 1 is young
        length = max_leaf_length * (1.2 - progress * 0.6)
        width = max_leaf_width * (1.0 - progress * 0.5)
        
        # Leaf geometry
        bm = bmesh.new()
        
        # Create a tapered, curved leaf
        segments = 12
        vertices = []
        
        for s in range(segments + 1):
            t = s / segments  # normalized length from base (0) to tip (1)
            
            # Tapering width: widest near base, sharp at tip
            current_width = width * (1.0 - t**2) if t < 0.9 else width * (1.0 - t) / 0.1
            if t == 1.0: current_width = 0
            
            # Curvature: arching outwards and slightly downwards
            # Using a simple quadratic curve for the spine
            spine_z = t * length * 0.3 * (1.0 - t * 0.5) # Slight upward rise then plateau
            spine_dist = t * length * 0.8
            
            # Local offsets for width
            v_left = bm.verts.new(Vector((-current_width/2, 0, 0)))
            v_right = bm.verts.new(Vector((current_width/2, 0, 0)))
            
            # Position the vertices along the curve
            # We'll translate and rotate them later via matrix to fit the spiral
            vertices.append((v_left, v_right))
            
        # Connect vertices into faces
        for s in range(segments):
            bm.faces.new((vertices[s][0], vertices[s+1][0], vertices[s+1][1], vertices[s][1]))

        # Now we need to "bend" the leaf's structure since it was built as a flat strip
        # We can do this by iterating through vertices and applying a curve formula
        for s in range(segments + 1):
            t = s / segments
            dist = t * length
            # Calculate local Z (up) and Y (outward from center)
            local_y = dist * math.cos(0.4 * t) # Curving downwards slightly
            local_z = dist * math.sin(0.4 * t) * 0.5
            
            # Shift the pair of vertices for this segment
            for v in [vertices[s][0], vertices[s][1]]:
                v.co.y = local_y
                v.co.z = local_z

        # Add thickness to the leaf (convert faces to volume)
        # Extrude all faces along their normals slightly
        bmesh.ops.extrude_face_region(bm, geom=bm.faces[:])
        for v in bm.verts:
            v.co += v.normal * (leaf_thickness / 2)

        # Finalize mesh object
        leaf_mesh = bpy.data.meshes.new(f"Leaf_{i}")
        bm.to_mesh(leaf_mesh)
        bm.free()
        
        leaf_obj = bpy.data.objects.new(f"Leaf_{i}", leaf_mesh)
        bpy.context.collection.objects.link(leaf_obj)
        
        # Position and Rotate the leaf to align with the spiral center
        # Rotation: Point from origin (0,0,0) to (x,y,z)
        rotation_vec = Vector((x, y, z))
        if rotation_vec.length > 0:
            rot_quat = rotation_vec.to_track_quat('Y', 'Z')
            leaf_obj.rotation_mode = 'QUATERNION'
            leaf_obj.rotation_quaternion = rot_quat
        
        leaf_obj.location = Vector((x, y, z))
        
        # Apply material with subtle variation
        mat = create_material(f"Mat_Leaf_{i}", get_random_green())
        leaf_obj.data.materials.append(mat)

    # Combine leaves into the base for a cleaner scene (optional, but keeps hierarchy tidy)
    # We'll leave them as separate objects for now to ensure materials apply correctly per-leaf
