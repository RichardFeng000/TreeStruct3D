def create_box(name, size, location):
    """Creates a simple box mesh using bmesh for precise control."""
    mesh = bpy.data.meshes.new(name)
    obj = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()
    bmesh.ops.create_cube(bm, size=1.0)
    # Scale to desired dimensions
    for v in bm.verts:
        v.co.x *= size[0] / 2
        v.co.y *= size[1] / 2
        v.co.z *= size[2] / 2
    
    bm.to_mesh(mesh)
    bm.free()
    obj.location = location
    return obj

def create_tapered_leg(name, height, top_rad, bot_rad, location):
    """Creates a tapered leg using bmesh.ops.create_cone."""
    mesh = bpy.data.meshes.new(name)
    obj = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()
    # Create cone for the leg (radius1 is bottom, radius2 is top)
    bmesh.ops.create_cone(
        bm, 
        cap_ends=True, 
        segments=16, 
        radius1=bot_rad, 
        radius2=top_rad, 
        depth=height
    )
    
    bm.to_mesh(mesh)
    bm.free()
    # Shift so the top is at 'location' and it grows downwards
    obj.location = (location[0], location[1], location[2] - height/2)
    return obj

def create_decorative_panel(name, width, height, depth, location):
    """Creates a panel with molded/carved detail using insets and extrusions."""
    mesh = bpy.data.meshes.new(name)
    obj = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()
    bmesh.ops.create_cube(bm, size=1.0)
    
    # Scale to main panel size
    for v in bm.verts:
        v.co.x *= width / 2
        v.co.y *= depth / 2
        v.co.z *= height / 2
    
    bm.faces.ensure_lookup_table()
    # We want to carve the faces that are perpendicular to the Y axis (the "front/back" of the panel)
    for f in bm.faces:
        if abs(f.normal.y) > 0.9:
            # Inset for a border
            res = bmesh.ops.inset_individual(bm, faces=[f], thickness=width * 0.05)
            inner_face = res['faces'][0]
            
            # Push the center inward to create depth (carving)
            direction = f.normal # Use the normal of the original face to push in
            push_vec = direction * -depth * 1.5
            bmesh.ops.translate(bm, verts=inner_face.verts, vec=push_vec)
            
            # Create an additional inner detail border
            res2 = bmesh.ops.inset_individual(bm, faces=[inner_face], thickness=width * 0.03)
            inner_detail = res2['faces'][0]
            bmesh.ops.translate(bm, verts=inner_detail.verts, vec=push_vec * 0.5)

    bm.to_mesh(mesh)
    bm.free()
    obj.location = location
    return obj
