def create_cabinet_door():
    """
    Creates a tall, narrow cabinet door panel with vertical edge detailing.
    Dimensions: Height=2.1m, Width=0.4m, Thickness=0.02m
    """
    # Parameters
    width = 0.4
    height = 2.1
    depth = 0.02
    border_width = 0.03  # The width of the outer frame detail
    inset_depth = 0.005   # How far the center panel is recessed
    bevel_amount = 0.002 # Subtle rounding for realism

    # Create BMesh
    bm = bmesh.new()

    # Create the main slab (the door)
    # Using a cube as a base and scaling it
    bmesh.ops.create_cube(bm, size=1.0)
    
    # Scale to dimensions
    # Note: create_cube makes a 2x2x2 cube centered at origin
    # We need to transform the vertices manually or scale the object later.
    # To be precise in BMesh, we'll move the vertices.
    for v in bm.verts:
        v.co.x *= (width / 2)
        v.co.y *= (depth / 2)
        v.co.z *= (height / 2)

    # Find the front face (the one facing -Y or +Y, let's use +Y as front)
    front_face = None
    for f in bm.faces:
        if f.normal.y > 0.9:
            front_face = f
            break

    if front_face:
        # To create "vertical edge detailing", we will inset the front face
        # and extrude it slightly inwards to create a recessed center panel.
        # Then add thin vertical grooves on the sides of that recession.
        
        # Inset the front face
        res = bmesh.ops.inset_individual(bm, faces=[front_face], width=border_width)
        inner_face = res['faces'][0]
        
        # Extrude inner face backwards (into the door)
        bmesh.ops.translate(bm, verts=inner_face.verts, vec=(0, -inset_depth, 0))

        # Add subtle vertical grooves on the left and right edges of the panel
        # We identify the vertical edges of the inner face
        inner_edges = [e for e in inner_face.edges]
        vertical_edges = [e for e in inner_edges if abs(e.verts[0].co.z - e.verts[1].co.z) > 1.0]

        for edge in vertical_edges:
            # Create a small groove by splitting the edge and extruding slightly
            # We'll use bevel on these edges to create a "U" shape detail
            bmesh.ops.bevel(bm, geom=[edge], offset=0.003, segments=2, affect='EDGES')

    # Final global bevel to remove CG sharp edges from the outer slab
    bmesh.ops.bevel(bm, geom=bm.edges, offset=bevel_amount, segments=3, affect='EDGES')

    # Create object and link to scene
    mesh = bpy.data.meshes.new("CabinetDoorPanel")
    bm.to_mesh(mesh)
    bm.free()

    obj = bpy.data.objects.new("CabinetDoorPanel", mesh)
    bpy.context.collection.objects.link(obj)
    
    # Center the object's origin
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.origin_set(type='ORIGIN_GEOMETRY', center='BOUNDS')

if __name__ == "__main__":
    setup_scene()
    create_cabinet_door()
