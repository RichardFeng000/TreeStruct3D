import bpy
import bmesh
import math
from mathutils import Vector, Matrix

def clear_scene():
    """Removes all objects from the current scene."""
    if bpy.context.active_object:
        bpy.ops.object.mode_set(mode='OBJECT')
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_lobed_profile(center, normal, radius, amplitude, ribs, resolution):
    """Generates a list of vertices forming a lobed circle."""
    vertices = []
    # Create local coordinate system for the ring
    up = normal
    if abs(up.x) < 0.9:
        right = up.cross(Vector((1, 0, 0))).normalized()
    else:
        right = up.cross(Vector((0, 1, 0))).normalized()
    forward = right.cross(up).normalized()

    for i in range(resolution):
        theta = (2 * math.pi * i) / resolution
        # The lobed radius calculation: base + oscillation
        r = radius + amplitude * math.cos(ribs * theta)
        x = math.cos(theta) * r
        y = math.sin(theta) * r
        pos = center + (right * x) + (forward * y)
        vertices.append(pos)
    return vertices

def build_cactus():
    clear_scene()

    # Parameters
    main_height = 8.0
    main_radius = 0.6
    rib_count = 8
    rib_amplitude = 0.15
    resolution = 32 # vertices per ring
    segments = 40   # rings along the length

    bm = bmesh.new()

    # 1. Create Main Stem
    main_rings = []
    for i in range(segments + 1):
        z = (i / segments) * main_height
        # Taper slightly towards top
        taper = 1.0 - (z / main_height) * 0.3
        r = main_radius * taper
        amp = rib_amplitude * taper
        
        center = Vector((0, 0, z))
        normal = Vector((0, 0, 1))
        
        ring_verts = create_lobed_profile(center, normal, r, amp, rib_count, resolution)
        
        # Add vertices to bmesh
        current_ring = [bm.verts.new(v) for v in ring_verts]
        main_rings.append(current_ring)

    # Connect main stem rings with faces
    for i in range(segments):
        r1 = main_rings[i]
        r2 = main_rings[i+1]
        for j in range(resolution):
            v1 = r1[j]
            v2 = r1[(j + 1) % resolution]
            v3 = r2[(j + 1) % resolution]
            v4 = r2[j]
            bm.faces.new((v1, v2, v3, v4))

    # 2. Create Curved Arm
    arm_segments = 30
    arm_radius_start = main_radius * 0.8
    arm_radius_end = main_radius * 0.5
    arm_height_start = 1.5 # branches from lower left
    
    # Path for the arm: Quadratic Bezier
    # Start: inside stem, Mid: out and left, End: up and slightly right
    p0 = Vector((0, 0, arm_height_start))
    p1 = Vector((-2.5, -0.5, arm_height_start + 1.0))
    p2 = Vector((-3.0, 0, arm_height_start + 6.0))

    def get_bezier_point(t):
        return (1-t)**2 * p0 + 2*(1-t)*t * p1 + t**2 * p2

    def get_bezier_tangent(t):
        return 2*(1-t)*(p1 - p0) + 2*t*(p2 - p1)

    arm_rings = []
    for i in range(arm_segments + 1):
        t = i / arm_segments
        center = get_bezier_point(t)
        tangent = get_bezier_tangent(t).normalized()
        
        # Radius and amplitude taper along the arm
        r = arm_radius_start + t * (arm_radius_end - arm_radius_start)
        amp = rib_amplitude * 0.8 * (1.0 - t * 0.5)
        
        ring_verts_pos = create_lobed_profile(center, tangent, r, amp, rib_count, resolution)
        current_ring = [bm.verts.new(v) for v in ring_verts_pos]
        arm_rings.append(current_ring)

    # Connect arm rings with faces
    for i in range(arm_segments):
        r1 = arm_rings[i]
        r2 = arm_rings[i+1]
        for j in range(resolution):
            v1 = r1[j]
            v2 = r1[(j + 1) % resolution]
            v3 = r2[(j + 1) % resolution]
            v4 = r2[j]
            bm.faces.new((v1, v2, v3, v4))

    # Cap the top of main stem and arm
    def cap_ring(ring):
        center = sum((v.co for v in ring), Vector((0, 0, 0))) / len(ring)
        cv = bm.verts.new(center)
        for j in range(len(ring)):
            bm.faces.new((cv, ring[j], ring[(j + 1) % len(ring)]))

    cap_ring(main_rings[-1])
    cap_ring(arm_rings[-1])

    # Finalize Mesh
    bm.to_mesh("CactusMesh")
    obj = bpy.data.objects.new("ColumnarCactus", bpy.data.meshes["CactusMesh"])
    bpy.context.collection.objects.link(obj)
    bm.free()

    # Smooth Shading
    for poly in obj.data.polygons:
        poly.use_smooth = True

    # Apply a simple white material
    mat = bpy.data.materials.new(name="CactusWhite")
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = (1, 1, 1, 1)
        bsdf.inputs['Roughness'].default_value = 0.8
    obj.data.materials.append(mat)

    # Add a subtle subdivision surface modifier for the "base mesh" high fidelity look
    mod = obj.modifiers.new(name="Subdiv", type='SUBSURF')
    mod.levels = 1
    mod.render_levels = 2

if __name__ == "__main__":
    build_cactus()
