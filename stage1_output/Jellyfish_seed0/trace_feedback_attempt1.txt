import bpy
import bmesh
import math
import random
from mathutils import Vector

def clear_scene():
    """Clear all existing objects from the scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color, alpha=1.0, transmission=0.0):
    """Create a Principled BSDF material with specific transparency and color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    
    # Create shader node
    node_principled = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    
    node_principled.inputs['Base Color'].default_value = color
    node_principled.inputs['Alpha'].default_value = alpha
    node_principled.inputs['Transmission Weight'].default_value = transmission
    node_principled.inputs['Roughness'].default_value = 0.1
    
    # Connect nodes
    links = mat.node_tree.links
    links.new(node_principled.outputs['BSDF'], node_output.inputs['Surface'])
    
    # Set blend mode for transparency in EEVEE/Cycles
    mat.blend_method = 'BLEND' 
    return mat

def create_jellyfish_bell():
    """Create the lobed, translucent bell of the jellyfish."""
    # Create a UV sphere as the base
    bpy.ops.mesh.primitive_uv_sphere_add(segments=32, ring_count=16, radius=1.0)
    bell = bpy.context.active_object
    bell.name = "JellyfishBell"

    # Scale it to be more dome-like and slightly flattened
    bell.scale = (1.2, 1.2, 0.8)
    bpy.ops.object.transform_apply(scale=True)

    # Edit geometry for lobed effect and faceting
    bm = bmesh.new()
    bm.from_mesh(bell.data)
    
    # We only want the top half (the dome)
    # Remove vertices below a certain Z threshold to make it an open bell
    for v in bm.verts:
        if v.co.z < -0.2:
            bm.verts.remove(v)
    
    # Fix faces after removing vertices
    bmesh.ops.contextual_delete(bm, verts=set()) # cleanup any stray elements

    # Create the lobed rim and subtle facets
    num_lobes = 8
    lobe_amplitude = 0.15
    frequency = num_lobes
    
    for v in bm.verts:
        # Apply a sine wave distortion based on angle for the lobes
        angle = math.atan2(v.co.y, v.co.x)
        dist = (math.sin(angle * frequency) * lobe_amplitude)
        
        # Apply it more strongly to vertices closer to the bottom rim
        weight = 1.0 - (v.co.z + 0.2) / 1.0
        if weight < 0: weight = 0
        
        v.co.x += math.cos(angle) * dist * weight
        v.co.y += math.sin(angle) * dist * weight
        
        # Add subtle "faceted" jitter to the surface
        jitter = 0.015
        v.co.x += random.uniform(-jitter, jitter)
        v.co.y += random.uniform(-jitter, jitter)
        v.co.z += random.uniform(-jitter, jitter)

    bm.to_mesh(bell.data)
    bm.free()

    # Add a subdivision surface modifier to smooth it slightly while keeping some facets
    subsurf = bell.modifiers.new(name="Subdiv", type='SUBSURF')
    subsurf.levels = 1
    subsurf.render_levels = 2

    # Material: Pale pink-lavender, translucent
    mat_bell = create_material("BellMat", (0.9, 0.75, 0.85, 1.0), alpha=0.6, transmission=0.8)
    bell.data.materials.append(mat_bell)

    return bell

def create_tentacles():
    """Create numerous thin, wispy tentacles."""
    num_tentacles = 70
    tentacle_length = 5.0
    segments_per_tentacle = 15
    
    # Material: Translucent pink thread
    mat_tentacle = create_material("TentacleMat", (1.0, 0.6, 0.7, 1.0), alpha=0.4)

    tentacles_container = []

    for i in range(num_tentacles):
        # Random starting position on the bell rim
        angle = random.uniform(0, 2 * math.pi)
        r = random.uniform(0.8, 1.1)
        start_pos = Vector((math.cos(angle) * r, math.sin(angle) * r, -0.2))

        # Create a curve for the tentacle
        curve_data = bpy.data.curves.new(name=f"TentacleCurve_{i}", type='CURVE')
        curve_data.dimensions = '3D'
        curve_data.fill_mode = 'FULL'
        curve_data.bevel_depth = 0.005  # Extremely thin
        curve_data.bevel_resolution = 3

        spline = curve_data.splines.new('BEZIER')
        spline.bezier_points.add(segments_per_tentacle - 1)

        curr_pos = start_pos.copy()
        for j in range(segments_per_tentacle):
            p = spline.bezier_points[j]
            p.co = curr_pos
            
            # Create wispy, tangled movement
            noise_offset = Vector((
                random.uniform(-0.4, 0.4),
                random.uniform(-0.4, 0.4),
                -random.uniform(0.2, 0.6)
            ))
            curr_pos += noise_offset
            
            # Set handles for smooth curves
            p.handle_left = p.co - (noise_offset * 0.5)
            p.handle_right = p.co + (noise_offset * 0.5)

        obj = bpy.data.objects.new(f"Tentacle_{i}", curve_data)
        bpy.context.collection.objects.link(obj)
        obj.data.materials.append(mat_tentacle)
        tentacles_container.append(obj)

    return tentacles_container

def create_oral_arm():
    """Create a single thicker, ribbon-like central oral arm."""
    # Material: Slightly more opaque pink
    mat_arm = create_material("OralArmMat", (1.0, 0.5, 0.6, 1.0), alpha=0.7)

    curve_data = bpy.data.curves.new(name="OralArmCurve", type='CURVE')
    curve_data.dimensions = '3D'
    curve_data.fill_mode = 'FULL'
    curve_data.bevel_resolution = 4

    spline = curve_data.splines.new('BEZIER')
    segments = 20
    spline.bezier_points.add(segments - 1)

    # Parametric spiral: r decreases, z goes down, theta increases
    t_max = 6.0
    for i in range(segments):
        t = (i / segments) * t_max
        radius = 0.4 * math.exp(-t * 0.15) # Tapering width/reach
        x = radius * math.cos(t * 2.0)
        y = radius * math.sin(t * 2.0)
        z = - (t * 0.8)
        
        p = spline.bezier_points[i]
        p.co = Vector((x, y, z))
        
        # Dynamic bevel depth for ribbon-like tapering
        # Since we can't set per-point radius on a standard curve in basic API easily, 
        # we use the 'radius' property of the bezier point
        p.radius = 0.08 * (1.0 - (i / segments)) + 0.02
        
        p.handle_left = p.co - Vector((0.2, 0, 0.2))
        p.handle_right = p.co + Vector((0.2, 0, 0.2))

    # We need to set the object's bevel depth to use the point radii
    curve_data.bevel_factor = 0 # This tells Blender to use point radius for thickness
    # Wait, actually in Bpy, if we want varying thickness, 
    # we can't just use 'bevel_depth'. We must set 'bevel_factor' and 'radius'.
    # Correction: To have varied thickness, bevel_depth is the base, then point.radius multiplies it.
    curve_data.bevel_depth = 0.1 

    obj = bpy.data.objects.new("OralArm", curve_data)
    bpy.context.collection.objects.link(obj)
    obj.data.materials.append(mat_arm)
