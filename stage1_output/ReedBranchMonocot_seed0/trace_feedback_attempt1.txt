import bpy
import bmesh
import math
import random
from mathutils import Vector

def clear_scene():
    """Clear default Blender scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Create a basic material with a specific base color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
    return mat

def create_reed_inflorescence():
    # Parameters for the reed structure
    stem_radius = 0.015
    stem_height = 2.5
    plume_start_z = 1.4
    plume_end_z = 2.6
    plume_max_radius = 0.22
    plume_min_radius = 0.05 # Tapering to a point/small tip
    spikelet_count = 1200  # Number of feathery spikelets
    spikelet_length_min = 0.08
    spikelet_length_max = 0.18
    curvature_factor = 0.4 # How much the spikelets bend

    # Create BMesh for the whole structure to keep it as one object
    bm = bmesh.new()

    # --- 1. Main Stem ---
    # Create a cylinder for the stem
    bmesh.ops.create_cone(
        bm, 
        cap_ends=True, 
        segments=12, 
        radius1=stem_radius, 
        radius2=stem_radius * 0.8, 
        depth=stem_height
    )
    # Move stem so it starts at origin and goes up
    bmesh.ops.translate(bm, vec=Vector((0, 0, stem_height / 2)), verts=bm.verts)

    # --- 2. The Plume (Inflorescence) ---
    # We'll generate spikelets as sets of vertices and edges
    for _ in range(spikelet_count):
        # Distribute along the Z axis within the plume region
        t = random.random() # normalized position [0, 1]
        z_pos = plume_start_z + t * (plume_end_z - plume_start_z)
        
        # Radius tapers as we go higher in the cone
        current_max_r = plume_max_radius * (1.0 - t * 0.7) # Tapers to 30% of max radius
        
        # Random position on the central axis or slightly offset
        angle = random.uniform(0, 2 * math.pi)
        dist_from_center = random.uniform(0, stem_radius * 1.2)
        start_x = math.cos(angle) * dist_from_center
        start_y = math.sin(angle) * dist_from_center
        
        # Direction of the spikelet (outwards and slightly upwards)
        # Higher t (top of plume) means more concentrated angles
        dir_x = math.cos(angle)
        dir_y = math.sin(angle)
        dir_z = random.uniform(0.2, 0.6) # Upward lean
        
        direction = Vector((dir_x, dir_y, dir_z)).normalized()
        length = random.uniform(spikelet_length_min, spikelet_length_max)
        
        # Create the spikelet as a curved path of 3 vertices
        v1 = bm.verts.new((start_x, start_y, z_pos))
        
        mid_point = Vector((start_x, start_y, z_pos)) + direction * (length * 0.5)
        # Add some organic curvature/jitter to the midpoint
        jitter = Vector((random.uniform(-0.02, 0.02), random.uniform(-0.02, 0.02), random.uniform(-0.02, 0.02)))
        mid_point += jitter + (Vector((0, 0, 1)) * curvature_factor * length * 0.2)
        v2 = bm.verts.new(mid_point)
        
        end_point = Vector((start_x, start_y, z_pos)) + direction * length
        # Ensure the end point feels "feathery" by adding a slight curve
        end_point += jitter * 0.5
        v3 = bm.verts.new(end_point)
        
        bm.edges.new((v1, v2))
        bm.edges.new((v2, v3))

    # To give the spikelets volume without huge polycounts, we will 
    # use a skin modifier or convert edges to thin cylinders.
    # Since BMesh is raw geometry, let's finish it and then apply modifiers.
    
    mesh_data = bpy.data.meshes.new("ReedInflorescence")
    bm.to_mesh(mesh_data)
    bm.free()

    obj = bpy.data.objects.new("ReedInflorescence", mesh_data)
    bpy.context.collection.objects.link(obj)

    # --- 3. Volumetric Polish ---
    # We use a Skin modifier to give the edges thickness, then Subsurf to smooth them
    skin_mod = obj.modifiers.new(name="Skin", type='SKIN')
    
    # Adjust skin radius (the thickness of the spikelets)
    # The Skin modifier stores radius in the vertex data.
    for v in obj.data.vertices:
        # Stem vertices should be thicker, plume vertices thinner
        if v.co.z < plume_start_z:
            obj.data.vertices[v.index].skin_radius = [stem_radius * 2, stem_radius * 2]
        else:
            obj.data.vertices[v.index].skin_radius = [0.004, 0.004]

    subsurf_mod = obj.modifiers.new(name="Subdivision", type='SUBSURF')
    subsurf_mod.levels = 1

    # --- 4. Coloration ---
    # Muted green: a desaturated olive/sage tone
    muted_green = (0.35, 0.42, 0.28, 1.0) 
    mat = create_material("MutedGreen", muted_green)
    obj.data.materials.append(mat)

if __name__ == "__main__":
    clear_scene()
    create_reed_inflorescence()
