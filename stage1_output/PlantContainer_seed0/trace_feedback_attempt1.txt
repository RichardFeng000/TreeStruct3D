def clear_scene():
    """Removes all existing objects from the scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Creates a simple diffuse material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    
    # Principled BSDF for basic color and roughness
    node_bsdf = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_bsdf.inputs['Base Color'].default_value = color
    node_bsdf.inputs['Roughness'].default_value = 0.8
    
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    
    links = mat.node_tree.links
    links.new(node_bsdf.outputs['BSDF'], node_output.inputs['Surface'])
    return mat

def create_bowl():
    """Creates a wide, shallow bowl with a pastel purple exterior."""
    # Parameters
    radius = 5.0
    height = 1.5
    wall_thickness = 0.3
    
    bm = bmesh.new()
    
    # Create the outer shell as a revolved profile (Screw)
    # Profile: from center bottom to edge, then up and inward
    verts = []
    segments = 32
    
    # Define profile line
    profile_points = [
        (0, 0, 0),
        (radius * 0.9, 0, 0),
        (radius, 0, height),
        (radius - wall_thickness, 0, height + 0.2)
    ]
    
    # We'll use a simple approach: Create a cylinder, scale it, and delete the top face
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, location=(0,0,0))
    bowl = bpy.context.active_object
    bowl.name = "PlantContainer"
    
    # Squash and shape to make it a shallow bowl
    bowl.scale[2] = 0.3 # Flatten Z

    # Enter edit mode to remove top half and thicken
    bm = bmesh.new()
    bm.from_mesh(bowl.data)
    
    # Remove vertices above Z = 0 (essentially the top hemisphere)
    # Since we scaled the sphere, Z=0 is center. Let's keep Z < 0.1 and remove everything else
    # Actually, let's just delete faces where centroid z > 0
    faces_to_delete = [f for f in bm.faces if f.calc_center_median().z > 0]
    bmesh.ops.delete(bm, geom=faces_to_delete, context='FACES')
    
    # Add thickness using Solidify modifier later or manual extrusion
    bm.to_mesh(bowl.data)

    bowl.data.materials.append(purple_mat)
    
    return bowl

def create_soil():
    """Creates a sandy soil disk inside the bowl."""
    radius = 4.6 # Slightly smaller than bowl radius
    height = 1.2
    
    bm = bmesh.new()
    bmesh.ops.create_cone(bm, cap_ends=True, segments=64, radius1=radius, radius2=radius, depth=0.5)
    
    # Randomly jitter the top vertices to make it look "sandy/natural"
    for v in bm.verts:
        if v.co.z > 0:
            v.co.z += random.uniform(-0.15, 0.15)
            v.co.x += random.uniform(-0.1, 0.1)
            v.co.y += random.uniform(-0.1, 0.1)

    mesh = bpy.data.meshes.new("SoilMesh")
    bm.to_mesh(mesh)
    bm.free()
    
    soil_obj = bpy.data.objects.new("Soil", mesh)

def create_mushroom(location):
    """Creates a single mushroom plant at the given location."""
    # Randomize dimensions slightly
    stem_height = random.uniform(1.5, 3.0)
    cap_radius = random.uniform(0.6, 1.2)
    cap_thickness = 0.3
    
    # Create Stem
    bm_stem = bmesh.new()
    bmesh.ops.create_cone(bm_stem, cap_ends=True, segments=12, radius1=0.3, radius2=0.15, depth=stem_height)
    
    stem_mesh = bpy.data.meshes.new("StemMesh")
    bm_stem.to_mesh(stem_mesh)
    bm_stem.free()
    
    stem_obj = bpy.data.objects.new("MushroomStem", stem_mesh)
    bpy.context.collection.objects.link(stem_obj)
    
    # Position stem
    stem_obj.location = (location[0], location[1], stem_height / 2)
    
    # Create Cap
    bm_cap = bmesh.new()
    bmesh.ops.create_uv_sphere(bm_cap, u_segments=16, v_segments=16, radius=cap_radius)
    
    # Flatten the sphere into a cap shape
    for v in bm_cap.verts:
        v.co.z *= 0.4 # Squash it
        # Slightly jitter for organic look
        if v.co.z > 0:
            v.co.z += random.uniform(-0.05, 0.05)

    # Remove the bottom half of the sphere to make a dome
    faces_to_delete = [f for f in bm_cap.faces if f.calc_center_median().z < -0.1]
    bmesh.ops.delete(bm_cap, geom=faces_to_delete, context='FACES')

    cap_mesh = bpy.data.meshes.new("CapMesh")
    bm_cap.to_mesh(cap_mesh)
    bm_cap.free()
    
    cap_obj = bpy.data.objects.new("MushroomCap", cap_mesh)
    bpy.context.collection.objects.link(cap_obj)
    
    # Position cap on top of stem
    cap_obj.location = (location[0], location[1], stem_height)
    
    # Material: Off-white/Cream
    cream_mat = create_material("MushroomCream", (0.95, 0.9, 0.8, 1.0))
    stem_obj.data.materials.append(cream_mat)
    cap_obj.data.materials.append(cream_mat)
    
    # Smoothing
    bpy.context.view_layer.objects.active = stem_obj
    bpy.ops.object.shade_smooth()
    bpy.context.view_layer.objects.active = cap_obj
    bpy.ops.object.shade_smooth()
    
    return [stem_obj, cap_obj]
