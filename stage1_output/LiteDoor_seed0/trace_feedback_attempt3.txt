import bpy
import bmesh
import math
from mathutils import Vector

def clear_scene():
    """Clears the default Blender scene."""
    if bpy.context.object and bpy.context.object.mode != 'OBJECT':
        bpy.ops.object.mode_set(mode='OBJECT')
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_dark_navy_material():
    """Creates a dark navy material."""
    mat = bpy.data.materials.new(name="DarkNavy")
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    
    node_bsdf = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    
    # Dark Navy Color (RGBA) - very dark blue/black
    node_bsdf.inputs['Base Color'].default_value = (0.01, 0.03, 0.12, 1.0)
    node_bsdf.inputs['Roughness'].default_value = 0.35
    
    links = mat.node_tree.links
    links.new(node_bsdf.outputs['BSDF'], node_output.inputs['Surface'])
    
    return mat

def create_arched_door():
    # Parameters
    width = 0.9
    height_spring = 1.8  # Height where the arch begins
    thickness = 0.06
    radius = width / 2
    total_height = height_spring + radius
    
    lite_w = 0.35
    lite_h = 0.6
    lite_z_center = (height_spring + radius) * 0.6  # Upper-center position
    
    # Create Mesh and Object
    mesh = bpy.data.meshes.new("ArchedDoor")
    obj = bpy.data.objects.new("ArchedDoor", mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()
    
    # 1. Create the front profile vertices
    segments = 32
    profile_verts = []
    
    # Bottom edge
    v0 = bm.verts.new((-width/2, 0, 0))
    v1 = bm.verts.new((width/2, 0, 0))
    profile_verts.append(v0)
    profile_verts.append(v1)
    
    # Right side up to spring line
    v2 = bm.verts.new((width/2, 0, height_spring))
    profile_verts.append(v2)
    
    # Arch (top half circle) from right to left
    for i in range(segments):
        angle = (math.pi / 2) - (math.pi * i / segments)
        tx = radius * math.cos(angle)
        tz = height_spring + radius * math.sin(angle)
        profile_verts.append(bm.verts.new((tx, 0, tz)))
    
    # Final vertex of arch (left side at spring line) to close the loop before returning to v0
    v_last = bm.verts.new((-width/2, 0, height_spring))
    profile_verts.append(v_last)
    
    # Create the front face
    face = bm.faces.new(profile_verts)
    
    # 2. Extrude to create thickness
    # We use bmesh.ops.extrude_face. It returns a result containing the new geometry.
    res = bmesh.ops.extrude_face(bm, vec=Vector((0, thickness, 0)), faces=[face])
    
    # Center the door on Y axis so it's centered at origin
    for v in bm.verts:
        v.co.y -= thickness / 2
    
    bm.to_mesh(mesh)
    bm.free()

    # 3. Create the Lite (Panel Opening) and Inset using Booleans
    # We create a cutter for the main hole
    bpy.ops.mesh.primitive_cube_add(size=1.0)
    cutter = bpy.context.active_object
    cutter.name = "LiteCutter"
    cutter.scale = (lite_w, thickness * 3.0, lite_h) # Overlap Y for clean cut
    cutter.location = (0, 0, lite_z_center)
    
    bool_mod = obj.modifiers.new(name="LiteCut", type='BOOLEAN')
    bool_mod.operation = 'DIFFERENCE'
    bool_mod.object = cutter
    
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier="LiteCut")
    bpy.data.objects.remove(cutter, do_unlink=True)

    # 4. Create the Inset Frame (the "inset" part of the lite)
    # This is a slightly larger rectangle cut shallowly into the front face
    inset_depth = 0.015
    frame_w = lite_w + 0.04
    frame_h = lite_h + 0.04
    
    bpy.ops.mesh.primitive_cube_add(size=1.0)
    inset_cutter = bpy.context.active_object
    inset_cutter.name = "InsetCutter"
    # Scale it to be the frame size, and thickness only for the inset depth
    inset_cutter.scale = (frame_w, inset_depth * 2.0, frame_h)
    # Position it so it bites into the front face (Y-negative side of center if door is on Y)
    # The door spans from -thickness/2 to +thickness/2. 
    # We want it to cut from -thickness/2 up to -thickness/2 + inset_depth.
    inset_cutter.location = (0, -thickness/2 + (inset_depth / 2), lite_z_center)
    
    bool_mod_inset = obj.modifiers.new(name="InsetCut", type='BOOLEAN')
    bool_mod_inset.operation = 'DIFFERENCE'
    bool_mod_inset.object = inset_cutter
    
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier="InsetCut")
    bpy.data.objects.remove(inset_cutter, do_unlink=True)

    # 5. Final Polish: Bevel edges for realism
    bev = obj.modifiers.new(name="DetailBevel", type='BEVEL')
    bev.width = 0.004
    bev.segments = 3
    bpy.ops.object.modifier_apply(modifier="DetailBevel")

    return obj

def main():
    clear_scene()
    
    # Create material
    navy_mat = create_dark_navy_material()
    
    # Create the door
    door_obj = create_arched_door()
    
    # Assign material
    if not door_obj.data.materials:
        door_obj.data.materials.append(navy_mat)
    else:
        door_obj.data.materials[0] = navy_mat

if __name__ == "__main__":
    main()
