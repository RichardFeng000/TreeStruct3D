def clear_scene():
    """Clears all objects from the current scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Creates a simple principled BSDF material with a specific color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
    return mat

def create_board(width, depth, thickness, material):
    """Creates the main horizontal shelf board."""
    bpy.ops.mesh.primitive_cube_add(size=1)
    board = bpy.context.active_object
    board.name = "ShelfBoard"
    board.scale = (width, depth, thickness)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    board.data.materials.append(material)
    return board

def create_arch_bracket(width, height, thickness, material):
    """Creates a rounded arch-shaped bracket using bmesh."""
    bm = bmesh.new()
    
    # Parameters for the arch shape
    # We define it as a rectangle with a semi-circle on top
    radius = width / 2
    rect_height = height - radius
    
    # Create vertices for the outer profile of the arch
    # Bottom left, bottom right, and then points along the arc
    segments = 16
    verts = []
    
    # Bottom base
    verts.append(bm.verts.new((-radius, 0, 0)))
    verts.append(bm.verts.new((radius, 0, 0)))
    
    # Arc points
    for i in range(segments + 1):
        angle = math.pi * (i / segments) # 0 to pi
        x = radius * math.cos(angle)
        z = rect_height + radius * math.sin(angle)
        # To keep the arc on top, we use sin for z and cos for x
        # Wait, standard circle: x=r*cos, z=r*sin. 
        # For a top arch: angle from 0 to pi, center is (0, rect_height)
        # But vertices should go clockwise or counter-clockwise.
    
    # Let's redefine the arc points more cleanly
    arc_verts = []
    for i in range(segments + 1):
        angle = math.pi - (math.pi * (i / segments)) # pi down to 0
        vx = radius * math.cos(angle)
        vz = rect_height + radius * math.sin(angle)
        arc_verts.append(bm.verts.new((vx, 0, vz)))
    
    # Connect them into a face
    # Base line: vert[0] -> vert[1]
    # Then arc points...
    full_polygon = [bm.verts[0], bm.verts[1]] + arc_verts

    # I will make the board sit in XY, thickness in Z. Brackets will be in YZ plane.

    bmesh.ops.translate(bm, verts=bm.verts, vec=(0, 0, 0))
    
    # Extrude along the depth of the bracket (width of support)
    # We'll use the 'thickness' parameter as the depth of the bracket strip
    geom = bm.faces[:]
    res = bmesh.ops.extrude_face_region(bm, geom=geom)
    verts_extruded = [v for v in res['geom'] if isinstance(v, bmesh.types.BVVert)]

    mesh = bpy.data.meshes.new("ArchBracketMesh")
    bm.to_mesh(mesh)
    bm.free()
    
    obj = bpy.data.objects.new("Bracket", mesh)
    bpy.context.collection.objects.link(obj)
    obj.data.materials.append(material)
    return obj
