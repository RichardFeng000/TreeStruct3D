import bpy
import bmesh
import random
import math
from mathutils import Vector, Euler

def clear_scene():
    """Clear default objects from the scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Create a simple diffuse material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
    return mat

def create_gnarled_cylinder(name, start_pos, end_pos, radius_start, radius_end, segments=8, resolution=4, material=None):
    """Creates a cylinder between two points with random jitter for a 'gnarled' effect."""
    bm = bmesh.new()
    
    # Create vertices for the start and end caps
    dir_vec = (end_pos - start_pos).normalized()
    ortho_vec = Vector((0, 1, 0)) if abs(dir_vec.dot(Vector((0, 0, 1)))) < 0.9 else Vector((1, 0, 0))
    right = dir_vec.cross(ortho_vec).normalized()
    up = dir_vec.cross(right).normalized()

    verts_start = []
    verts_end = []

    for i in range(segments):
        angle = (2 * math.pi / segments) * i
        offset = (right * math.cos(angle) + up * math.sin(angle)) * radius_start
        verts_start.append(bm.verts.new((start_pos + offset)))

    for i in range(segments):
        angle = (2 * math.pi / segments) * i
        offset = (right * math.cos(angle) + up * math.sin(angle)) * radius_end
        # Add a bit of randomness to the end position for gnarled look
        jitter = Vector((random.uniform(-0.1, 0.1), random.uniform(-0.1, 0.1), random.uniform(-0.1, 0.1)))
        verts_end.append(bm.verts.new((end_pos + offset + jitter)))

    # Create faces connecting the caps
    for i in range(segments):
        v1 = verts_start[i]
        v2 = verts_start[(i + 1) % segments]
        v3 = verts_end[(i + 1) % segments]
        v4 = verts_end[i]
        bm.faces.new((v1, v2, v3, v4))

    # Fill caps
    bm.faces.new(verts_start)
    bm.faces.new(verts_end)

    obj = bpy.data.objects.new(name, bm.to_mesh())
    bpy.context.collection.objects.link(obj)
    bm.free()
    
    if material:
        obj.data.materials.append(material)
    
    return obj

def create_leaf(position, rotation, material):
    """Creates a small leaf mesh."""
    bm = bmesh.new()
    # Simple diamond shape for the leaf
    v1 = bm.verts.new((0, 0, 0))
    v2 = bm.verts.new((0, 0.3, 0))
    v3 = bm.verts.new((0.15, 0.15, 0))
    v4 = bm.verts.new((-0.15, 0.15, 0))
    bm.faces.new((v1, v3, v2, v4))
    
    mesh = bm.to_mesh("leaf")
    obj = bpy.data.objects.new("Leaf", mesh)
    bpy.context.collection.objects.link(obj)
    obj.location = position
    obj.rotation_euler = rotation
    obj.data.materials.append(material)
    bm.free()
    return obj

def create_flower_cluster(position, material):
    """Creates a small cluster of spheres for flowers."""
    for _ in range(random.randint(3, 6)):
        offset = Vector((random.uniform(-0.1, 0.1), random.uniform(-0.1, 0.1), random.uniform(-0.1, 0.1)))
        bpy.ops.mesh.primitive_uv_sphere_add(radius=0.05, location=position + offset)
        flower = bpy.context.active_object
        flower.data.materials.append(material)

def generate_tree():
    clear_scene()

    # Materials
    brown_mat = create_material("Brown", (0.2, 0.1, 0.05, 1.0))
    green_mat = create_material("Green", (0.1, 0.4, 0.1, 1.0))
    pink_mat = create_material("Pink", (0.9, 0.4, 0.6, 1.0))

    # Trunk: Short and thick
    trunk_start = Vector((0, 0, 0))
    trunk_end = Vector((0, 0, 1.2))
    trunk = create_gnarled_cylinder("Trunk", trunk_start, trunk_end, 0.5, 0.3, material=brown_mat)

    # Branching logic
    branch_endpoints = [trunk_end]
    num_primary_branches = 6
    
    for i in range(num_primary_branches):
        start = trunk_end
        angle_z = (2 * math.pi / num_primary_branches) * i
        angle_y = random.uniform(0.3, 1.0) # Outward spread for top-down view
        
        dir = Vector((math.cos(angle_z) * math.sin(angle_y), 
                     math.sin(angle_z) * math.sin(angle_y), 
                     math.cos(angle_y)))
        
        end = start + dir * random.uniform(1.5, 2.5)
        branch = create_gnarled_cylinder(f"Primary_{i}", start, end, 0.3, 0.15, material=brown_mat)
        branch_endpoints.append(end)

    # Secondary branches for a broader canopy
    secondary_endpoints = []
    for ep in branch_endpoints[1:]: # Skip the trunk bottom
        for j in range(3):
            start = ep
            dir = Vector((random.uniform(-1, 1), random.uniform(-1, 1), random.uniform(-0.5, 1))).normalized()
            end = start + dir * random.uniform(0.8, 1.5)
            create_gnarled_cylinder(f"Secondary_{j}", start, end, 0.15, 0.05, material=brown_mat)
            secondary_endpoints.append(end)

    # Canopy: Leaves and Flowers
    # We distribute leaves around the endpoints of secondary branches
    for ep in secondary_endpoints:
        # Create a cloud of leaves around each endpoint
        num_leaves = random.randint(15, 30)
        for _ in range(num_leaves):
            offset = Vector((random.uniform(-0.6, 0.6), 
                            random.uniform(-0.6, 0.6), 
                            random.uniform(-0.6, 0.6)))
            rot = Euler((random.uniform(0, 6.28), random.uniform(0, 6.28), random.uniform(0, 6.28)))
            create_leaf(ep + offset, rot, green_mat)
        
        # Occasional flower clusters
        if random.random() < 0.4:
            create_flower_cluster(ep + Vector((0,0,0.1)), pink_mat)

    # Final clean up - join meshes if necessary (optional, but here we keep separate for detail)
    # To avoid cluttering the outliner with thousands of objects, we can join them into a few groups.
    bpy.ops.object.select_all(action='DESELECT')
    for obj in bpy.data.objects:
        if "Leaf" in obj.name or "Primary" in obj.name or "Secondary" in obj.name or "Trunk" in obj.name:
            obj.select_set(True)
    # We don't join the flowers because they are separate spheres, but we could join leaves.
    # However, for procedural generation, keeping them as is for a few thousand verts is fine.

if __name__ == "__main__":
    generate_tree()
