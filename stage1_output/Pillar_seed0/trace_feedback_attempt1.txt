import bpy
import bmesh
import math
import random
from mathutils import Vector

def clear_scene():
    """Clears all objects from the current scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_plinth():
    """Creates a stepped rectangular base (plinth)."""
    steps = [
        {"w": 1.2, "d": 1.2, "h": 0.3}, # Bottom step
        {"w": 1.0, "d": 1.0, "h": 0.2}, # Middle step
        {"w": 0.8, "d": 0.8, "h": 0.15} # Top step
    ]
    
    current_z = 0
    objs = []
    
    for i, dim in enumerate(steps):
        bpy.ops.mesh.primitive_cube_add(size=1.0)
        obj = bpy.context.active_object
        obj.name = f"PlinthStep_{i}"
        obj.scale = (dim["w"], dim["d"], dim["h"])
        obj.location = (0, 0, current_z + dim["h"]/2)
        bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
        
        # Add some "weathering" by perturbing vertices of the base slightly
        bm = bmesh.new()
        bm.from_mesh(obj.data)
        for v in bm.verts:
            if random.random() > 0.7:
                v.co += Vector((random.uniform(-0.02, 0.02), 
                                random.uniform(-0.02, 0.02), 
                                random.uniform(-0.02, 0.02)))
        bm.to_mesh(obj.data)
        bm.free()
        
        objs.append(obj)
        current_z += dim["h"]
    
    return current_z

def create_fluted_shaft(start_z):
    """Creates a cylindrical shaft with vertical concave grooves (fluting)."""
    height = 6.0
    radius = 0.35
    flutes = 24
    depth = 0.03 # Depth of the groove
    
    # Create profile for the fluting
    bm = bmesh.new()
    
    # Generate a circle with concave segments
    verts = []
    for i in range(flutes * 2):
        angle = (math.pi * 2 * i) / (flutes * 2)
        x = math.cos(angle) * radius
        y = math.sin(angle) * radius
        
        # Every other vertex is the center of a flute (pushed inward)
        if i % 2 == 0:
            # Midpoint of flute - push in
            dist = radius - depth
            x = math.cos(angle) * dist
            y = math.sin(angle) * dist
        
        verts.append(bm.verts.new((x, y, 0)))

    # Connect vertices into a loop (face profile)
    for i in range(len(verts)):
        bm.edges.new((verts[i], verts[(i + 1) % len(verts)]))
    
    # Create the face for extrusion
    bm.faces.new(verts)
    
    # Extrude upwards to create height
    # We'll extrude in segments if we want a slight taper, 
    # but classical pillars usually have an entasis (slight bulge).
    segments = 10
    seg_height = height / segments
    
    last_face = bm.faces[0]
    for s in range(segments):
        # Calculate tapering/entasis: radius slightly larger in the middle
        t = s / segments
        # Simple entasis formula: subtle bulge at 1/3 height
        bulge = 0.02 * math.sin(math.pi * t)
        
        # Extrude
        extrude_result = bmesh.ops.extrude_face_region(bm, geom=[last_face])
        verts_new = [v for v in extrude_result['geom'] if isinstance(v, bmesh.types.BVVert)]
        
        # Move extruded vertices up and apply bulge
        for v in verts_new:
            v.co.z += seg_height
            # Apply radial push/pull for entasis
            direction = Vector((v.co.x, v.co.y, 0)).normalized()
            v.co += direction * bulge
            
        last_face = [f for f in extrude_result['geom'] if isinstance(f, bmesh.types.BVFace)][0]

    # Close the top and bottom
    # The initial face is at Z=0, the last extrusion creates the top face.
    
    # Shift entire shaft to start at start_z
    for v in bm.verts:
        v.co.z += start_z
        
    mesh = bpy.data.meshes.new("ShaftMesh")
    bm.to_mesh(mesh)
    bm.free()
    
    obj = bpy.data.objects.new("PillarShaft", mesh)
    bpy.context.collection.objects.link(obj)
    return obj, height

def create_capital(start_z):
    """Creates a decorative capital consisting of an echinus and abacus."""
    # Echinus (rounded transition)
    echinus_radius_bot = 0.38
    echinus_radius_top = 0.45
    echinus_height = 0.3
    
    bm = bmesh.new()
    
    # Create a torus-like bulge for the echinus
    res = 32
    for i in range(res):
        angle = (math.pi * 2 * i) / res
        x_bot = math.cos(angle) * echinus_radius_bot
        y_bot = math.sin(angle) * echinus_radius_bot
        x_top = math.cos(angle) * echinus_radius_top
        y_top = math.sin(angle) * echinus_radius_top
        
        v_bot = bm.verts.new((x_bot, y_bot, start_z))
        v_top = bm.verts.new((x_top, y_top, start_z + echinus_height))
        
        # Connect these vertically in the next step loop
    
    # Fix vertices and create faces for echinus
    all_verts = list(bm.verts)
    for i in range(res):
        v1 = all_verts[i*2] # bottom
        v2 = all_verts[(i*2 + 1)] # top
        v3 = all_verts[((i+1)%res)*2] # next bottom
        v4 = all_verts[((i+1)%res)*2 + 1] # next top
        bm.faces.new((v1, v3, v4, v2))

    # Create Abacus (Flat square slab)
    abacus_size = 0.6
    abacus_height = 0.15
    
    # We can use a separate mesh for the abacus or add to this bmesh
    # Adding to same bmesh for simplicity of "one object" feel, though normally separate is fine
    
    # Define corners of the square top
    s = abacus_size / 2
    z_top = start_z + echinus_height
    
    c1 = bm.verts.new((-s, -s, z_top))
    c2 = bm.verts.new((s, -s, z_top))
    c3 = bm.verts.new((s, s, z_top))
    c4 = bm.verts.new((-s, s, z_top))
    
    # Top face of abacus
    bm.faces.new((c1, c2, c3, c4))
    
    # Extrude top face upwards for thickness
    ext = bmesh.ops.extrude_face_region(bm, geom=[bm.faces[-1]])
    v_top_corners = [v for v in ext['geom'] if isinstance(v, bmesh.types.BVVert)]
    for v in v_top_corners:
        v.co.z += abacus_height

    # Finish and create object
    mesh = bpy.data.meshes.new("CapitalMesh")
    bm.to_mesh(mesh)
    bm.free()
    
    obj = bpy.data.objects.new("PillarCapital", mesh)
    bpy.context.collection.objects.link(obj)
    return obj

def main():
    clear_scene()
    
    # 1. Base
    z_after_base = create_plinth()
    
    # 2. Shaft
    shaft_obj, shaft_height = create_fluted_shaft(z_after_base)
    
    # 3. Capital
    create_capital(z_after_base + shaft_height)
    
    # Final touch: Combine into one object for convenience (optional)
    bpy.ops.object.select_all(action='SELECT')
    # We don't join so we can keep the geometric distinction, but ensure they are centered
    
    # Apply a subtle bevel to everything to reduce "CG sharpness" and look more weathered
    for obj in bpy.context.scene.objects:
        if obj.type == 'MESH':
            bevel = obj.modifiers.new(name="WeatheringBevel", type='BEVEL')
            bevel.width = 0.005
            bevel.segments = 2

if __name__ == "__main__":
    main()
