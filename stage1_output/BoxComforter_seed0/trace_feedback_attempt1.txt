import bpy
import bmesh
import mathutils
import random

def clear_scene():
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
    return mat

def create_comforter():
    # Parameters
    width = 2.2
    length = 3.0
    thickness = 0.05
    rows = 6
    cols = 8
    puffiness = 0.03
    
    # Materials
    mat_peach = create_material("PeachPink", (1.0, 0.75, 0.65, 1.0))
    mat_lavender = create_material("Lavender", (0.7, 0.6, 0.9, 1.0))
    mat_cream = create_material("Cream", (0.95, 0.92, 0.8, 1.0))
    mat_blue = create_material("PaleBlue", (0.7, 0.85, 0.95, 1.0))

    # Create the main body mesh
    bm = bmesh.new()
    
    # Generate grid vertices
    verts = []
    for i in range(rows + 1):
        row_verts = []
        for j in range(cols + 1):
            x = (j / cols - 0.5) * width
            y = (i / rows - 0.5) * length
            row_verts.append(bm.verts.new((x, y, 0)))
        verts.append(row_verts)

    # Create faces
    faces = []
    for i in range(rows):
        for j in range(cols):
            v1 = verts[i][j]
            v2 = verts[i+1][j]
            v3 = verts[i+1][j+1]
            v4 = verts[i][j+1]
            f = bm.faces.new((v1, v2, v3, v4))
            faces.append(f)

    # Add puffiness to each box segment
    # To do this properly with BMesh, we split faces or subdivide
    bmesh.ops.subdivide_edges(bm, edges=bm.edges, cuts=1, use_grid_fill=True)
    
    # Now identify the centers of the original boxes and push them up
    # We can do this by iterating over all vertices and checking if they are "centers"
    # Since we used a grid subdivide, center points are those not on boundaries and with specific indices
    for v in bm.verts:
        if abs(v.co.x) < width/2 - 0.01 and abs(v.co.y) < length/2 - 0.01:
            # Check if it's a cell center or edge center
            # A simple way to simulate "pillows" is using a sine-like wave or noise, 
            # but for box-stitch we want specific centers pushed up.
            if (round(v.co.x / (width/cols)*100) % 20 == 0 and 
                round(v.co.y / (length/rows)*100) % 20 == 0):
                 v.co.z += puffiness

    # Assign random material patches to faces
    material_pool = [mat_peach, mat_peach, mat_peach, mat_lavender, mat_cream, mat_blue]
    for f in bm.faces:
        if not f.material:
            f.material = random.choice(material_pool)

    # Finalize main mesh
    mesh_data = bpy.data.meshes.new("ComforterMesh")
    bm.to_mesh(mesh_data)
    bm.free()
    
    comforter_obj = bpy.data.objects.new("Comforter", mesh_data)
    bpy.context.collection.objects.link(comforter_obj)
    
    # Add modifiers for smoothness and thickness
    subsurf = comforter_obj.modifiers.new(name="Subdiv", type='SUBSURF')
    subsurf.levels = 2
    subsurf.rental_type = 'CATMULL_CLARK'
    
    solidify = comforter_obj.modifiers.new(name="Solidify", type='SOLIDIFY')
    solidify.thickness = thickness
    
    # Create the border strip (lavender)
    border_w = width + 0.1 # slight overlap
    border_h = 0.3
    
    bm_border = bmesh.new()
    # Rectangular slab for the border
    bmesh.ops.create_cube(bm_border, size=1.0)
    
    # Scale and position the border strip
    # It's a strip at one short end (Y max)
    # Target dimensions: width x height x thickness
    scale = mathutils.Vector((border_w/2, border_h/2, thickness/2))
    for v in bm_border.verts:
        v.co *= scale
        
    # Move to the end of the comforter and fold it slightly
    # The "folded" look is achieved by adding a bend or simply layering it
    offset_y = (length / 2) - (border_h / 4)
    for v in bm_border.verts:
        v.co.y += offset_y
        # Slight tilt to simulate fold
        if v.co.z > 0:
            v.co.y -= 0.02

    border_mesh = bpy.data.meshes.new("BorderMesh")
    bm_border.to_mesh(border_mesh)
    bm_border.free()
    
    border_obj = bpy.data.objects.new("BorderStrip", border_mesh)
    bpy.context.collection.objects.link(border_obj)
    
    # Material for border
    border_mat = create_material("LavenderBorder", (0.6, 0.5, 0.8, 1.0))
    border_obj.data.materials.append(border_mat)

    return comforter_obj

if __name__ == "__main__":
    clear_scene()
    create_comforter()
