import bpy
import bmesh
import math
from mathutils import Vector

def clear_scene():
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color, metallic=0.0, roughness=0.5):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    bsdf.inputs['Base Color'].default_value = color
    bsdf.inputs['Metallic'].default_value = metallic
    bsdf.inputs['Roughness'].default_value = roughness
    return mat

def create_box(name, w, h, d, loc=(0, 0, 0), rot=(0, 0, 0), mat=None):
    bpy.ops.mesh.primitive_cube_add(size=1.0, location=loc, rotation=rot)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = (w, d, h)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    if mat:
        obj.data.materials.append(mat)
    return obj

def create_wire_segment(name, start, end, radius, mat):
    # Create a cylinder to represent a wire
    direction = end - start
    length = direction.length
    if length == 0: return None
    
    midpoint = (start + end) / 2
    phi = math.atan2(direction.y, direction.x)
    theta = math.acos(direction.z / length)
    
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=length, location=midpoint)
    wire = bpy.context.active_object
    wire.name = name
    # Rotate cylinder to align with direction vector
    rot_quat = Vector((0, 0, 1)).rotation_difference(direction)
    wire.rotation_mode = 'QUATERNION'
    wire.rotation_quaternion = rot_quat
    if mat:
        wire.data.materials.append(mat)
    return wire

def create_dish_rack(loc, mat):
    # Container for the rack parts
    rack_container = bpy.data.objects.new("DishRackContainer", None)
    bpy.context.collection.objects.link(rack_container)
    
    rw, rd, rh = 0.54, 0.52, 0.60 # dimensions of the rack
    wire_r = 0.005
    
    # Base grid (X direction wires)
    for i in range(8):
        z = 0
        y = -rd/2 + (i * rd / 7)
        create_wire_segment(f"WireX_{i}", Vector((-rw/2, y, z)) + loc, Vector((rw/2, y, z)) + loc, wire_r, mat)
    
    # Base grid (Y direction wires)
    for i in range(8):
        z = 0
        x = -rw/2 + (i * rw / 7)
        create_wire_segment(f"WireY_{i}", Vector((x, -rd/2, z)) + loc, Vector((x, rd/2, z)) + loc, wire_r, mat)

    # Vertical slots for plates
    for i in range(6):
        x = -0.2 + (i * 0.08)
        z_top = rh * 0.5
        y_pos = -0.1
        create_wire_segment(f"Slot_{i}", Vector((x, y_pos, 0)) + loc, Vector((x, y_pos, z_top)) + loc, wire_r, mat)
        create_wire_segment(f"SlotB_{i}", Vector((x, -y_pos, 0)) + loc, Vector((x, -y_pos, z_top)) + loc, wire_r, mat)

    # Top frame rim
    z_top = rh * 0.5
