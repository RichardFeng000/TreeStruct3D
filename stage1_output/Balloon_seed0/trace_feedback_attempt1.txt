def clear_scene():
    """Clear all default objects from the scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color, metallic, roughness):
    """Create a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    
    bsdf.inputs['Base Color'].default_value = color
    bsdf.inputs['Metallic'].default_value = metallic
    bsdf.inputs['Roughness'].default_value = roughness
    return mat

def create_balloon_letter(char, index, navy_mat, vivid_mat):
    """Creates a single tubular balloon letter."""
    # Create Text object
    bpy.ops.object.text_add(location=(index * 1.2, 0, 0))
    obj = bpy.context.active_object
    obj.data.body = char
    
    # Set text properties for a rounded look
    obj.data.size = 1.0
    obj.data.extrude = 0.0  # We will use curve bevel instead for true tubularity
    
    # Convert to Curve to utilize the bevel_depth (tubular) property
    bpy.ops.object.convert(target='CURVE')
    curve = obj.data
    curve.fill_mode = 'NONE' # Only the outline, no filling
    curve.bevel_depth = 0.15
    curve.bevel_resolution = 6
    
    # Convert to Mesh for final geometry and subdivision
    bpy.ops.object.convert(target='MESH')
    mesh = obj.data
    
    # Apply Subdivision Surface modifier for the 'inflated' look
    subsurf = obj.modifiers.new(name="Subdiv", type='SUBSURF')
    subsurf.levels = 2
    subsurf.render_levels = 2
    
    # Shade smooth
    bpy.ops.object.shade_smooth()
    
    # Apply the modifier to make geometry real for material assignment by normal
    bpy.ops.object.modifier_apply(modifier="Subdiv")
    
    # Calculate centroid of this letter object to determine 'interior' faces
    # The interior are faces pointing towards the center of the letter's loops (like in O, B, A)
    centroid = Vector((0, 0, 0))
    verts = mesh.vertices
    for v in verts:
        centroid += v.co
    centroid /= len(verts)
    
    # Material assignment
    obj.data.materials.append(navy_mat) # Slot 0
    obj.data.materials.append(vivid_mat) # Slot 1
    
    for poly in mesh.polygons:
        # Calculate face center and normal
        face_center = (poly.vertices[0].co + poly.vertices[1].co + poly.vertices[2].co) / 3 if len(poly.vertices) >= 3 else poly.vertices[0].co
        normal = poly.normal
        
        # Vector from face center to centroid
        to_centroid = centroid - face_center
        
        # If the normal points significantly towards the centroid, it's likely an interior wall of a hole
        if normal.dot(to_centroid) > 0.4:
            poly.material_index = 1 # Vivid Blue
        else:
            poly.material_index = 0 # Navy Blue
            
    return obj

def main():
    clear_scene()
    
    # Setup materials
    # Dark navy-blue glossy outer surface
    navy_mat = create_material(
        "NavyGloss", 
        (0.01, 0.02, 0.1, 1.0), # Navy blue
        0.2, 
        0.15
    )
    
    # Vivid blue-tinted reflective interior
    vivid_mat = create_material(
        "VividReflect", 
        (0.0, 0.4, 1.0, 1.0), # Bright vivid blue
        0.9, 
        0.05
    )
    
    text_string = "BALLOONS"
    letters = []
    
    # Create each character as a separate object to handle local centroids for material logic
    for i, char in enumerate(text_string):
        letter_obj = create_balloon_letter(char, i, navy_mat, vivid_mat)
        letters.append(letter_obj)
        
    # Center the whole assembly at the origin
    total_width = len(text_string) * 1.2
    for obj in letters:
        obj.location.x -= total_width / 2
        
    # Join all letters into one coherent object
    bpy.ops.object.select_all(action='DESELECT')
    for obj in letters:
        obj.select_set(True)
    bpy.context.view_layer.objects.active = letters[0]
    bpy.ops.object.join()
    
    # Final cleanup to ensure only the object remains at origin
    final_obj = bpy.context.active_object
    final_obj.name = "BalloonSign"

if __name__ == "__main__":
    main()
