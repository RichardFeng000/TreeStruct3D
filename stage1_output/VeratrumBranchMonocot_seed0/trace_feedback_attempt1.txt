def clear_scene():
    """Clears the default scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Creates a simple diffuse material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    node_principled = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    node_principled.inputs['Base Color'].default_value = color
    mat.node_tree.links.new(node_principled.outputs['BSDF'], node_output.inputs['Surface'])
    return mat

def create_leaf_mesh(length, width, curve_factor=1.0, twist_factor=0.0):

def create_organic_leaf(name, length, width, curve=1.0, twist=0.0):
    """Revised leaf generation using a BMesh grid and deformation."""
    bm = bmesh.new()
    res_l = 15
    res_w = 4
    
    # Create a flat plane first
    for i in range(res_l + 1):
        t = i / res_l
        # Width taper: 0 -> width -> 0
        current_w = width * math.sin(t * math.pi)
        for j in range(res_w + 1):
            u = (j / res_w) - 0.5
            x = t * length
            y = u * current_w
            z = 0
            bm.verts.new(Vector((x, y, z)))

    # Indexing for faces
    for i in range(res_l):
        for j in range(res_w):
            v1 = i * (res_w + 1) + j
            v2 = i * (res_w + 1) + j + 1
            v3 = (i + 1) * (res_w + 1) + j + 1
            v4 = (i + 1) * (res_w + 1) + j
            bm.faces.new((bm.verts[v1], bm.verts[v2], bm.verts[v3], bm.verts[v4]))

    # Deform the leaf: Curve it and Twist it
    for v in bm.verts:
        t = v.co.x / length
        # Spine curve (Y-axis)
        bend = math.sin(t * math.pi * 0.5) * curve
        v.co.y += bend
        # Twist the leaf along X axis
        angle = t * twist
        rot_mat = Matrix.Rotation(angle, 4, 'X')
        # Only rotate relative to the spine (the current x position)
        offset = Vector((0, v.co.y - (math.sin(t * math.pi * 0.5) * curve), v.co.z))
        v.co.y = (rot_mat @ offset).y + (math.sin(t * math.pi * 0.5) * curve)
        v.co.z = (rot_mat @ offset).z

    obj = bpy.data.objects.new(name, bm)
    bpy.context.collection.objects.link(obj)
    bm.to_mesh(obj.data)
    bm.free()
    
    # Add subdivision for smoothness
    mod = obj.modifiers.new(name="Subsurf", type='SUBSURF')
    mod.levels = 1
    mod.render_levels = 2
    
    return obj

def create_stalk(height, radius):
    """Creates the main central stem."""
    bm = bmesh.new()
    # Use a cylinder with slightly tapered top
    bmesh.ops.create_cone(bm, cap_ends=True, segments=12, diameter1=radius*2, diameter2=radius*1.2, depth=height)
    
    # Move center to origin (base at 0)
    for v in bm.verts:
        v.co.z += height / 2
        
    obj = bpy.data.objects.new("Stalk", bm)
    bpy.context.collection.objects.link(obj)
    bm.to_mesh(obj.data)
    bm.free()
    return obj
