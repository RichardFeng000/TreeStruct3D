import bpy
import bmesh
import random
import math
from mathutils import Vector

def clear_scene():
    """Removes all default objects from the scene."""
    if "Cube" in bpy.data.objects:
        bpy.data.objects.remove(bpy.data.objects["Cube"], do_unlink=True)
    if "Camera" in bpy.data.objects:
        bpy.data.objects.remove(bpy.data.objects["Camera"], do_unlink=True)
    if "Light" in bpy.data.objects:
        bpy.data.objects.remove(bpy.data.objects["Light"], do_unlink=True)
    # Clear any remaining orphan data
    for mesh in bpy.data.meshes:
        bpy.data.meshes.remove(mesh)

def create_bowl():
    """
    Creates a wide shallow bowl with a rim using a screw modifier approach 
    via BMesh for precision.
    """
    # Define the profile of the bowl's wall and base (X=radius, Z=height)
    # The sequence follows the cross-section from center bottom to top outer edge, then back in.
    profile_points = [
        (0.0, 0.0),      # Inner bottom center
        (1.6, 0.0),      # Bottom inner edge
        (1.65, 0.05),    # Bottom outer corner
        (1.7, 0.3),      # Mid wall
        (2.1, 0.8),      # Top outer rim edge
        (1.8, 0.8),      # Top inner rim edge
        (1.6, 0.6),      # Interior wall slope
        (1.5, 0.1),      # Inside bottom corner
        (0.0, 0.1)       # Back to center (filling the base)
    ]

    mesh = bpy.data.meshes.new("BowlMesh")
    obj = bpy.data.objects.new("FruitBowl", mesh)
    bpy.context.collection.objects.link(obj)

    bm = bmesh.new()
    
    # Create vertices for the profile
    verts = []
    for p in profile_points:
        verts.append(bm.verts.new(p))
    
    # Connect them into an edge chain
    for i in range(len(verts) - 1):
        bm.edges.new((verts[i], verts[i+1]))

    bm.to_mesh(mesh)
    bm.free()

    # Apply Screw Modifier to rotate the profile 360 degrees
    screw_mod = obj.modifiers.new(name="Screw", type='SCREW')
    screw_mod.angle = 2 * math.pi
    screw_mod.steps = 64
    screw_mod.render_steps = 64
    screw_mod.axis = (0, 0, 1)

    # Apply the modifier to make it a real mesh for potential further adjustments
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier="Screw")
    
    return obj

def create_fruit(name, radius, location):
    """Creates a smooth spherical fruit."""
    bpy.ops.mesh.primitive_uv_sphere_add(
        radius=radius, 
        location=location, 
        segments=32, 
        ring_count=16
    )
    fruit = bpy.context.active_object
    fruit.name = name
    
    # Apply smooth shading
    for poly in fruit.data.polygons:
        poly.use_smooth = True
        
    return fruit

def populate_bowl():
    """Places several spherical fruits randomly inside the bowl."""
    num_fruits = 18
    # Bowl interior dimensions roughly: radius < 1.6, height approx 0.1 to 0.8
    inner_radius = 1.5
    min_z = 0.2
    max_z = 0.7

    fruits = []
    for i in range(num_fruits):
        # Random radius for variety
        r = random.uniform(0.15, 0.25)
        
        # Simple physics-ish placement: try to find a spot that doesn't overlap too much
        # but is within the bowl boundaries
        placed = False
        attempts = 0
        while not placed and attempts < 50:
            # Random position in circle
            angle = random.uniform(0, 2 * math.pi)
            dist = random.uniform(0, inner_radius - r)
            x = math.cos(angle) * dist
            y = math.sin(angle) * dist
            z = random.uniform(min_z, max_z)
            
            loc = Vector((x, y, z))
            
            # Check distance against already placed fruits to avoid extreme overlap
            too_close = False
            for other_r, other_loc in fruits:
                if (loc - other_loc).length < (r + other_r) * 0.8:
                    too_close = True
                    break
            
            if not too_close:
                fruits.append((r, loc))
                create_fruit(f"Fruit_{i}", r, loc)
                placed = True
            attempts += 1

def main():
    clear_scene()
    
    # Create the wide shallow bowl
    bowl = create_bowl()
    
    # Fill it with round fruits
    populate_bowl()
    
    # Ensure everything is centered and clean
    bpy.context.view_layer.update()

if __name__ == "__main__":
    main()
