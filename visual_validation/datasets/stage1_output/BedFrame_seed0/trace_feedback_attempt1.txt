def create_box(name, size, location):
    """Creates a simple box mesh."""
    mesh = bpy.data.meshes.new(name)
    obj = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()
    bmesh.ops.create_cube(bm, size=1.0)
    # Scale to desired dimensions
    for v in bm.verts:
        v.co.x *= size[0] / 2
        v.co.y *= size[1] / 2
        v.co.z *= size[2] / 2
    
    bm.to_mesh(mesh)
    bm.free()
    obj.location = location
    return obj

def create_tapered_leg(name, height, top_rad, bot_rad, location):
    """Creates a tapered leg with metallic accents."""
    mesh = bpy.data.meshes.new(name)
    obj = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()
    # Create cylinder for the leg
    bmesh.ops.create_cone(
        bm, 
        cap_ends=True, 
        segments=16, 
        radius1=bot_rad, 
        radius2=top_rad,

    )

    bm.free()

    return obj

def create_decorative_panel(name, width, height, depth, location):
    """Creates a panel with molded/carved detail."""
    mesh = bpy.data.meshes.new(name)
    obj = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()
    bmesh.ops.create_cube(bm, size=1.0)
    
    # Scale to main panel size
    for v in bm.verts:
        v.co.x *= width / 2
        v.co.y *= depth / 2
        v.co.z *= height / 2
    
    # Inset and extrude faces to create "carved" effect
    bm.faces.ensure_lookup_table()
    # Find the front face (positive Y)
    front_face = None
    for f in bm.faces:
        if f.normal.y > 0.9:
            front_face = f
            break
            
    if front_face:
        # Inset for border
        res = bmesh.ops.inset_individual(bm, faces=[front_face], thickness=0.05)
        inner_face = res['faces'][0]
        # Push inward to create the panel depth
        bmesh.ops.translate(bm, verts=inner_face.verts, vec=(0, -depth*0.3, 0))
        
        # Create a further inner detail
        res2 = bmesh.ops.inset_individual(bm, faces=[inner_face], thickness=0.03)
        inner_detail = res2['faces'][0]
        bmesh.ops.translate(bm, verts=inner_detail.verts, vec=(0, -depth*0.1, 0))

    bm.to_mesh(mesh)
    bm.free()
    obj.location = location
    return obj
