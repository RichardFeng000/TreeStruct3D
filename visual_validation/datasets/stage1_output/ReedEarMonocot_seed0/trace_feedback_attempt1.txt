import bpy
import bmesh
import math
import random
from mathutils import Vector, Euler

def clear_scene():
    """Removes all default objects from the scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Creates a simple diffuse material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
    return mat

def create_floret_mesh():
    """Creates a single scale-like floret using bmesh."""
    bm = bmesh.new()
    
    # Create a teardrop/scale shape
    # Base circle (small)
    verts = []
    segments = 6
    for i in range(segments):
        angle = (2 * math.pi / segments) * i
        verts.append(bm.verts.new(Vector((math.cos(angle)*0.01, math.sin(angle)*0.01, 0))))
    
    # Create the base face
    bm.faces.new(verts)
    
    # Extrude and taper to a point
    # We create layers of vertices moving up Z
    prev_layer = verts
    layers = 5
    for l in range(1, layers + 1):
        curr_layer = []
        z_pos = l * 0.02
        scale = 1.0 - (l / layers)
        # Add some curvature by offsetting the X center slightly as we go up
        offset_x = (l**2) * 0.005 
        
        for i in range(segments):
            angle = (2 * math.pi / segments) * i
            v = bm.verts.new(Vector((
                math.cos(angle) * 0.01 * scale + offset_x, 
                math.sin(angle) * 0.01 * scale, 
                z_pos
            )))
            curr_layer.append(v)
        
        # Bridge between layers
        for i in range(segments):
            next_i = (i + 1) % segments
            bm.faces.new((prev_layer[i], prev_layer[next_i], curr_layer[next_i], curr_layer[i]))
            
        prev_layer = curr_layer

    # Close the tip if not already closed (the last layer is very small, but for a seed head we want it sharp)
    if len(prev_layer) > 0:
        # Merge the last vertices into one point to make it a sharp tip
        tip_v = bm.verts.new(Vector((prev_layer[0].co.x, prev_layer[0].co.y, prev_layer[0].co.z + 0.01)))
        # This is simplified; in bmesh we just create faces from the last layer to this tip
        for i in range(segments):
            next_i = (i + 1) % segments
            bm.faces.new((prev_layer[i], prev_layer[next_i], tip_v))

    bm.verts.ensure_lookup_table()
    bm.faces.ensure_lookup_table()
    
    mesh = bpy.data.meshes.new("Floret")
    bm.to_mesh(mesh)
    bm.free()
    return mesh

def generate_reed_ear():
    clear_scene()
    
    # Parameters
    num_florets = 280
    spike_height = 1.8
    stem_radius = 0.012
    golden_angle = math.radians(137.5)
    bend_intensity = 0.15
    
    # Create materials
    pale_green = (0.4, 0.6, 0.3, 1.0)
    mat = create_material("ReedGreen", pale_green)
    
    # Central axis stem
    bm_stem = bmesh.new()
    # Create a thin cylinder for the rachis
    bmesh.ops.create_cone(bm_stem, cap_ends=True, segments=8, diameter=stem_radius*2, depth=spike_height)
    
    # We'll move the stem and florets together into one object
    # To make the posture "slightly angled", we'll define a path for them
    # Path: x = sin(z * 1.5) * bend_intensity, y = cos(z * 1.5) * bend_intensity / 2
    
    floret_mesh = create_floret_mesh()
    floret_obj = bpy.data.objects.new("FloretTemplate", floret_mesh)
    # We won't link it to the scene, just use its data
    
    # Final combined object
    final_bm = bmesh.new()
    
    # 1. Build the stem geometry based on the curve path
    # Since create_cone is straight, we manually bend the vertices of the stem
    stem_verts = []
    segments_z = 20
    for i in range(segments_z + 1):
        z = (i / segments_z) * spike_height
        off_x = math.sin(z * 1.2) * bend_intensity
        off_y = math.cos(z * 1.2) * (bend_intensity * 0.5)
        # Create a ring of vertices at this height
        for j in range(8):
            angle = (2 * math.pi / 8) * j
            v = final_bm.verts.new(Vector((
                off_x + math.cos(angle) * stem_radius,
                off_y + math.sin(angle) * stem_radius,
                z
            )))
            stem_verts.append(v)
            
    # Face the stem vertices
    for i in range(segments_z):
        layer_start = i * 8
        next_layer_start = (i + 1) * 8
        for j in range(8):
            n_j = (j + 1) % 8
            final_bm.faces.new((
                stem_verts[layer_start + j],
                stem_verts[layer_start + n_j],
                stem_verts[next_layer_start + n_j],
                stem_verts[next_layer_start + j]
            ))

    # 2. Distribute Florets along the curve
    # We use a separate BMesh for florets to avoid complex vertex tracking, then join
    for i in range(num_florets):
        t = i / num_florets # normalized height [0, 1]
        z = t * spike_height
        
        # Curve position of the center axis
        pos_x = math.sin(z * 1.2) * bend_intensity
        pos_y = math.cos(z * 1.2) * (bend_intensity * 0.5)
        center = Vector((pos_x, pos_y, z))
        
        # Phyllotaxis rotation
        angle = i * golden_angle
        
        # Scaling: slightly smaller at the tip
        scale_factor = 1.0 - (t * 0.4)
        
        # Create floret geometry manually into final_bm
        # To avoid overhead, we'll use a small set of offsets based on the template floret mesh
        # But for procedural accuracy and speed in Blender Python, we can just duplicate the BMesh logic
        
        # Local coordinates for one floret (mimicking create_floret_mesh)
        floret_verts = []
        # Base ring
        for j in range(6):
            a = (2 * math.pi / 6) * j
            v_local = Vector((math.cos(a)*0.01, math.sin(a)*0.01, 0))
            # Rotate around Z
            rot_z = Euler((0, 0, angle), 'XYZ')
            v_rotated = rot_z @ v_local
            # Scale and Translate
            v_final = center + (v_rotated * scale_factor)
            floret_verts.append(final_bm.verts.new(v_final))
        
        # Florets are usually angled upwards and outwards
        # We'll add the "body" of the floret
        prev_layer = floret_verts
        for l in range(1, 5):
            curr_layer = []
            z_off = l * 0.02 * scale_factor
            s_off = (1.0 - l/5) * scale_factor
            curve_off = (l**2) * 0.004 * scale_factor
            
            for j in range(6):
                a = (2 * math.pi / 6) * j
                v_local = Vector((math.cos(a)*0.01*s_off + curve_off, math.sin(a)*0.01*s_off, z_off))
                rot_z = Euler((0, 0, angle), 'XYZ')
                # Also tilt the floret slightly outwards
                rot_tilt = Euler((0.2, 0, 0), 'XYZ') 
                v_rotated = rot_z @ (rot_tilt @ v_local)
                v_final = center + v_rotated
                curr_layer.append(final_bm.verts.new(v_final))
            
            for j in range(6):
                n_j = (j + 1) % 6
                final_bm.faces.new((prev_layer[j], prev_layer[n_j], curr_layer[n_j], curr_layer[j]))
            prev_layer = curr_layer

        # Tip of floret
        tip_v = final_bm.verts.new(center + (Euler((0, 0, angle), 'XYZ') @ Vector((0, 0, 0.1 * scale_factor))))
        for j in range(6):
            n_j = (j+1)%6
            final_bm.faces.new((prev_layer[j], prev_layer[n_j], tip_v))

    # Finalize mesh
    mesh_data = bpy.data.meshes.new("ReedEar")
    final_bm.to_mesh(mesh_data)
    final_bm.free()
    
    obj = bpy.data.objects.new("ReedEar", mesh_data)
    bpy.context.collection.objects.link(obj)
    
    # Assign material
    obj.data.materials.append(mat)
    
    # Set final rotation to make it slightly angled as per request
    obj.rotation_euler = Euler((0.2, 0, 0), 'XYZ')

if __name__ == "__main__":
