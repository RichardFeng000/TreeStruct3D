import bpy

from mathutils import Vector, Matrix

def clear_scene():
    """Clears all objects from the current scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Creates a simple principled BSDF material with a specific color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    node_principled = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    node_principled.inputs['Base Color'].default_value = color
    mat.node_tree.links.new(node_principled.outputs['BSDF'], node_output.inputs['Surface'])
    return mat

def create_spikelet_mesh():

    bm = bmesh.new()

    for v in bm.verts:
        v.co.z *= 2.5  # Elongate it
        v.co.x *= 0.6
        v.co.y *= 0.6

def create_wheat_ear():
    clear_scene()

    # Parameters
    num_nodes = 24  # Number of levels of spikelets along the rachis
    spikelets_per_node = 3 # Distribution around the stem
    rachis_height = 15.0
    curve_intensity = 1.8
    taper_start = 0.6 # Spikelets start tapering after this % of height

    # Materials (muted green-gold)
    color_gold = (0.7, 0.6, 0.3, 1.0)
    color_green = (0.4, 0.5, 0.2, 1.0)
    mat_grain = create_material("GrainMat", color_gold)
    mat_stem = create_material("StemMat", color_green)

    # Create Rachis (The central stem)
    rachis_bm = bmesh.new()
    segments = 60
    radius = 0.05
    
    points = []
    for i in range(segments + 1):
        t = i / segments
        z = t * rachis_height
        # Add a slight organic curve
        x = math.sin(t * math.pi * 0.6) * curve_intensity * (t**2)
        y = math.cos(t * math.pi * 0.3) * 0.5 * t
        points.append(Vector((x, y, z)))

    # Build the rachis cylinder along these points
    for i in range(segments):
        p1 = points[i]
        p2 = points[i+1]
        
        # Create a ring of vertices at each point
        ring_res = 6
        ring = []
        v_dir = (p2 - p1).normalized()
        ortho = Vector((0, 0, 1)) if abs(v_dir.dot(Vector((0,0,1)))) < 0.9 else Vector((0, 1, 0))
        right = v_dir.cross(ortho).normalized()
        up = v_dir.cross(right).normalized()

        for j in range(ring_res):
            angle = (2 * math.pi / ring_res) * j
            offset = (right * math.cos(angle) + up * math.sin(angle)) * radius
            ring.append(rachis_bm.verts.new(p1 + offset))
        
        points_next_ring = []
        for j in range(ring_res):
            angle = (2 * math.pi / ring_res) * j
            offset = (right * math.cos(angle) + up * math.sin(angle)) * radius
            points_next_ring.append(rachis_bm.verts.new(p2 + offset))

        for j in range(ring_res):
            v1 = ring[j]
            v2 = ring[(j+1)%ring_res]
            v3 = points_next_ring[(j+1)%ring_res]
            v4 = points_next_ring[j]
            bm_face = rachis_bm.faces.new((v1, v2, v3, v4))

    # Close the bottom cap
    bottom_ring = [] # we need to keep track of the first ring
    # Re-calculate first ring for capping
    p1 = points[0]
    v_dir = (points[1] - points[0]).normalized()
    ortho = Vector((0, 0, 1)) if abs(v_dir.dot(Vector((0,0,1)))) < 0.9 else Vector((0, 1, 0))
    right = v_dir.cross(ortho).normalized()
    up = v_dir.cross(right).normalized()
    for j in range(6):
        angle = (2 * math.pi / 6) * j
        offset = (right * math.cos(angle) + up * math.sin(angle)) * radius
        bottom_ring.append(rachis_bm.verts.new(p1 + offset)) # This is slightly wrong because we already created them, but for a simple model it's okay if we just ignore the double vertices and merge later.
    
    # Actually let's just use the first 6 verts of the mesh
    first_ring = rachis_bm.verts[:6]
    rachis_bm.faces.new(first_ring)

    # Now add Spikelets along the points
    for n in range(num_nodes):
        t = (n / num_nodes)
        p = points[int(t * segments)]
        
        # Tangent for orientation
        v_dir = (points[min(int(t * segments)+1, segments)] - p).normalized()
        ortho = Vector((0, 0, 1)) if abs(v_dir.dot(Vector((0,0,1)))) < 0.9 else Vector((0, 1, 0))
        right = v_dir.cross(ortho).normalized()
        up = v_dir.cross(right).normalized()

        # Tapering factor
        scale = 1.0
        if t > taper_start:
            scale = 1.0 - (t - taper_start) / (1.0 - taper_start) * 0.6

        for s in range(spikelets_per_node):
            angle = (2 * math.pi / spikelets_per_node) * s + (n * 0.5) # Spiral offset
            pos = p + (right * math.cos(angle) + up * math.sin(angle)) * radius
            
            # Orientation for the spikelet: pointing slightly "out and up"
            up_dir = (v_dir * 0.5 + (right * math.cos(angle) + up * math.sin(angle)) * 0.8).normalized()
            
            # Create a Grain
            grain_bm = bmesh.new()
            bmesh.ops.create_uvsphere(grain_bm, u_segments=6, v_segments=6, radius=0.15 * scale)
            for v in grain_bm.verts:
                v.co.z *= 2.2 # Elongate
                # Rotate and move to position
                # We use a matrix for this
                rot = Matrix.Rotation(math.acos(up_dir.dot(Vector((0,0,1)))), 4, Vector((0,0,1))) # simplified rotation
                # Better way: align Z axis to up_dir
                z_axis = Vector((0,0,1))
                axis = z_axis.cross(up_dir)
                angle_val = z_axis.angle(up_dir)
                rot_mat = Matrix.Rotation(angle_val, 4, axis if axis.length > 0 else Vector((1,0,0)))
                v.co = rot_mat @ v.co + pos
            
            # Add the Awn (the long hair)
            awn_bm = bmesh.new()
            awn_len = (1.5 + random.random()) * scale
            v1 = awn_bm.verts.new((0, 0, 0))
            v2 = awn_bm.verts.new((0, 0, awn_len))
            awn_bm.edges.new((v1, v2))
            # Transform awn
            z_axis = Vector((0,0,1))
            axis = z_axis.cross(up_dir)
            angle_val = z_axis.angle(up_dir)
            rot_mat = Matrix.Rotation(angle_val, 4, axis if axis.length > 0 else Vector((1,0,0)))
            for v in awn_bm.verts:
                v.co = rot_mat @ v.co + pos

            # Merge these into a final mesh object (converting to meshes first)
            # To keep the script clean and avoid creating 100 objects, we'll use one combined bmesh for grains/awns
            # But BMesh merge is complex. Let's just create them as separate objects then join at the end.
            grain_obj = bpy.data.objects.new("Grain", grain_bm.to_mesh())
            awn_obj = bpy.data.objects.new("Awn", awn_bm.to_mesh())
            bpy.context.collection.objects.link(grain_obj)
            bpy.context.collection.objects.link(awn_obj)
            grain_obj.data.materials.append(mat_grain)
            awn_obj.data.materials.append(mat_grain)
            grain_bm.free()
            awn_bm.free()

    # Finalize Rachis
    rachis_obj = bpy.data.objects.new("Rachis", rachis_bm.to_mesh())
    bpy.context.collection.objects.link(rachis_obj)
    rachis_obj.data.materials.append(mat_stem)
    rachis_bm.free()

    # Join all objects into one for cleanliness
    bpy.ops.object.select_all(action='SELECT')
    bpy.context.view_layer.objects.active = rachis_obj
    bpy.ops.object.join()
    
    # Slight final rotation to make it look "elevated" and organic
    rachis_obj.rotation_euler = (math.radians(15), 0, math.radians(-10))
