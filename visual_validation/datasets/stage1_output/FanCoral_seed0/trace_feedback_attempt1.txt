def clear_scene():
    """Clears the default scene of any objects."""
    bpy.ops.object.select_all(action='SELECT')

def create_material(name, color):
    """Creates a simple material with a base color (no textures)."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
    return mat

def rotate_vector(vec, axis, angle):
    """Rotates a vector around a given axis by an angle in radians."""
    import mathutils
    rot = mathutils.Matrix.Rotation(angle, 4, axis)
    return rot @ vec

def generate_coral_structure():
    """Generates the geometric structure of the sea fan coral."""
    # Parameters
    ROOT_HEIGHT = 1.2
    ROOT_RADIUS_BASE = 0.15
    ROOT_RADIUS_TOP = 0.1
    
    ITERATIONS = 6
    INITIAL_LENGTH = 1.0
    BIFURCATION_ANGLE = math.radians(45)
    LENGTH_DECAY = 0.78
    RADIUS_DECAY = 0.72
    PLANAR_STRICTNESS = 0.1  # How much it deviates from the XZ plane (small value = more planar)

    # To store all segments as (start, end, start_radius, end_radius)
    segments = []
    
    # Root segment
    root_start = Vector((0, 0, 0))
    root_end = Vector((0, 0, ROOT_HEIGHT))
    segments.append((root_start, root_end, ROOT_RADIUS_BASE, ROOT_RADIUS_TOP))

    def grow(start, direction, length, radius, depth):
        if depth <= 0:
            return [start]
        
        # Calculate end point
        end = start + direction * length
        new_radius = radius * RADIUS_DECAY
        segments.append((start, end, radius, new_radius))
        
        # Create two diverging directions
        # We want the fan to be roughly in XZ plane (direction mostly Z and X)
        # The growth is primarily "up" and "out"
        
        # Define a local coordinate system for branching
        # Base direction is 'direction'. We need two orthogonal vectors.
        ortho1 = Vector((0, 1, 0)) # Y axis as the "plane normal"
        if abs(direction.dot(ortho1)) > 0.9:
            ortho1 = Vector((1, 0, 0))
        
        right = direction.cross(ortho1).normalized()
        up_local = right.cross(direction).normalized()
        
        # Randomize angles slightly for organic feel
        angle1 = BIFURCATION_ANGLE + random.uniform(-0.2, 0.2)
        angle2 = -BIFURCATION_ANGLE + random.uniform(-0.2, 0.2)
        
        # Add a small amount of "out-of-plane" jitter (Y axis)
        jitter1 = Vector((0, 1, 0)) * random.uniform(-PLANAR_STRICTNESS, PLANAR_STRICTNESS)
        jitter2 = Vector((0, 1, 0)) * random.uniform(-PLANAR_STRICTNESS, PLANAR_STRICTNESS)

        # Calculate new directions
        # Branch 1: rotate direction around 'ortho1' then add jitter
        dir1 = rotate_vector(direction, ortho1, angle1).normalized() + jitter1
        dir1 = dir1.normalized()
        
        # Branch 2: rotate direction around 'ortho1' the other way
        dir2 = rotate_vector(direction, ortho1, angle2).normalized() + jitter2
        dir2 = dir2.normalized()
        
        points = []
        points.extend(grow(end, dir1, length * LENGTH_DECAY, new_radius, depth - 1))
        points.extend(grow(end, dir2, length * LENGTH_DECAY, new_radius, depth - 1))
        return points

    # Start growing from the top of the root
    endpoints = grow(root_end, Vector((0, 0, 1)), INITIAL_LENGTH, ROOT_RADIUS_TOP, ITERATIONS)
    
    # Create a lattice-like net by adding cross-links between nearby endpoints
    # We only link vertices that are within a certain distance and at similar depths
    all_nodes = []
    for seg in segments:
        all_nodes.append(seg[0])
        all_nodes.append(seg[1])
    
    unique_nodes = list(set(tuple(v) for v in all_nodes))
    unique_nodes = [Vector(v) for v in unique_nodes]
    
    cross_link_dist = INITIAL_LENGTH * 0.6
    for i in range(len(unique_nodes)):
        for j in range(i + 1, len(unique_nodes)):
            p1 = unique_nodes[i]
            p2 = unique_nodes[j]
            dist = (p1 - p2).length
            if dist < cross_link_dist and dist > 0.1:
                # Add a thin connecting branch
                avg_radius = ROOT_RADIUS_TOP * 0.3
                segments.append((p1, p2, avg_radius, avg_radius))

    return segments

def build_mesh_from_segments(segments):
    """Converts segment data into actual BMesh geometry."""
    bm = bmesh.new()
    
    # To avoid overlapping vertices at joints, we'll use a dictionary to map positions
    vert_map = {}

    def get_vert(pos):
        pos_tuple = tuple(round(c, 5) for c in pos)
        if pos_tuple not in vert_map:
            vert_map[pos_tuple] = bm.verts.new(pos)
        return vert_map[pos_tuple]

    # We can't simply use edges if we want thickness; 
    # we need to build tubes for each segment.
    # However, creating thousands of cylinders is slow. 
    # Instead, we create a skeleton and then apply the Skin modifier.
    for start, end, r1, r2 in segments:
        v1 = get_vert(start)
        v2 = get_vert(end)
        if v1 != v2:
            bm.edges.new((v1, v2))

    # Finalize the mesh object
    mesh = bpy.data.meshes.new("SeaFanCoral")
    bm.to_mesh(mesh)
    bm.free()
    
    obj = bpy.data.objects.new("SeaFanCoral", mesh)
    bpy.context.collection.objects.link(obj)
    
    # Apply Skin Modifier for thickness
    skin_mod = obj.modifiers.new(name="Skin", type='SKIN')
    
    # Adjust vertex radii to match the intended tapering
    # In Blender, skin radius is stored in a specific way (vertex data)
    # We need to iterate through vertices and set their size.
    for v in obj.data.vertices:
        # To simulate variable thickness, we map the vertex position 
        # back to our segments or use height as a proxy for taper.
        z = v.co.z
        dist_from_center = (Vector((v.co.x, v.co.y, 0)).length)
        
        # Base is thickest, tips are thinnest
        # Radius formula: base + decay based on height and distance from origin
        radius = 0.15 * (1.0 / (1.0 + z * 0.8)) * (1.0 / (1.0 + dist_from_center * 0.5))
        # Ensure it doesn't get too thin or too thick
        radius = max(0.01, min(0.15, radius))
        
        # Set skin radius (this is stored in the vertex weights/data)
        v.skin_radius = [radius, radius] # For Blender 4.x/5.0

    # Apply Subdivision Surface for organic look
    subdiv = obj.modifiers.new(name="Subdiv", type='SUBSURF')
    subdiv.levels = 2
    subdiv.render_levels = 3
    
    return obj
