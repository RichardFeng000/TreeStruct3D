import bpy
import bmesh
import math
import random
from mathutils import Vector

def clear_scene():
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    node_principled = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    node_principled.inputs['Base Color'].default_value = color
    mat.node_tree.links.new(node_principled.outputs['BSDF'], node_output.inputs['Surface'])
    return mat

def create_wing(name, position, rotation, scale=(1.0, 2.5, 1.0)):
    """Creates a dragonfly wing with a geometric vein network."""
    mesh = bpy.data.meshes.new(name)
    obj = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(obj)

    bm = bmesh.new()
    
    # Wing Shape: Elliptical plane
    segments = 32
    rings = 32
    for i in range(segments + 1):
        u = (i / segments) * math.pi * 2
        x = math.cos(u) * scale[0]
        y = math.sin(u) * scale[1]
        bm.verts.new((x, y, 0))

    # This is a bit crude for an oval, let's use a parametric approach
    bm.verts.clear()
    
    # Define boundary vertices for the wing
    res = 40
    for i in range(res):
        angle = (i / res) * math.pi * 2
        vx = math.cos(angle) * scale[0]
        vy = math.sin(angle) * scale[1]
        bm.verts.new((vx, vy, 0))
    
    # Create the main face (the membrane)
    center_vert = bm.verts.new((0, 0, 0))
    for i in range(res):
        v1 = bm.verts[i]
        v2 = bm.verts[(i + 1) % res]
        bm.faces.new((center_vert, v1, v2))

    # Create the veins (Geometric lines)
    # Radial veins
    for i in range(res):
        # We already have edges from center to boundary
        pass
        
    # Cross-veins (Connecting radial veins)
    # To simulate a network, we'll add random internal vertices and connect them
    internal_verts = []
    num_internal = 80
    for _ in range(num_internal):
        r = random.random()
        angle = random.random() * math.pi * 2
        vx = r * math.cos(angle) * scale[0]
        vy = r * math.sin(angle) * scale[1]
        internal_verts.append(bm.verts.new((vx, vy, 0)))

    # Connect internal vertices to nearest neighbors and boundaries
    for v in internal_verts:
        # Connect to boundary
        nearest_boundary = min(bm.verts[:res], key=lambda bv: (bv.co - v.co).length)
        bm.edges.new((v, nearest_boundary))
        
        # Connect to a few other random internal vertices
        targets = random.sample(internal_verts, 2)
        for t in targets:
            if (t - v).length < scale[1] * 0.4:
                bm.edges.new((v, t))

    bm.to_mesh(mesh)
    bm.free()

    obj.location = position
    obj.rotation_euler = rotation
    return obj

def create_thorax():
    # Bulbous thorax
    bpy.ops.mesh.primitive_uv_sphere_add(radius=1, location=(0, 0, 0))
    thorax = bpy.context.active_object
    thorax.name = "Thorax"
    thorax.scale = (0.6, 0.8, 0.7)
    
    # Add fine hairs using a particle-like approach with geometry
    bm = bmesh.new()
    bmesh.ops.create_uvsphere(bm, u_segments=32, v_segments=16, radius=1)
    
    hairs = []
    for v in bm.verts:
        if random.random() > 0.7: # Only some vertices get hairs
            # Create a small line for the hair
            start = v.co * Vector((0.6, 0.8, 0.7))
            end = start + (v.co.normalized() * 0.1)
            hairs.append((start, end))
    
    bm.free()
    
    # Create the hair geometry as a separate mesh to avoid bloating one object
    hair_mesh = bpy.data.meshes.new("ThoraxHairs")
    hair_obj = bpy.data.objects.new("ThoraxHairs", hair_mesh)
    bpy.context.collection.objects.link(hair_obj)
    
    bm_hairs = bmesh.new()
    for s, e in hairs:
        v1 = bm_hairs.verts.new(s)
        v2 = bm_hairs.verts.new(e)
        bm_hairs.edges.new((v1, v2))
    bm_hairs.to_mesh(hair_mesh)
    bm_hairs.free()

    return thorax

def create_abdomen():
    # Long slender segmented abdomen
    segments = 10
    total_length = 6.0
    radius_start = 0.4
    radius_end = 0.05
    
    abdomen_parts = []
    for i in range(segments):
        r_curr = radius_start - (radius_start - radius_end) * (i / segments)
        r_next = radius_start - (radius_start - radius_end) * ((i + 1) / segments)
        
        # Create a segment as a cylinder/cone
        bpy.ops.mesh.primitive_cylinder_add(
            vertices=12, 
            radius1=r_curr, 
            radius2=r_next, 
            depth=total_length / segments, 
            location=(0, -(i + 0.5) * (total_length/segments) + 0.8, 0)
        )
        seg = bpy.context.active_object
        seg.name = f"AbdomenSeg_{i}"
        # Add a small bevel or ring to signify segmentation
        abdomen_parts.append(seg)

    return abdomen_parts

def create_head():
    # Small rounded head
    bpy.ops.mesh.primitive_uv_sphere_add(radius=0.4, location=(0, 1.2, 0))
    head = bpy.context.active_object
    head.name = "Head"
    head.scale = (0.8, 0.7, 0.6)
    
    # Compound eyes
    for side in [-1, 1]:
        bpy.ops.mesh.primitive_uv_sphere_add(radius=0.3, location=(0.4 * side, 1.3, 0.2))
        eye = bpy.context.active_object
        eye.name = f"Eye_{side}"
        eye.scale = (0.6, 0.8, 0.8)
    
    return head
