import bpy
import bmesh
import math
from mathutils import Vector

def clear_scene():
    """Clears the default Blender scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_heart_material():
    """Creates a pale soft green material for the leaf."""
    mat = bpy.data.materials.new(name="LeafMaterial")
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        # Pale soft green color
        bsdf.inputs['Base Color'].default_value = (0.55, 0.82, 0.45, 1.0)
        bsdf.inputs['Roughness'].default_value = 0.6
    return mat

def generate_heart_leaf():
    """Generates a heart-shaped leaf with midrib and veins."""
    # Parameters for the parametric heart shape
    # x = 16 sin^3(t)
    # y = 13 cos(t) - 5 cos(2t) - 2 cos(3t) - cos(4t)
    res = 128
    points = []
    for i in range(res):
        t = (i / res) * 2 * math.pi
        x = 16 * (math.sin(t)**3)
        y = 13 * math.cos(t) - 5 * math.cos(2*t) - 2 * math.cos(3*t) - math.cos(4*t)
        points.append(Vector((x * 0.1, y * 0.1, 0)))

    # Create Mesh and BMesh
    mesh = bpy.data.meshes.new("HeartLeaf")
    obj = bpy.data.objects.new("HeartLeaf", mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()

    # Create boundary vertices
    verts = [bm.verts.new(p) for p in points]
    
    # Connect boundary to form a loop
    for i in range(len(verts)):
        bm.edges.new((verts[i], verts[(i + 1) % len(verts)]))

    # Create a center vertex for triangulation (fan)
    center_v = bm.verts.new((0, 0, 0))
    for v in verts:
        bm.edges.new((center_v, v))
    
    # Fill the faces
    bmesh.ops.contextualize(bm, edges=bm.edges) # Ensure consistency
    bpy.context.view_layer.objects.active = obj
    
    # To properly fill a heart shape with BMesh, we can use bmesh.ops.face_join 
    # or just manually create faces in the fan.
    for i in range(len(verts)):
        try:
            bm.faces.new((center_v, verts[i], verts[(i + 1) % len(verts)]))
        except ValueError:
            pass # Handle duplicate edges/faces

    # Subdivide heavily to allow for detailed surface deformation (midrib and veins)
    # We'll use bmesh subdivision operators
    for _ in range(5):
        bmesh.ops.subdivide_edges(bm, edges=bm.edges, cuts=1, use_grid_fill=True)

    # Now deform the vertices for organic look
    # Midrib: a slight dip along the Y axis (center line)
    # Veins: subtle ridges branching out
    for v in bm.verts:
        x, y, z = v.co
        
        # 1. The Central Crease (Midrib)
        # Create a narrow valley centered at x=0
        midrib_depth = 0.05 * math.exp(-(x**2) / (0.01))
        v.co.z -= midrib_depth
        
        # 2. Slight overall curvature of the leaf (bowing)
        v.co.z += (x**2 + y**2) * 0.05
        
        # 3. Vein-like surface lines
        # Create diagonal ridges branching from center to edges
        # We use a sine wave combined with distance to create 'ribs'
        vein_freq = 8.0
        vein_strength = 0.015
        
        # Distort Y slightly to make veins look more natural (diagonal)
        angle_offset = x * 2.0
        vein_val = math.sin((y + angle_offset) * vein_freq)
        
        # Veins should fade out as they get further from the center, but not disappear entirely
        fade = math.exp(-abs(x)*3.0) + 0.2
        v.co.z += vein_val * vein_strength * fade

    # Finalize BMesh
    bm.to_mesh(mesh)
    bm.free()

    # Add a Solidify modifier for slight thickness (leaves aren't 2D planes)
    solid = obj.modifiers.new(name="Solidify", type='SOLIDIFY')
    solid.thickness = 0.005
    solid.offset = 0 # Center the thickness

    # Add a Subdivision Surface modifier for smoothness
    subdiv = obj.modifiers.new(name="Subsurf", type='SUBSURF')
    subdiv.levels = 2
    subdiv.render_levels = 3

    # Set smooth shading
    for poly in mesh.polygons:
        poly.use_smooth = True

    return obj

def main():
    clear_scene()
    
    # Create the material
    leaf_mat = create_heart_material()
    
    # Generate the leaf object
    leaf_obj = generate_heart_leaf()
    
    # Assign material
    if leaf_obj.data.materials:
        leaf_obj.data.materials[0] = leaf_mat
    else:
        leaf_obj.data.materials.append(leaf_mat)

    # Rotate for better presentation (elevated view feel)
    leaf_obj.rotation_euler[0] = math.radians(-20)
    leaf_obj.rotation_euler[2] = math.radians(45)

if __name__ == "__main__":
    main()
