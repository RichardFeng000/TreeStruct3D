def clear_scene():
    """Clears the default blender scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Creates a simple diffuse material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
    return mat

def create_spines(body_obj):
    """Creates dense clusters of spines across the cactus surface."""
    # We'll sample face centers and place small cones
    spine_mesh = bpy.data.meshes.new("Spines")
    spine_obj = bpy.data.objects.new("Spines", spine_mesh)
    bpy.context.collection.objects.link(spine_obj)
    
    bm_spines = bmesh.new()
    
    # Access the body mesh to find face centers and normals
    body_mesh = body_obj.data
    
    # Saguaro spines are clustered in 'areoles' along the ribs
    # We sample a subset of faces
    num_clusters = 400
    
    for _ in range(num_clusters):
        # Pick a random face
        face_idx = random.randint(0, len(body_mesh.polygons) - 1)
        face = body_mesh.polygons[face_idx]
        center = face.center
        normal = face.normal
        
        # Each cluster has multiple spines
        spines_per_cluster = random.randint(5, 12)
        for _ in range(spines_per_cluster):
            # Randomly perturb the normal for the spine direction
            spine_dir = Vector((
                normal.x + random.uniform(-0.3, 0.3),
                normal.y + random.uniform(-0.3, 0.3),
                normal.z + random.uniform(-0.3, 0.3)
            )).normalized()
            
            length = random.uniform(0.05, 0.15)
            
            # Create a simple thin cylinder/cone for each spine
            # For performance in BMesh, we'll just use a line (two vertices) 
            # and then maybe skin it or keep as lines? No, the prompt asks for detail.
            # To optimize: create one cone template and instance it? 
            # But BMesh is better for merging into one object.
            
            v1 = bm_spines.verts.new(center)
            v2 = bm_spines.verts.new(center + spine_dir * length)
            bm_spines.edges.new((v1, v2))

    # Since we just made edges (lines), let's convert them to thin cylinders 
    # using a skin modifier or simply create actual geometry.
    # Actually, creating thousands of tiny cones in BMesh is slow.
    # Let's use a more efficient method: generate vertices and then Use a Modifier.
    bm_spines.to_mesh(spine_mesh)
    bm_spines.free()
    
    # To make the lines look like spines, we can add a skin modifier or simply 
    # just leave them as edges if we were rendering with a specific shader.
    # Since we need "real geometry", I'll use a Skin Modifier and then Subsurf for volume.
    skin_mod = spine_obj.modifiers.new(name="Skin", type='SKIN')
    
    # Adjust skin radius to be very thin
    for v in spine_mesh.vertices:
        spine_obj.data.skin_vertices[0].radius = [0.005, 0.005]

    return spine_obj

def main():
    clear_scene()
    
    # Create materials
    green_mat = create_material("CactusGreen", (0.1, 0.4, 0.1, 1.0))
    white_mat = create_material("SpineWhite", (0.9, 0.9, 0.9, 1.0))
