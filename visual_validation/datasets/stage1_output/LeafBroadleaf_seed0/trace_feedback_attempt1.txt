import bpy
import bmesh
import math
from mathutils import Vector

def clear_scene():
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_leaf():
    # Parameters
    length = 4.0
    width = 2.0
    res_u = 60  # length resolution
    res_v = 30  # width resolution
    cup_long = 0.15 # longitudinal curvature
    cup_trans = 0.2  # transverse cupping
    ovate_factor = 0.4 # how much it bulges at the base
    stem_radius = 0.06
    stem_length = 1.2

    bm = bmesh.new()

    # Create a grid of vertices for the blade
    verts = []
    for i in range(res_u + 1):
        row = []
        u = (i / res_u) * 2 - 1  # u from -1 to 1
        # Width function for an ovate shape: wider at the base (u=-1), narrower at tip (u=1)
        # w(u) = sqrt(1-u^2) * (1 + factor * (-u))
        w_factor = math.sqrt(max(0, 1 - u**2)) * (1 + ovate_factor * (-u))
        
        for j in range(res_v + 1):
            v = (j / res_v) * 2 - 1  # v from -1 to 1
            
            x = u * length * 0.5
            y = v * w_factor * width * 0.5
            
            # Calculate curvature
            # Longitudinal bend (arching the leaf)
            z_long = cup_long * (u**2) * length * 0.1
            # Transverse cupping (curving across the width)
            z_trans = cup_trans * (v**2) * width * 0.1
            
            # Central vein subtle protrusion
            vein_bump = 0
            if abs(v) < 0.1:
                vein_bump = (1.0 - abs(v)/0.1) * 0.05
            
            z = z_long + z_trans + vein_bump
            
            vert = bm.verts.new(Vector((x, y, z)))
            row.append(vert)
        verts.append(row)

    # Create faces for the blade
    for i in range(res_u):
        for j in range(res_v):
            bm.faces.new((
                verts[i][j], 
                verts[i+1][j], 
                verts[i+1][j+1], 
                verts[i][j+1]
            ))

    # Create the central vein geometry (a thin strip slightly raised)
    # We identify the center column of vertices and offset them slightly more if needed, 
    # but the bump is already in Z. To make it a "feature", we can add edges or just rely on subdiv.

    # Create the petiole (stem)
    # The stem starts at the base u=-1, v=0.
    base_u = verts[0][res_v // 2]
    base_pos = base_u.co
    
    # Generate a simple cylinder for the stem
    stem_segments = 8
    stem_rings = 10
    stem_verts = []
    for r in range(stem_rings + 1):
        ring = []
        z_off = - (r / stem_rings) * stem_length
        for s in range(stem_segments):
            angle = (s / stem_segments) * 2 * math.pi
            # The stem should align with the base of the leaf
            # Since we are at origin, and x is length, the stem goes in -x direction usually,
            # but since the leaf starts at -length/2, it extends further back.
            offset_x = - (r / stem_rings) * stem_length * 0.5 # tilt slightly
            offset_y = math.cos(angle) * stem_radius
            offset_z = math.sin(angle) * stem_radius + z_off * 0.2
            
            v_stem = bm.verts.new(Vector((base_pos.x + offset_x, base_pos.y + offset_y, base_pos.z + offset_z)))
            ring.append(v_stem)
        stem_verts.append(ring)

    for r in range(stem_rings):
        for s in range(stem_segments):
            s_next = (s + 1) % stem_segments
            bm.faces.new((
                stem_verts[r][s], 
                stem_verts[r+1][s], 
                stem_verts[r+1][s_next], 
                stem_verts[r][s_next]
            ))

    # Bridge the stem to the leaf base
    # Connect the top ring of the stem to the center vertices of the leaf base
    for s in range(stem_segments):
        s_next = (s + 1) % stem_segments
        # Approximate a connection to the bottom-most row of the leaf mesh
        v1 = stem_verts[0][s]
        v2 = stem_verts[0][s_next]
        v3 = verts[0][res_v // 2 + 1] if res_v // 2 + 1 <= res_v else verts[0][res_v // 2]
        v4 = verts[0][res_v // 2]
        # We just create some triangles to fill the gap
        bm.faces.new((v1, v2, v3))
        bm.faces.new((v1, v3, v4))

    bm.to_mesh("Leaf")
    bm.free()

def apply_modifiers(obj):
    # Smooth shading
    for poly in obj.data.polygons:
        poly.use_smooth = True

    # Subdivision Surface for high fidelity organic look
    subsurf = obj.modifiers.new(name="Subdiv", type='SUBSURF')
    subsurf.levels = 2
    subsurf.render_levels = 3

    # Solidify to give the leaf a tiny bit of physical thickness
    solid = obj.modifiers.new(name="Solidify", type='SOLIDIFY')
    solid.thickness = 0.01

def main():
    clear_scene()
    
    create_leaf()
    
    # Get the object created in create_leaf (it's named "Leaf")
    obj = bpy.data.objects.get("Leaf")
    if obj:
        # Center the object roughly
        bpy.context.view_layer.objects.active = obj
        bpy.ops.object.select_all(action='DESELECT')
        obj.select_set(True)
        
        apply_modifiers(obj)
        
        # Rotate for a three-quarter perspective look in the scene (though requested as model only)
        # The prompt asks for it "rendered from 3/4", but we provide geometry at origin.
        # We'll keep it slightly tilted so its natural form is apparent.
        obj.rotation_euler = (math.radians(15), 0, math.radians(45))

if __name__ == "__main__":
    main()
