def create_mussel_material(name, color):
    """Create a simple principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    node_principled = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    node_principled.inputs['Base Color'].default_value = color
    # Make shell slightly glossy/pearly
    node_principled.inputs['Roughness'].default_value = 0.3
    mat.node_tree.links.new(node_principled.outputs['BSDF'], node_output.inputs['Surface'])
    return mat

def generate_mussel_valve(name, is_top=True):
    """
    Procedurally generates one half of a mussel shell using BMesh.
    The shape is an elongated teardrop with geometric growth rings.
    """
    bm = bmesh.new()
    
    # Parametric constants
    segments_long = 60  # Resolution along the length
    segments_arc = 32   # Resolution around the shell arc
    length = 4.0        # Total length of the valve
    width_max = 1.8     # Maximum width
    height_max = 1.2    # Maximum height/thickness
    
    # Generate vertices for a half-shell (semi-ellipsoid teardrop)
    # u: distance from hinge [0, 1]
    # v: arc angle [0, PI]
    verts = []
    for i in range(segments_long + 1):
        u = i / segments_long
        
        # Shape function for a teardrop/mussel profile
        # Tapers at the hinge (u=0) and rounds at the end (u=1)
        scale_factor = math.sin(math.pi * u**0.7) 
        if u == 0: scale_factor = 0.05 # Small point at hinge
        
        # Growth ring pattern: add geometric displacement based on length
        ring_freq = 12.0
        ring_amp = 0.03
        displacement = 1.0 + ring_amp * math.sin(u * ring_freq * 2 * math.pi)
        
        current_w = width_max * scale_factor * displacement
        current_h = height_max * scale_factor * displacement
        
        row = []
        for j in range(segments_arc + 1):
            v = (j / segments_arc) * math.pi
            
            # X: width, Y: depth/thickness, Z: length axis
            x = current_w * math.cos(v)
            y = current_h * math.sin(v)
            z = u * length
            
            row.append(bm.verts.new(Vector((x, y, z))))
        verts.append(row)

    # Create faces for the outer shell surface
    for i in range(segments_long):
        for j in range(segments_arc):
            bm.faces.new((
                verts[i][j], 
                verts[i+1][j], 
                verts[i+1][j+1], 
                verts[i][j+1]
            ))

    # Create the "lip" of the shell (the open edge)
    # This connects the arc from v=0 to v=PI at u=1
    # Since our loop creates a semi-circle, the ends are naturally lines.
    # We need to close the teardrop end and the hinge end.
    
    # Close the rounded end (u=1)
    end_verts = verts[segments_long]
    bm.faces.new(end_verts[::-1]) # Reverse list for correct normal
    
    # Close the hinge end (u=0)
    start_verts = verts[0]
    bm.faces.new(start_verts)

    # Give the shell thickness by duplicating and flipping (Solidify effect)
    # We use bmesh to extrude slightly inwards along normals
    bmesh.ops.duplicate(bm, geom=bm.faces[:])
    
    # Offset duplicated faces inwards
    for face in bm.faces:
        if face.index >= segments_long * segments_arc: # Only the new faces
            # Move vertices slightly toward center of shell
            v_center = Vector((0, 0, face.calc_center().z))
            for v in face.verts:
                dir_vec = (v.co - v_center).normalized()
                v.co -= dir_vec * 0.05

    # Bridge the original and duplicated surfaces at the edges
    # To keep it simple, we use a modifier later or manually connect.
    # For high fidelity, let's just finish this mesh.
    
    obj = bpy.data.objects.new(name, bm.to_mesh())
    bpy.context.collection.objects.link(obj)
    bm.free()
    return obj
