import bpy
import bmesh
import math

def clear_scene():
    """Removes all objects from the current scene."""
    if bpy.context.active_object:
        bpy.ops.object.mode_set(mode='OBJECT')
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color, emission=0.0, roughness=0.5):
    """Creates a Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    
    # Create Principled BSDF node
    node_bsdf = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_bsdf.inputs['Base Color'].default_value = color
    node_bsdf.inputs['Roughness'].default_value = roughness
    
    if emission > 0:
        node_bsdf.inputs['Emission Strength'].default_value = emission
        # Set Emission color (defaults to white if Base Color is used, but we can set it specifically)
        node_bsdf.inputs['Emission Color'].default_value = color

    # Create Output node
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    
    # Link them
    links = mat.node_tree.links
    links.new(node_bsdf.outputs['BSDF'], node_output.inputs['Surface'])
    
    return mat

def create_ceiling_light():
    # Parameters
    radius = 0.4
    height = 0.06
    resolution = 64
    bevel_width = 0.01
    
    # Create the basic cylinder for the light fixture
    bpy.ops.mesh.primitive_cylinder_add(
        vertices=resolution, 
        radius=radius, 
        depth=height, 
        location=(0, 0, 0)
    )
    obj = bpy.context.active_object
    obj.name = "FlushMountLight"

    # Use BMesh to refine the geometry and assign materials
    bm = bmesh.new()
    bm.from_mesh(obj.data)
    
    # Create Materials
    mat_diffuser = create_material("Mat_Diffuser", (1, 1, 1, 1), emission=0.0, roughness=0.2)
    mat_glow = create_material("Mat_Glow", (1, 1, 1, 1), emission=15.0, roughness=0.0)
    mat_rim = create_material("Mat_Rim", (0.8, 0.8, 0.8, 1), emission=0.0, roughness=0.4)

    obj.data.materials.append(mat_diffuser) # Index 0
    obj.data.materials.append(mat_glow)     # Index 1
    obj.data.materials.append(mat_rim)      # Index 2

    # Identify faces: Top, Bottom, and Sides
    # In a standard Blender cylinder:
    # The face with normal (0,0,1) is the top
    # The face with normal (0,0,-1) is the bottom
    
    for face in bm.faces:
        # Normal check
        norm = face.normal
        if norm.z > 0.9:
            face.material_index = 0 # Top - Diffuser
        elif norm.z < -0.9:
            face.material_index = 1 # Bottom - Glow
        else:
            face.material_index = 2 # Sides - Rim

    # Add a subtle bevel to the top and bottom edges for realism
    # We'll use the Bevel modifier instead of bmesh manual beveling for cleaner results
    bm.to_mesh(obj.data)
    bm.free()

    # Apply modifiers
    # 1. Bevel for smoothed edges
    bevel_mod = obj.modifiers.new(name="Bevel", type='BEVEL')
    bevel_mod.width = bevel_width
    bevel_mod.segments = 3
    bevel_mod.limit_method = 'ANGLE'
    bevel_mod.angle_limit = math.radians(30)

    # 2. Subdivision Surface for perfectly smooth cylindrical look
    subsurf_mod = obj.modifiers.new(name="Subdiv", type='SUBSURF')
    subsurf_mod.levels = 1
    subsurf_mod.render_levels = 2

    # Shade smooth
    bpy.ops.object.shade_smooth()
    
    # To prevent the "shading artifacts" on a cylinder with subsurf, we need to ensure normals are clean
    # Or just use auto-smooth (in 4.1+ it's a modifier or attribute)
    # For Blender 5.0, let's use the Smooth by Angle modifier
    smooth_mod = obj.modifiers.new(name="SmoothByAngle", type='SMOOTH_BY_ANGLE')
    smooth_mod.angle = math.radians(30)

def setup_scene():
    clear_scene()
    create_ceiling_light()

if __name__ == "__main__":
    setup_scene()
