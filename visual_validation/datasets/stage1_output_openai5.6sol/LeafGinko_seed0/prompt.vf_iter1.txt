def outer_radius(theta):
    normalized = (theta - theta_min) / (theta_max - theta_min)
    side_shaping = -0.22 * (abs(2.0 * normalized - 1.0) ** 1.8)
    broad_lobes = 0.12 * math.cos(2.0 * (theta - math.pi * 0.5))
    scallops = 0.085 * math.cos(12.0 * (theta - theta_min) + 0.25)
    fine_wave = 0.025 * math.sin(25.0 * theta)
    central_notch = -0.67 * math.exp(-((theta - math.pi * 0.5) / 0.105) ** 2)
    asymmetry = 0.045 * math.sin(theta - math.pi * 0.5)
    return 2.62 + side_shaping + broad_lobes + scallops + fine_wave + central_notch + asymmetry

def blade_position(theta, fraction):
    radius = outer_radius(theta) * fraction
    x = radius * math.cos(theta)
    y = radius * math.sin(theta)
    cup = 0.115 * fraction * fraction * (x / 2.5) ** 2
    sweep = 0.045 * fraction * math.sin(theta - math.pi * 0.5)
    edge_wave = 0.032 * fraction ** 3 * math.sin(10.0 * theta + 0.4)
    longitudinal_curve = 0.035 * fraction * fraction
    z = cup + sweep + edge_wave + longitudinal_curve
    return Vector((x, y, z))

def make_poly_curve(name, points, bevel_depth, material, cyclic=False, bevel_resolution=1):
    curve_data = bpy.data.curves.new(name + " Data", type='CURVE')
    curve_data.dimensions = '3D'
    curve_data.resolution_u = 1
    curve_data.bevel_depth = bevel_depth
    curve_data.bevel_resolution = bevel_resolution
    curve_data.resolution_u = 1
    curve_data.materials.append(material)

    spline = curve_data.splines.new('POLY')
    spline.points.add(len(points) - 1)
    for point, coordinate in zip(spline.points, points):
        point.co = (coordinate.x, coordinate.y, coordinate.z, 1.0)
    spline.use_cyclic_u = cyclic

    obj = bpy.data.objects.new(name, curve_data)
    bpy.context.collection.objects.link(obj)
    curve_objects.append(obj)
    return obj
