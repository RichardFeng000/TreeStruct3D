def local_frame(tangent):
    tangent = tangent.normalized()
    if abs(tangent.z) < 0.88:
        side = tangent.cross(Vector((0.0, 0.0, 1.0))).normalized()
    else:
        side = tangent.cross(Vector((1.0, 0.0, 0.0))).normalized()
    up = side.cross(tangent).normalized()
    return side, up

def add_tube(points, radii, sides=8, material_index=1, flatten=1.0,
             cap_start=True, cap_end=True, irregularity=0.08):
    if len(points) < 2:
        return

    start_index = len(verts)
    ring_indices = []

    for i, point in enumerate(points):
        if i == 0:
            tangent = points[1] - points[0]
        elif i == len(points) - 1:
            tangent = points[-1] - points[-2]
        else:
            tangent = points[i + 1] - points[i - 1]

        side, up = local_frame(tangent)
        ring = []
        for j in range(sides):
            angle = 2.0 * math.pi * j / sides
            wave = (
                math.sin(j * 2.31 + i * 1.17) * 0.52 +
                math.sin(j * 4.07 - i * 0.83) * 0.25
            )
            jitter = 1.0 + irregularity * wave + random.uniform(-irregularity * 0.25, irregularity * 0.25)
            radial = side * math.cos(angle) + up * (math.sin(angle) * flatten)
            ring.append(len(verts))
            verts.append(tuple(point + radial * radii[i] * jitter))
        ring_indices.append(ring)

    for i in range(len(ring_indices) - 1):
        for j in range(sides):
            face = (
                ring_indices[i][j],
                ring_indices[i][(j + 1) % sides],
                ring_indices[i + 1][(j + 1) % sides],
                ring_indices[i + 1][j],
            )
            faces.append(face)
            color_roll = random.random()
            if color_roll < 0.08:
                mat = 2
            elif color_roll < 0.13:
                mat = 0
            else:
                mat = material_index
            face_materials.append(mat)
            face_smooth.append(True)

    if cap_start:
        faces.append(tuple(reversed(ring_indices[0])))
        face_materials.append(material_index)
        face_smooth.append(False)

    if cap_end:
        faces.append(tuple(ring_indices[-1]))

def random_perpendicular(direction):
    trial = Vector((
        random.uniform(-1.0, 1.0),
        random.uniform(-1.0, 1.0),
        random.uniform(-0.5, 1.0)
    ))
    perpendicular = trial - direction * trial.dot(direction)
    if perpendicular.length < 0.001:
        perpendicular = direction.cross(Vector((0.0, 0.0, 1.0)))
    if perpendicular.length < 0.001:
        perpendicular = Vector((1.0, 0.0, 0.0))
    return perpendicular.normalized()

def grow_branch(start, initial_direction, length, radius, depth, tone=1):
    direction = initial_direction.normalized()
    point_count = 5 if depth >= 2 else 4
    points = [start.copy()]
    current = start.copy()
    current_dir = direction.copy()

    side_bias = random_perpendicular(current_dir) * random.uniform(-0.12, 0.12)
    for i in range(1, point_count):
        progress = i / (point_count - 1)
        wander = Vector((
            random.uniform(-0.22, 0.22),
            random.uniform(-0.22, 0.22),
            random.uniform(-0.13, 0.20)
        ))
        lift = Vector((0.0, 0.0, 0.10 if depth >= 2 else -0.015))
        current_dir = (current_dir * 0.86 + wander + side_bias + lift).normalized()
        step = length / (point_count - 1)
        step *= random.uniform(0.88, 1.10)
        current = current + current_dir * step
        points.append(current.copy())

    if depth > 0:
        end_radius = max(radius * random.uniform(0.62, 0.70), 0.018)
    else:
        end_radius = max(radius * random.uniform(0.15, 0.25), 0.008)
