def create_mesh_object(name, verts, faces, materials, smooth=True, material_indices=None):
    mesh = bpy.data.meshes.new(name + "Mesh")
    mesh.from_pydata(verts, [], faces)
    mesh.materials.clear()
    for material in materials:
        mesh.materials.append(material)

            poly.material_index = idx

        for poly in mesh.polygons:

    return obj

def append_tube(verts, faces, points, radii, sides=8):
    if len(points) < 2:
        return
    start = len(verts)
    frames = []
    previous_n = None
    for i, point in enumerate(points):
        if i == 0:
            tangent = (points[1] - points[0]).normalized()
        elif i == len(points) - 1:
            tangent = (points[-1] - points[-2]).normalized()
        else:
            tangent = (points[i + 1] - points[i - 1]).normalized()
        if previous_n is None:
            ref = Vector((0.0, 0.0, 1.0))
            if abs(tangent.dot(ref)) > 0.88:
                ref = Vector((1.0, 0.0, 0.0))
            n = tangent.cross(ref).normalized()
        else:
            n = previous_n - tangent * previous_n.dot(tangent)
            if n.length < 0.001:
                ref = Vector((1.0, 0.0, 0.0))
                n = tangent.cross(ref).normalized()
            else:
                n.normalize()
        b = tangent.cross(n).normalized()
        previous_n = n
        frames.append((n, b))
        radius = radii[i] if hasattr(radii, "__len__") else radii
        for j in range(sides):
            angle = 2.0 * math.pi * j / sides
            verts.append(tuple(point + radius * (math.cos(angle) * n + math.sin(angle) * b)))
    rings = len(points)
    for i in range(rings - 1):
        for j in range(sides):
            a = start + i * sides + j
            b = start + i * sides + (j + 1) % sides
            c = start + (i + 1) * sides + (j + 1) % sides
            d = start + (i + 1) * sides + j
            faces.append((a, b, c, d))
    faces.append(tuple(start + j for j in reversed(range(sides))))
    end_start = start + (rings - 1) * sides
    faces.append(tuple(end_start + j for j in range(sides)))

def append_petal(verts, faces, center, axis, radial, scale):
    radial = radial.normalized()
    side = axis.cross(radial).normalized()
    inner = center + radial * (0.010 * scale)
    middle = center + radial * (0.052 * scale) + axis * (0.007 * scale)
    tip = center + radial * (0.102 * scale) + axis * (0.020 * scale)
    width = 0.027 * scale
    thick = 0.006 * scale
    polygon = [
        inner,
        middle - side * width,
        tip,
        middle + side * width
    ]
    start = len(verts)
    for p in polygon:
        verts.append(tuple(p + axis * thick))
    for p in polygon:
        verts.append(tuple(p - axis * thick))
