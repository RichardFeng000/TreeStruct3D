def organic_noise(q, phi):
    theta = math.acos(max(0.0, min(1.0, q)))
    side_weight = 0.35 + 0.65 * math.sin(theta)
    value = (
        0.050 * math.sin(phi * 5.0 + theta * 8.0)
        * math.sin(phi * 9.0 - theta * 3.0)
        + 0.027 * math.cos(phi * 17.0 + theta * 11.0)
        + 0.015 * math.sin(phi * 31.0 - theta * 19.0)
        + 0.009 * math.cos(phi * 47.0 + theta * 29.0)
    )
    return value * side_weight

def ellipsoid_point(q, phi):
    q = max(0.0, min(1.0, q))
    radial = math.sqrt(max(0.0, 1.0 - q * q))
    point = Vector((
        RX * radial * math.cos(phi),
        RY * radial * math.sin(phi),
        RZ * q
    ))
    normal = Vector((
        point.x / (RX * RX),
        point.y / (RY * RY),
        point.z / (RZ * RZ)
    )).normalized()
    point += normal * organic_noise(q, phi)
    return point, normal

def project_to_dome(candidate):
    scaled = Vector((
        candidate.x / RX,
        candidate.y / RY,
        max(0.012, candidate.z / RZ)
    ))
    if scaled.length == 0.0:
        scaled = Vector((0.0, 0.0, 1.0))
    scaled.normalize()
    q = max(0.0, min(1.0, scaled.z))
    phi = math.atan2(scaled.y, scaled.x)
    return ellipsoid_point(q, phi)
