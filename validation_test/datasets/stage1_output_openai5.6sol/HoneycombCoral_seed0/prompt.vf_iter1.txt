def surface_noise(q, phi):
    theta = math.acos(max(-1.0, min(1.0, q)))
    s = math.sin(theta)
    a = math.sin(phi * 5.0 + theta * 7.0)
    b = math.sin(phi * 11.0 - theta * 4.0)
    c = math.cos(phi * 17.0 + theta * 13.0)
    d = math.sin(phi * 29.0 - theta * 9.0)
    return (0.042 * a * b + 0.021 * c + 0.010 * d) * (0.28 + 0.72 * s)

def dome_surface(q, phi):
    q = max(0.0, min(1.0, q))
    radial = math.sqrt(max(0.0, 1.0 - q * q))
    p = Vector((
        RX * radial * math.cos(phi),
        RY * radial * math.sin(phi),
        RZ * q
    ))
    n = Vector((
        p.x / (RX * RX),
        p.y / (RY * RY),
        p.z / (RZ * RZ)
    )).normalized()
    p += n * surface_noise(q, phi)
    return p, n
