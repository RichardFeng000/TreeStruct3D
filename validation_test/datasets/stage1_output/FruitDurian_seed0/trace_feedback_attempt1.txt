import bpy
import bmesh
import math
import random
from mathutils import Vector

def clear_scene():
    """Clears the default scene objects."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Creates a simple principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
        # Give it a slightly matte look for fruit skin
        bsdf.inputs['Roughness'].default_value = 0.8
    return mat

def create_spine_mesh(base_radius, height, sides=6):
    """Creates a pyramid-like spine as a bmesh."""
    bm = bmesh.new()
    # Create base vertices
    verts = []
    for i in range(sides):
        angle = (2 * math.pi * i) / sides
        vx = math.cos(angle) * base_radius
        vy = math.sin(angle) * base_radius
        verts.append(bm.verts.new((vx, vy, 0)))
    
    # Create tip vertex
    tip = bm.verts.new((0, 0, height))
    
    # Create faces (sides)
    for i in range(sides):
        v1 = verts[i]
        v2 = verts[(i + 1) % sides]
        bm.faces.new((v1, v2, tip))
    
    # Create base face
    bm.faces.new(verts)
    
    return bm

def generate_durian():
    clear_scene()

    # --- Parameters ---
    body_radius = 1.0
    body_stretch_z = 1.2  # Ovoid shape
    num_spines = 1800
    spine_base_rad = 0.035
    spine_height = 0.14
    stem_radius = 0.06
    stem_height = 0.3

    # --- Materials ---
    mat_body = create_material("DurianBody", (0.6, 0.7, 0.2, 1.0)) # Pale yellow-green
    mat_stem = create_material("DurianStem", (0.4, 0.3, 0.2, 1.0))  # Tan-brown

    # --- Body Construction ---
    # Create the main body ovoid
    bpy.ops.mesh.primitive_uv_sphere_add(radius=body_radius, segments=64, ring_count=32)
    body_obj = bpy.context.active_object
    body_obj.name = "Durian_Body"
    body_obj.scale = (1.0, 1.0, body_stretch_z)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    
    # Assign material to body
    body_obj.data.materials.append(mat_body)

    # --- Spine Distribution (Fibonacci Sphere on Ellipsoid) ---
    # We use a Fibonacci spiral for uniform distribution of points on the surface
    spines_bm = bmesh.new()
    
    for i in range(num_spines):
        phi = math.acos(1 - 2 * (i + 0.5) / num_spines)
        theta = math.pi * (1 + 5**0.5) * (i + 0.5)
        
        # Base unit sphere point
        x = math.cos(theta) * math.sin(phi)
        y = math.sin(theta) * math.sin(phi)
        z = math.cos(phi)
        
        # Scale to match body ovoid shape
        pos = Vector((x, y, z * body_stretch_z))
        
        # Calculate normal for the ellipsoid surface (grad of x^2 + y^2 + (z/s)^2 - 1)
        # Normal is proportional to (2x, 2y, 2z/s^2)
        normal = Vector((x, y, z / (body_stretch_z**2))).normalized()
        
        # Create a local rotation matrix to align the spine with the normal
        up = Vector((0, 0, 1))
        rot_quat = up.rotation_difference(normal)
        mat_rot = rot_quat.to_matrix()
        trans_mat = Vector(pos)

        # Generate a small pyramidal spine at this location/orientation
        spine_base_bm = create_spine_mesh(spine_base_rad, spine_height)
        
        # Transform vertices of the spine into world space (relative to durian center)
        for v in spine_base_bm.verts:
            v.co = mat_rot @ v.co + trans_mat
            
        # Merge spine geometry into the master spines bmesh
        # To avoid excessive memory, we'll merge these later
        bmesh.ops.join_meshes([spines_bm, spine_base_bm])
        spine_base_bm.free()

    # Create object from combined spines
    spine_mesh = bpy.data.meshes.new("SpinesMesh")
    spines_bm.to_mesh(spine_mesh)
    spines_bm.free()
    spine_obj = bpy.data.objects.new("Durian_Spines", spine_mesh)
    bpy.context.collection.objects.link(spine_obj)
    spine_obj.data.materials.append(mat_body)

    # --- Stem Construction ---
    # The stem is a small, slightly irregular cylinder at the top
    stem_bm = bmesh.new()
    
    # Create circular base for stem
    segments = 12
    base_verts = []
    for i in range(segments):
        angle = (2 * math.pi * i) / segments
        v = stem_bm.verts.new((math.cos(angle) * stem_radius, math.sin(angle) * stem_radius, 0))
        base_verts.append(v)
    
    # Create top vertices (slightly tapered and offset for a natural look)
    top_verts = []
    for i in range(segments):
        angle = (2 * math.pi * i) / segments
        # Taper the tip slightly and add small randomness
        r_tip = stem_radius * 0.7 + random.uniform(-0.01, 0.01)
        offset_x = random.uniform(-0.02, 0.02)
        offset_y = random.uniform(-0.02, 0.02)
        v = stem_bm.verts.new((math.cos(angle) * r_tip + offset_x, math.sin(angle) * r_tip + offset_y, stem_height))
        top_verts.append(v)
    
    # Create faces for the stem cylinder
    for i in range(segments):
        v1 = base_verts[i]
        v2 = base_verts[(i + 1) % segments]
        v3 = top_verts[(i + 1) % segments]
        v4 = top_verts[i]
        stem_bm.faces.new((v1, v2, v3, v4))
    
    # Cap the stem
    stem_bm.faces.new(base_verts)
    stem_bm.faces.new(reversed(top_verts))

    stem_mesh = bpy.data.meshes.new("StemMesh")
    stem_bm.to_mesh(stem_mesh)
    stem_bm.free()
    stem_obj = bpy.data.objects.new("Durian_Stem", stem_mesh)
    bpy.context.collection.objects.link(stem_obj)
    stem_obj.data.materials.append(mat_stem)
    
    # Position stem at the top of the ovoid body
    stem_obj.location = (0, 0, body_radius * body_stretch_z)

    # Final touch: Join Body and Spines for a cleaner hierarchy if desired,
    # but keeping them separate is fine too. Let's just ensure they are centered.
    body_obj.location = (0, 0, 0)
    spine_obj.location = (0, 0, 0)

if __name__ == "__main__":
    generate_durian()
