def clear_scene():
    """Clear the default Blender scene."""
    if "Cube" in bpy.data.objects:
        bpy.data.objects.remove(bpy.data.objects["Cube"], do_unlink=True)
    if "Camera" in bpy.data.objects:
        bpy.data.objects.remove(bpy.data.objects["Camera"], do_unlink=True)
    if "Light" in bpy.data.objects:
        bpy.data.objects.remove(bpy.data.objects["Light"], do_unlink=True)
    # Clear all meshes and materials to start fresh
    for mesh in bpy.data.meshes:
        bpy.data.meshes.remove(mesh)
    for mat in bpy.data.materials:
        bpy.data.materials.remove(mat)

def create_material(name, color):
    """Create a simple material with a base color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
    return mat

def create_pouch():
    # Dimensions
    width = 1.2
    depth = 0.4
    height = 2.5
    res_x = 32
    res_z = 64
    
    # Create a mesh and object
    mesh = bpy.data.meshes.new("PouchMesh")
    obj = bpy.data.objects.new("FoodPackagingPouch", mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()
    
    # Create a basic box shape (thin pillow)
    # We'll generate two sheets (front and back) and connect them
    verts_front = []
    verts_back = []
    
    for i in range(res_z + 1):
        z = (i / res_z) * height - (height / 2)
        # Pillow bulge factor: maximum at center of Z, zero at ends
        bulge_z = math.sin(math.pi * (i / res_z)) 
        
        for j in range(res_x + 1):
            x_norm = (j / res_x) * 2 - 1 # -1 to 1
            x = x_norm * (width / 2)
            
            # Pillow bulge factor: maximum at center of X, zero at edges
            bulge_x = math.cos(math.pi * x_norm / 2)
            
            # Total displacement for the "pillow" effect
            offset = depth * 0.5 * (bulge_z * bulge_x)
            
            # Front face
            verts_front.append(bm.verts.new((x, offset, z)))
            # Back face
            verts_back.append(bm.verts.new((x, -offset, z)))

    # Create faces for front and back
    for i in range(res_z):
        for j in range(res_x):
            v1 = verts_front[i * (res_x + 1) + j]
            v2 = verts_front[i * (res_x + 1) + j + 1]
            v3 = verts_front[(i + 1) * (res_x + 1) + j + 1]
            v4 = verts_front[(i + 1) * (res_x + 1) + j]
            bm.faces.new((v1, v2, v3, v4))
            
            v1b = verts_back[i * (res_x + 1) + j]
            v2b = verts_back[i * (res_x + 1) + j + 1]
            v3b = verts_back[(i + 1) * (res_x + 1) + j + 1]
            v4b = verts_back[(i + 1) * (res_x + 1) + j]
            bm.faces.new((v1b, v3b, v2b, v4b)) # Reversed winding

    # Connect front and back to seal the pouch
    for i in range(res_z + 1):
        for j in range(res_x + 1):
            # Only connect edges (sides, top, bottom)
            if j == 0 or j == res_x or i == 0 or i == res_z:
                vf = verts_front[i * (res_x + 1) + j]
                vb = verts_back[i * (res_x + 1) + j]
                # We handle edges separately to avoid overlapping faces
                pass

    # Side seals
    for i in range(res_z):
        # Left edge
        v1f = verts_front[i * (res_x + 1)]
        v2f = verts_front[(i + 1) * (res_x + 1)]
        v1b = verts_back[i * (res_x + 1)]
        v2b = verts_back[(i + 1) * (res_x + 1)]
        bm.faces.new((v1f, v1b, v2b, v2f))
        
        # Right edge
        v1f = verts_front[i * (res_x + 1) + res_x]
        v2f = verts_front[(i + 1) * (res_x + 1) + res_x]
        v1b = verts_back[i * (res_x + 1) + res_x]
        v2b = verts_back[(i + 1) * (res_x + 1) + res_x]
        bm.faces.new((v1f, v2f, v2b, v1b))

    # Top and Bottom seals
    for j in range(res_x):
        # Bottom seal
        v1f = verts_front[j]
        v2f = verts_front[j + 1]
        v1b = verts_back[j]
        v2b = verts_back[j + 1]

        bm.faces.new((v1f, v1b, v2b, v2f))

    # --- Geometry Details: Pinching and Crimping ---
    # Side seals are usually narrower/pinched
    for v in bm.verts:

def assign_materials(obj):
    # Define colors as requested
    colors = {
        "GreenPolka": (0.1, 0.6, 0.2, 1.0),
        "CoralOrange": (1.0, 0.5, 0.3, 1.0),
        "DarkBrownBlack": (0.1, 0.08, 0.05, 1.0),
        "WhiteStriped": (0.9, 0.9, 0.9, 1.0),
        "BlackStriped": (0.1, 0.1, 0.1, 1.0),
        "TanCream": (0.85, 0.75, 0.6, 1.0)
    }
    
    mats = {name: create_material(name, color) for name, color in colors.items()}
    obj.data.materials.join([mats[n] for n in colors])

    # Map faces to materials based on coordinates
    # Note: we use the original mesh data before modifiers
    mesh = obj.data
    height = 2.5
    width = 1.2
    
    for face in mesh.polygons:
        center = face.center
        z = center.z
        x = center.x
        
        # Side seams (Tan Cream) - based on x distance from center
        if abs(x) > (width / 2) * 0.85:
            face.material_index = 5 # TanCream
            continue
            
        # Top/Bottom seals (can also be tan or follow bands, usually follow bands but distinct)
        # For this model, let's prioritize the horizontal color bands
        
        # Vertical Bands logic
        # Green Polka-dot pattern area (Top)
        if z > height * 0.15:
            face.material_index = 0 # GreenPolka
        # Coral Orange middle section
        elif z > -height * 0.15:
            # Add some "white and black striped accents" here
            # If Z is near the boundaries of the coral section, make it a stripe
            if abs(z) > height * 0.1:
                if (int(z * 20)) % 2 == 0:
                    face.material_index = 3 # WhiteStriped
                else:
                    face.material_index = 4 # BlackStriped
            else:
                face.material_index = 1 # CoralOrange
        # Dark Brown-Black lower zone
        else:
            face.material_index = 2 # DarkBrownBlack
