def clear_scene():
    """Clears all default objects from the scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Creates a simple diffuse material with a specific color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    
    # Create Principled BSDF
    node_bsdf = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_bsdf.inputs['Base Color'].default_value = color
    
    # Create Output node
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    
    # Link them
    links = mat.node_tree.links
    links.new(node_bsdf.outputs['BSDF'], node_output.inputs['Surface'])
    return mat

def create_pot_lid():
    # Dimensions
    radius = 5.0
    thickness = 0.15
    dome_height = 0.4
    rim_width = 0.2
    knob_radius = 0.4
    knob_height = 1.2

    # Materials
    beige_mat = create_material("Beige", (0.93, 0.85, 0.78, 1.0)) # Light pinkish-beige
    pink_mat = create_material("Pink", (1.0, 0.4, 0.6, 1.0))      # Bright Pink
    blue_mat = create_material("DarkBlue", (0.02, 0.05, 0.25, 1.0)) # Dark Blue

    # --- Create Lid Body ---
    bm = bmesh.new()
    
    # Create a circular base disk
    bmesh.ops.create_circle(bm, cap_ends=True, radius=radius, segments=64)
    
    # Slightly dome the profile
    # We move vertices based on their distance from center: z = h * (1 - r^2/R^2)
    for v in bm.verts:
        dist_sq = v.co.x**2 + v.co.y**2
        v.co.z = dome_height * (1.0 - dist_sq / (radius**2))

    # Extrude for thickness
    # We extrude the whole thing along Z to give it volume
    geom = bm.verts[:] + bm.edges[:] + bm.faces[:]
    res = bmesh.ops.extrude_face_region(bm, geom=geom)
    
    # Move the extruded vertices down by thickness
    verts_extruded = [v for v in res['geom'] if isinstance(v, bmesh.types.BMVert)]
    for v in verts_extruded:
        v.co.z -= thickness

    # Create a rim: slightly widen the bottom edge or bevel the outer edges
    # For simplicity and precision, we'll create the rim by extruding the side wall outwards
    bm.edges.index_update()
    # Find perimeter edges of the original top disc
    outer_edges = [e for e in bm.edges if len(e.link_faces) == 1] # Simplified check for boundary
    # More robust: find edges where vertices are at radius distance
    rim_edges = [e for e in bm.edges if all(round(v.co.length, 2) >= radius - 0.1 for v in e.verts)]
    
    # Bevel the outer edge to create a rounded rim lip
    bmesh.ops.bevel(bm, geom=rim_edges, offset=rim_width, segments=3, affect='EDGES')

    # Finalize Lid Mesh
    lid_obj = bpy.data.objects.new("PotLid", bm)
    bpy.context.collection.objects.link(lid_obj)
    bm.to_mesh(lid_obj.data)
    bm.free()
    
    # Assign beige material to lid
    lid_obj.data.materials.append(beige_mat)

    # --- Create Knob Handle ---
    # The knob consists of two colored segments (pink and blue)
    def create_knob_segment(name, radius, height, z_offset, material):
        bm_k = bmesh.new()
        bmesh.ops.create_cylinder(
            bm_k, 
            cap_ends=True, 
            radius=radius, 
            depth=height, 
            segments=32
        )
        # Shift Z so the bottom of the cylinder is at z_offset
        for v in bm_k.verts:
            v.co.z += (z_offset + height/2)
            
        # Rounded top bevel for the upper segment
        if name == "KnobTop":
            top_edges = [e for e in bm_k.edges if all(abs(v.co.z - (z_offset + height)) < 0.1 for v in e.verts)]
            bmesh.ops.bevel(bm_k, geom=top_edges, offset=0.2, segments=8, affect='EDGES')

        obj_k = bpy.data.objects.new(name, bm_k)
        bpy.context.collection.objects.link(obj_k)
        bm_k.to_mesh(obj_k.data)
        bm_k.free()
        obj_k.data.materials.append(material)
        return obj_k

    # Place knob on top of the dome (z = dome_height)
    seg_h = knob_height / 2
    knob_bottom = create_knob_segment("KnobBottom", knob_radius, seg_h, dome_height, blue_mat)
    knob_top = create_knob_segment("KnobTop", knob_radius, seg_h, dome_height + seg_h, pink_mat)

    # Join all parts into one object for clean output
    bpy.ops.object.select_all(action='DESELECT')
    lid_obj.select_set(True)
    knob_bottom.select_set(True)
    knob_top.select_set(True)
    bpy.context.view_layer.objects.active = lid_obj
    bpy.ops.object.join()
