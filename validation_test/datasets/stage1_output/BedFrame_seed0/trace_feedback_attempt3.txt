import bpy

def clear_scene():
    """Clears the default scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Creates a simple material with a specific color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    node_principled = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    node_principled.inputs['Base Color'].default_value = color
    mat.node_tree.links.new(node_principled.outputs['BSDF'], node_output.inputs['Surface'])
    return mat

def add_box(bm, size, location):
    """Adds a box to the bmesh."""
    bmesh.ops.create_cube(bm, size=1.0)
    # The cube is created at origin with size 1 (extents -0.5 to 0.5)
    # We scale and translate the most recent vertices
    verts = bm.verts[-8:]
    for v in verts:
        v.co.x *= size[0]
        v.co.y *= size[1]
        v.co.z *= size[2]
        v.co += tuple(location)

def add_tapered_leg(bm, top_rad, bot_rad, height, location):
    """Adds a tapered leg to the bmesh."""
    # Create cone (radius1=bot, radius2=top, depth=height)
    # Note: create_cone is centered at origin along Z axis.
    bmesh.ops.create_cone(bm, cap_ends=True, segments=16, radius1=bot_rad, radius2=top_rad, depth=height)
    verts = bm.verts[-34:] # 16 * 2 + 2 caps
    for v in verts:
        # Shift from centered at origin to growing downwards from location
        v.co.z -= height / 2
        v.co += tuple(location)

def add_cylinder(bm, radius, height, location):
    """Adds a cylinder (accent ring) to the bmesh."""
    bmesh.ops.create_cone(bm, cap_ends=True, segments=16, radius1=radius, radius2=radius, depth=height)
    verts = bm.verts[-34:]
    for v in verts:
        v.co.z -= height / 2
        v.co += tuple(location)

def add_carved_panel(bm, width, height, depth, location):
    """Adds a decorative carved panel to the bmesh."""
    # Create base plate
    bmesh.ops.create_cube(bm, size=1.0)
    verts = bm.verts[-8:]
    for v in verts:
        v.co.x *= width / 2
        v.co.y *= depth / 2
        v.co.z *= height / 2
        v.co += tuple(location)
    
    # To simulate "carving" without complex boolean/inset ops that might be slow,
    # we add a slightly smaller recessed box inside it using vertices logic
    # However, for high fidelity as requested, let's use bmesh.ops.create_cube again 
    # and manually shift faces if needed, or just create the frame as separate pieces.
    # Simpler: Create the border (frame) and a recessed center.
    
    # Recessed inner panel
    inner_w = width * 0.85
    inner_h = height * 0.85
    inner_d = depth * 0.6
    bmesh.ops.create_cube(bm, size=1.0)
    iverts = bm.verts[-8:]
    for v in iverts:
        v.co.x *= inner_w / 2
        v.co.y *= inner_d / 2
        v.co.z *= inner_h / 2
        v.co += tuple(location)
