import bpy
import bmesh
import math
from mathutils import Vector

def clear_scene():
    """Clears the default scene of all objects."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_white_material():
    """Creates a simple clean white material."""
    mat = bpy.data.materials.new(name="WhiteCasingMat")
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = (0.95, 0.95, 0.95, 1.0)
        bsdf.inputs['Roughness'].default_value = 0.3
    return mat

def create_door_casing():
    # --- Parameters ---
    door_w = 0.9  # Standard width (~36 inches)
    door_h = 2.1  # Standard height (~84 inches)
    jamb_depth = 0.12 # Wall thickness/jamb depth
    trim_width = 0.08 # Width of the decorative casing
    trim_depth = 0.015 # Projection from wall
    
    # --- Create BMesh ---
    bm = bmesh.new()

    # 1. Create Jambs (The inner frame that sits in the wall)
    # Side jambs
    for side in [-1, 1]:
        x_start = (door_w / 2) * side
        # Creating a box for each side jamb
        bmesh.ops.create_cube(bm, size=1.0)
        # Scale and move to position
        # The cube is created at origin with size 1.0 (-0.5 to 0.5)
        # We want width = jamb_depth, depth = jamb_depth/2 (approx), height = door_h
        # Actually let's just use a helper function for boxes
        pass

    # Since bmesh.ops.create_cube is clunky with transforms, 
    # I will define vertices and faces manually or via small logic blocks.
    
    def add_box(bm, x, y, z, dx, dy, dz):
        """Helper to add a box into the BMesh."""
        verts = []
        for px in [0, dx]:
            for py in [0, dy]:
                for pz in [0, dz]:
                    verts.append(bm.verts.new(Vector((x + px, y + py, z + pz))))
        
        # Indices for faces
        faces = [
            [verts[0], verts[1], verts[3], verts[2]], # bottom
            [verts[4], verts[5], verts[7], verts[6]], # top
            [verts[0], verts[1], verts[5], verts[4]], # front
            [verts[2], verts[3], verts[7], verts[6]], # back
            [verts[0], verts[2], verts[6], verts[4]], # left
            [verts[1], verts[3], verts[7], verts[5]]  # right
        ]
        for f in faces:
            bm.faces.new(f)

    # JAMBS
    # Left jamb (X is negative, Y centered around 0)
    add_box(bm, -door_w/2 - jamb_depth, -jamb_depth/2, 0, jamb_depth, jamb_depth, door_h)
    # Right jamb
    add_box(bm, door_w/2, -jamb_depth/2, 0, jamb_depth, jamb_depth, door_h)
    # Top jamb (Header)
    add_box(bm, -door_w/2 - jamb_depth, -jamb_depth/2, door_h, jamb_depth * 2 + door_w, jamb_depth, jamb_depth / 2)

    # TRIM CASING PROFILE
    # We define a cross-section for the trim and extrude it.
    # Profile points (X=dist from wall, Y=height/width offset)
    # The profile is in the XY plane relative to its local orientation
    profile_pts = [
        (0, 0),                         # Point against the jamb
        (0, trim_depth),               # Outward projection
        (trim_width * 0.2, trim_depth), # Flat part
        (trim_width * 0.3, trim_depth * 0.7), # Bevel/Taper
        (trim_width, 0)                # Back to wall level (outer edge)
    ]

    def extrude_profile(bm, start_pos, direction, length, rotation_axis):
        """Extrudes a custom profile along a vector."""
        # Calculate the 'right' vector for the profile based on movement direction
        # If moving in Z (sides), profile is in XZ or YZ.
        # Here we keep it simple: 
        # Side trims move along Z, profile is in XY plane.
        # Top trim moves along X, profile is in XZ plane.
        
        verts = []
        for pt in profile_pts:
            v = bm.verts.new(Vector(start_pos) + Vector((pt[0], pt[1], 0)))
            verts.append(v)
        
        # Create face for the start cap
        bm.faces.new(verts)
        
        # Extrude along direction
        res = bmesh.ops.extrude_face(bm, face=[bm.faces[-1]])
        extruded_verts = [v for v in res['geom'] if isinstance(v, bmesh.types.BMVert)]
        
        # Move the extruded verts by length * direction
        for v in extruded_verts:
            v.co += Vector(direction) * length

    # We need to handle the trim pieces separately for different orientations.
    # Left Trim
    # Start pos: -door_w/2, 0 (at wall surface), Z=0
    # Direction: (0,0,1). Profile is in XY plane (x offset is distance from door)
    # Since it's the left side, the profile "width" should go further negative.
    def add_trim_segment(bm, start_pos, dir_vec, length, profile_dir):
        """
        start_pos: Vector
        dir_vec: Direction of extrusion (Vector)
        length: How far to extrude
        profile_dir: The direction the trim 'sticks out' from the wall (Vector)
        """
        # Create a local coordinate system for the profile
        # We need a perpendicular vector for the thickness/depth
        up = Vector((0, 1, 0)) if abs(dir_vec.dot(Vector((0, 1, 0)))) < 0.9 else Vector((0, 0, 1))
        right = dir_vec.cross(up).normalized()
        up = right.cross(dir_vec).normalized()
        
        # profile_dir is the 'outward' vector from wall
        out_vec = profile_dir.normalized()
        side_vec = dir_vec.cross(out_vec).normalized()

        verts = []
        for pt in profile_pts:
            # Point = start + (dist_from_wall * out_vec) + (offset * side_vec)
            # But for a door casing, the 'side_vec' is usually 0 because it's flush.
            # Actually, we want the profile to be in the plane of the wall face.
            # Let's rethink: Profile pts are (dist_from_wall, height_above_baseline)
            # No, let's use (dist_from_jamb, thickness_out_from_wall).
            v = bm.verts.new(start_pos + (Vector((pt[0], 0, 0)) * side_vec) + (Vector((0, pt[1], 0)) * out_vec))
            verts.append(v)
        
        face = bm.faces.new(verts)
        res = bmesh.ops.extrude_face(bm, face=[face])
        extruded_verts = [v for v in res['geom'] if isinstance(v, bmesh.types.BMVert)]
        for v in extruded_verts:
            v.co += dir_vec * length

    # Let's use a more robust approach for the trim segments: 3 boxes with bevels
    # because BMesh extrusion of profiles can be tricky to miter perfectly.
    # I will create 3 "slabs" and then apply a bevel modifier or manually cut them.
    
    def add_trim_slab(bm, x, y, z, dx, dy, dz):
        add_box(bm, x, y, z, dx, dy, dz)

    # LEFT TRIM SLAB
    # Positioned at -door_w/2, sticking out from wall (y > 0), width = trim_width
    add_trim_slab(bm, -door_w/2 - trim_width, 0, 0, trim_width, trim_depth, door_h)
    # RIGHT TRIM SLAB
    add_trim_slab(bm, door_w/2, 0, 0, trim_width, trim_depth, door_h)
    # TOP TRIM SLAB
    add_trim_slab(bm, -door_w/2 - trim_width, 0, door_h, 
                  door_w + 2*trim_width, trim_depth, trim_width)

    # Now we apply a "chamfer" to the inner and outer edges of these slabs
    # To make it look like architectural molding.
    # Since this is procedural script, I'll use a Modifier for the final polish.

    bm.to_mesh("DoorCasing")
    bm.free()

def finalize_object():
    obj = bpy.data.objects["DoorCasing"]
    bpy.context.view_layer.objects.active = obj
    
    # Add a bevel modifier to simulate the architectural profile (the "clean" look)
    bev = obj.modifiers.new(name="Bevel", type='BEVEL')
    bev.width = 0.01
    bev.segments = 3
    bev.limit_method = 'ANGLE'
    bev.angle_limit = 0.785 # 45 degrees
    
    # Apply material
    mat = create_white_material()
    obj.data.materials.append(mat)

if __name__ == "__main__":
    clear_scene()
    create_door_casing()
    finalize_object()
