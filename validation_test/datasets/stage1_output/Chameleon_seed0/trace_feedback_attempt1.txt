import bpy
import bmesh
import math
import random
from mathutils import Vector

def clear_scene():
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
    return mat

def add_bumpy_texture(obj):
    # Create a procedural noise texture for displacement to simulate leathery skin
    tex = bpy.data.textures.new("SkinNoise", type='NOISE')
    tex.noise_scale = 0.05
    
    mod = obj.modifiers.new(name="Bumps", type='DISPLACE')
    mod.texture = tex
    mod.strength = 0.03
    
    # Apply the modifier to make it real geometry as requested
    bpy.context.view_layer.objects.active = obj
    bpy.ops.object.modifier_apply(modifier="Bumps")

def create_chameleon():

    # Materials

    rings = 8

        t = i / segments

        y_pos = -1.8 - (t * length)
        
        # Create ring
        verts = []
        for j in range(rings):
            angle = (2 * math.pi / rings) * j
            vx = math.cos(angle) * r
            vz = math.sin(angle) * r
            verts.append(bm.verts.new((vx, y_pos, vz)))
        
        # Connect to previous ring
        if i > 0:
            prev_ring_start = (i-1) * rings
            for j in range(rings):
                v1 = bm.verts[prev_ring_start + j]
                v2 = bm.verts[i * rings + j]
                # Since we are doing indices, it's safer to track ring lists
                pass 

    # Re-do tail with explicit list for connectivity
    bm = bmesh.new()
    all_rings = []
    for i in range(segments + 1):
        t = i / segments
        r = radius_start * (1 - t) + radius_end * t
        y_pos = -1.8 - (t * length)
        ring = []
        for j in range(rings):
            angle = (2 * math.pi / rings) * j
            ring.append(bm.verts.new((math.cos(angle)*r, y_pos, math.sin(angle)*r)))
        all_rings.append(ring)

                bm.faces.new((v1, v2, v3, v4))

    last_ring = all_rings[-1]

    # We'll combine the main parts into one object for a unified skin bump effect
