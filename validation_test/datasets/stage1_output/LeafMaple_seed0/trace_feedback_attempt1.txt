import bpy
import bmesh
import math
import random
from mathutils import Vector

def clear_scene():
    """Clears the default Blender scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_maple_leaf():
    # 1. Define the rough outline of a maple leaf (Palmate shape)
    # Coordinates as (x, y) in 2D plane
    # We define key "anchor" points for the lobes and gaps
    outline_points = [
        (0.0, 0.0),             # Base
        (-0.3, 0.2),            # Inner gap L1
        (-1.5, 0.4),            # Tip Lobe 1 (Bottom Left)
        (-0.6, 0.7),            # Inner gap L2
        (-1.8, 1.5),            # Tip Lobe 2 (Mid Left)
        (-0.6, 1.3),            # Inner gap L3
        (-0.4, 2.2),            # Tip Lobe 3 (Top Left)
        (0.0, 1.8),             # Center Top Gap
        (0.4, 2.2),             # Tip Lobe 4 (Top Right)
        (0.6, 1.3),             # Inner gap L4
        (1.8, 1.5),             # Tip Lobe 5 (Mid Right)
        (0.6, 0.7),             # Inner gap L5
        (1.5, 0.4),             # Tip Lobe 6 (Bottom Right)
        (0.3, 0.2),             # Inner gap L6
        (0.0, 0.0),             # Close loop at base
    ]

    # Interpolate points to create a smoother high-res outline
    res = 15 # vertices per segment
    smooth_outline = []
    for i in range(len(outline_points) - 1):
        p1 = Vector(outline_points[i])
        p2 = Vector(outline_points[i+1])
        for j in range(res):
            t = j / res
            smooth_outline.append(p1.lerp(p2, t))

    # Create the Mesh and BMesh
    mesh = bpy.data.meshes.new("MapleLeaf")
    obj = bpy.data.objects.new("MapleLeaf", mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()

    # 2. Create the perimeter vertices and apply serration (sharp teeth)
    final_outline = []
    for i in range(len(smooth_outline)):
        p = smooth_outline[i]
        # Calculate a tangent/normal for the offset
        prev_p = smooth_outline[i-1] if i > 0 else smooth_outline[-1]
        next_p = smooth_outline[(i+1)%len(smooth_outline)]
        
        edge_dir = (next_p - prev_p).normalized()
        normal = Vector((-edge_dir.y, edge_dir.x, 0))
        
        # Serration: High frequency jitter/sine wave along the perimeter
        # We skip the base area for serrations
        if p.length() > 0.5:
            serration_amp = 0.04 * (1.0 + random.random() * 0.5)
            offset = normal * (math.sin(i * 0.8) * serration_amp)
            final_outline.append(p + offset)
        else:
            final_outline.append(p)

    # Create perimeter vertices in BMesh
    verts = [bm.verts.new((v.x, v.y, 0)) for v in final_outline]
    
    # Close the loop with edges
    for i in range(len(verts)):
        bm.edges.new((verts[i], verts[(i+1)%len(verts)]))

    # Fill the face to create a surface
    # Use bmesh.ops.contextual_create for filling complex polygons
    bm.faces.new(verts)

    # 3. Subdivide the leaf blade to allow for undulation and veins
    # We can't easily use modifiers in script before adding to scene,
    # so we do manual subdivision or use a grid approach.
    # Actually, let's create a denser internal mesh using a different method:
    # Create a center point and connect it to the perimeter (fan)
    center_vert = bm.verts.new((0, 0, 0))
    
    # Remove the large single face and replace with triangles from center
    bm.faces.ensure_lookup_table()
    large_face = bm.faces[0]
    bm.faces.remove(large_face)
    
    for i in range(len(verts)):
        bm.faces.new((center_vert, verts[i], verts[(i+1)%len(verts)]))

    # Subdivide the triangles for better geometry
    for _ in range(3): 
        bmesh.ops.subdivide_edges(bm, edges=bm.edges, cuts=1)

    # 4. Apply Undulations and Veins (Geometric Displacement)
    bm.verts.ensure_lookup_table()
    for v in bm.verts:
        x, y = v.co.x, v.co.y
        dist = math.sqrt(x*x + y*y)
        
        # Organic curling: Lift the center and drop the edges slightly
        # Use a Gaussian-like hump for the main body
        z_base = 0.15 * math.exp(-dist**2 / 1.5)
        
        # Add undulating waves (perlin-like simple sine)
        wave = 0.03 * math.sin(x * 4) * math.cos(y * 4)
        
        # Fold the leaf slightly across the x axis for a 3/4 look
        fold = 0.1 * (x**2) * 0.5 
        
        # Vein elevation: define paths from center to lobes
        vein_depth = 0.0
        lobe_tips = [Vector((0, 2.2)), Vector((1.8, 1.5)), Vector((-1.8, 1.5)), 
                     Vector((1.5, 0.4)), Vector((-1.5, 0.4))]
        for tip in lobe_tips:
            # distance to line segment (center -> tip)
            line = tip - Vector((0,0,0))
            proj = (Vector((x,y,0)).dot(line) / line.length_squared()) * line
            dist_to_vein = (Vector((x,y,0)) - proj).length()
            if dist_to_vein < 0.08:
                # Create a raised ridge for the vein
                vein_depth += (0.06 - dist_to_vein) * 1.2
        
        v.co.z = z_base + wave + fold + vein_depth

    # 5. The Petiole Stem
    # Create a cylinder for the stem starting at origin and going down
    stem_verts = []
    segments = 8
    rings = 8
    radius = 0.05
    length = 1.5
    
    for i in range(segments + 1):
        z = - (i / segments) * length
        # Slight curve in the stem
        x_offset = 0.2 * math.sin(i/segments * math.pi * 0.5)
        for j in range(rings):
            angle = (j / rings) * 2 * math.pi
            sx = x_offset + radius * math.cos(angle)
            sy = radius * math.sin(angle)
            stem_verts.append(bm.verts.new((sx, sy, z)))

    # Connect stem vertices into a tube
    for i in range(segments):
        for j in range(rings):
            v1 = stem_verts[i*rings + j]
            v2 = stem_verts[i*rings + (j+1)%rings]
            v3 = stem_verts[(i+1)*rings + (j+1)%rings]
            v4 = stem_verts[(i+1)*rings + j]
            bm.faces.new((v1, v2, v3, v4))

    # Close the bottom of the stem
    bottom_ring = [stem_verts[segments*rings + j] for j in range(rings)]
    bm.faces.new(bottom_ring)

    # Connect stem to leaf base (bridge the gap)
    # The top ring of the stem connects to center_vert and surrounding area
    top_ring = [stem_verts[0*rings + j] for j in range(rings)]
    for j in range(rings):
        bm.faces.new((top_ring[j], top_ring[(j+1)%rings], center_vert))

    # Finalize BMesh
    bm.to_mesh(mesh)
    bm.free()

    # Set rotation for 3/4 perspective
    obj.rotation_euler = (math.radians(-20), 0, math.radians(45))
    # Move slightly to center visual weight
    obj.location = (0, 0, 0)

if __name__ == "__main__":
    clear_scene()
    create_maple_leaf()
