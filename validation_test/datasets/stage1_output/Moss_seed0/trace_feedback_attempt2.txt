import bpy
import bmesh
import math
import random
from mathutils import Vector

def clear_scene():
    """Clears all default objects from the scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()
    for block in bpy.data.meshes:
        bpy.data.meshes.remove(block)
    for block in bpy.data.materials:
        bpy.data.materials.remove(block)

def create_moss_material():
    """Creates a muted olive-brown material."""
    mat = bpy.data.materials.new(name="MossMaterial")
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    
    node_bsdf = nodes.new(type='ShaderNodeBsdfPrincipled')
    # Muted olive-brown: Olive-greenish brown
    node_bsdf.inputs['Base Color'].default_value = (0.25, 0.3, 0.15, 1.0)
    node_bsdf.inputs['Roughness'].default_value = 0.9
    
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    links = mat.node_tree.links
    links.new(node_bsdf.outputs['BSDF'], node_output.inputs['Surface'])
    return mat

def create_moss_fragment():
    """Constructs the moss fragment geometry procedurally using BMesh."""
    # Parameters for the crescent wedge
    outer_radius = 1.0
    inner_radius = 0.6
    height = 0.3
    segments = 48
    arc_angle = math.pi * 1.2 # Slightly more than a semi-circle
    
    bm = bmesh.new()
    
    # 1. Create the bottom crescent face (flat base)
    outer_verts = []
    inner_verts = []
    
    for i in range(segments + 1):
        angle = (i / segments) * arc_angle - (arc_angle / 2)
        cos_a, sin_a = math.cos(angle), math.sin(angle)
        
        # Outer Arc
        ov = bm.verts.new(Vector((cos_a * outer_radius, sin_a * outer_radius, 0)))
        outer_verts.append(ov)
        
        # Inner Arc
        iv = bm.verts.new(Vector((cos_a * inner_radius, sin_a * inner_radius, 0)))
        inner_verts.append(iv)
    
    # Create base faces
    for i in range(segments):
        bm.faces.new((outer_verts[i], outer_verts[i+1], inner_verts[i+1], inner_verts[i]))
    
    # 2. Extrude upwards to create the wedge volume
    # We use bmesh.ops.extrude_face_region
    bottom_faces = bm.faces[:]
    res = bmesh.ops.extrude_face_region(bm, geom=bottom_faces)
    extruded_verts = [v for v in res['geom'] if isinstance(v, bmesh.types.BMVert)]
    
    # Move extruded vertices up and add slight variation to the top surface
    for v in extruded_verts:
        # Base height + random organic jitter
        jitter = random.uniform(-0.05, 0.1)
        v.co.z += (height + jitter)
        
    # 3. Add "clumps" and filaments to the top surface
    # Identify faces on the top (those with normals pointing roughly up)
    bm.faces.ensure_lookup_table()
    top_faces = [f for f in bm.faces if f.normal.z > 0.8]
    
    # Pick a random subset of top faces to extrude further into organic clumps
    num_clumps = int(len(top_faces) * 0.4)
    target_faces = random.sample(top_faces, num_clumps)
    
    for f in target_faces:
        # Extrude a small "bump"
        res_bump = bmesh.ops.extrude_face_region(bm, geom=[f])
        bump_verts = [v for v in res_bump['geom'] if isinstance(v, bmesh.types.BMVert)]
        
        # Move bump vertices up and scale them inwards to create a rounded blob
        center = f.calc_center_median()
        bump_height = random.uniform(0.05, 0.15)
        scale_factor = random.uniform(0.4, 0.8)
        
        for v in bump_verts:
            # Push up and pinch inwards relative to face center
            v.co.z += bump_height
            v.co.x = center.x + (v.co.x - center.x) * scale_factor
            v.co.y = center.y + (v.co.y - center.y) * scale_factor

    # 4. Finalize Mesh
    bm.to_mesh("moss_mesh")
    bm.free()
    
    obj = bpy.data.objects.new("MossFragment", bpy.data.meshes["moss_mesh"])
    bpy.context.collection.objects.link(obj)
    
    # 5. Smooth the object and round edges
    # Subdivision Surface modifier for that organic, rounded look
    subsurf = obj.modifiers.new(name="Subdiv", type='SUBSURF')
    subsurf.levels = 2
    subsurf.render_levels = 2
    
    # Set shading to smooth
    for poly in obj.data.polygons:
        poly.use_smooth = True

    return obj

def main():
    clear_scene()
    
    # Material setup
    moss_mat = create_moss_material()
    
    # Geometry construction
    fragment = create_moss_fragment()
    
    # Assign material
    fragment.data.materials.append(moss_mat)

if __name__ == "__main__":
    main()
