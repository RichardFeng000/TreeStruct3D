def clear_scene():
    """Clears the default blender scene."""

def create_material(name, color):
    """Creates a simple material with a specific diffuse color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    # Find the Principled BSDF node
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
        # Set slightly higher roughness for wood feel
        bsdf.inputs['Roughness'].default_value = 0.7
    return mat

def create_box(name, width, depth, height, location):
    """Creates a cube-based box primitive."""
    bpy.ops.mesh.primitive_cube_add(size=1.0, location=location)
    obj = bpy.context.active_object
    obj.name = name
    # Scale the unit cube (which is 2x2x2 by default) to desired dimensions
    obj.scale = (width / 2, depth / 2, height / 2)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    return obj
