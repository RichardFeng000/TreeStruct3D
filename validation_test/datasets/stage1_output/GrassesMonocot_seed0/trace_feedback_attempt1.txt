import bmesh
import math
import random
from mathutils import Vector

def clear_scene():
    """Clears all objects from the default Blender scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_grass_stalk():
    """Constructs a procedural grass stalk with a curved stem and alternating leaves."""
    
    # Parameters
    height = 4.0
    segments = 32
    stem_radius = 0.015
    num_leaves = 8
    leaf_length_max = 1.2
    leaf_width_max = 0.06
    curvature_amplitude = 0.4
    
    # --- STEM GENERATION ---
    # Generate points for the stem path with a gentle curve
    path_points = []
    for i in range(segments + 1):
        t = i / segments
        z = t * height
        # Simple parametric curve for organic look
        x = math.sin(t * math.pi * 0.8) * curvature_amplitude * t
        y = math.cos(t * math.pi * 0.5 + 1.0) * (curvature_amplitude * 0.5) * t
        path_points.append(Vector((x, y, z)))

    # Use BMesh to create the stem mesh
    bm_stem = bmesh.new()
    
    # Create circular rings along the path
    ring_res = 8
    rings = []
    for p in path_points:
        ring = []
        for j in range(ring_res):
            angle = (2 * math.pi * j) / ring_res
            vx = p.x + math.cos(angle) * stem_radius
            vy = p.y + math.sin(angle) * stem_radius
            vz = p.z
            ring.append(bm_stem.verts.new(Vector((vx, vy, vz))))
        rings.append(ring)

    # Bridge rings to form the tube
    for i in range(len(rings) - 1):
        r1 = rings[i]
        r2 = rings[i+1]
        for j in range(ring_res):
            v1 = r1[j]
            v2 = r1[(j + 1) % ring_res]
            v3 = r2[(j + 1) % ring_res]
            v4 = r2[j]
            bm_stem.faces.new((v1, v2, v3, v4))

    # Finalize stem object
    stem_obj = bpy.data.objects.new("GrassStem", bm_stem)
    bpy.context.collection.objects.link(stem_obj)
    bm_stem.to_mesh(stem_obj.data)
    bm_stem.free()

    # --- LEAF GENERATION ---
    # Create leaves at intervals along the stem
    for i in range(num_leaves):
        # Position leaf along the height (not at the very bottom or top)
        t = 0.2 + (i / (num_leaves - 1)) * 0.6 if num_leaves > 1 else 0.5
        idx = int(t * segments)
        p_start = path_points[idx]
        
        # Determine leaf orientation based on the stem's tangent
        if idx < segments:
            tangent = (path_points[idx+1] - path_points[idx]).normalized()
        else:
            tangent = (path_points[idx] - path_points[idx-1]).normalized()
            
        # Create a coordinate frame for the leaf
        up = Vector((0, 0, 1))
        right = tangent.cross(up).normalized()
        if right.length < 0.1: # Handle vertical stem case
            right = Vector((1, 0, 0))
        up_leaf = right.cross(tangent).normalized()

        # Alternate sides
        side_dir = right if (i % 2 == 0) else -right
        
        # Leaf parameters
        l_len = random.uniform(0.6, leaf_length_max)
        l_wid = random.uniform(0.03, leaf_width_max)
        l_segs = 12

        bm_leaf = bmesh.new()
        
        # Create leaf geometry as a tapered ribbon
        prev_v_left = None
        prev_v_right = None
        
        for s in range(l_segs + 1):
            st = s / l_segs
            # Curvature of the leaf (drooping)
            curv_z = -math.sin(st * math.pi * 0.5) * 0.5
            curv_out = math.cos(st * math.pi * 0.2) * side_dir
            
            # Taper width
            current_width = l_wid * (1.0 - st**0.7)
            
            # Position of center line
            center = p_start + (side_dir * st * l_len) + (Vector((0, 0, curv_z)))
            
            # Offset for left and right edges
            offset = up_leaf * (current_width / 2.0)
            v_left = bm_leaf.verts.new(center + offset)
            v_right = bm_leaf.verts.new(center - offset)
            
            if prev_v_left is not None:
                bm_leaf.faces.new((prev_v_left, v_left, v_right, prev_v_right))
                
            prev_v_left = v_left
            prev_v_right = v_right

        # Finalize leaf object
        leaf_obj = bpy.data.objects.new(f"Leaf_{i}", bm_leaf)
        bpy.context.collection.objects.link(leaf_obj)
        bm_leaf.to_mesh(leaf_obj.data)
        bm_leaf.free()

if __name__ == "__main__":
    clear_scene()
    create_grass_stalk()
