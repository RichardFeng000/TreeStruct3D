def create_material(name, color):
    """Creates a simple diffuse material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
    return mat

def create_cylinder(name, radius_bottom, radius_top, height, location, rotation=(0, 0, 0), material=None):
    """Creates a cylinder/cone based on radii and height."""
    mesh = bpy.data.meshes.new(name)
    obj = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(obj)

    bm = bmesh.new()
    # Create a cylinder with vertices at top and bottom
    # Using bmesh.ops.create_cone for truncated cones (tapered cylinders)
    bmesh.ops.create_cone(
        bm, 
        cap_ends=True, 
        segments=32, 
        radius1=radius_bottom, 
        radius2=radius_top, 
        depth=height
    )
    
    bm.to_mesh(mesh)
    bm.free()

    obj.location = location
    obj.rotation_euler = rotation
    
    if material:
        obj.data.materials.append(material)
        
    return obj
