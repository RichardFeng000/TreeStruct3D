import bpy
import bmesh
import math
import random
from mathutils import Vector, Matrix

def clear_scene():
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    node_principled = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    node_principled.inputs['Base Color'].default_value = color
    # Make fruit glossy
    if "Fruit" in name:
        node_principled.inputs['Roughness'].default_value = 0.2
    node_output.inputs['Surface'].default_value = node_principled.outputs['BSDF']
    mat.node_tree.links.new(node_principled.outputs['BSDF'], node_output.inputs['Surface'])
    return mat

def create_strawberry():
    # Materials
    fruit_mat = create_material("FruitMat", (0.9, 0.2, 0.3, 1.0)) # Salmon-red
    seed_mat = create_material("SeedMat", (0.95, 0.9, 0.7, 1.0))  # White-cream
    leaf_mat = create_material("LeafMat", (0.1, 0.6, 0.1, 1.0))   # Bright green

    # --- FRUIT BODY ---
    # Create a base sphere and deform it into a conical shape
    bpy.ops.mesh.primitive_uv_sphere_add(segments=64, ring_count=32, radius=1.0)
    fruit = bpy.context.active_object
    fruit.name = "StrawberryBody"
    
    bm = bmesh.new()
    bm.from_mesh(fruit.data)
    
    for v in bm.verts:
        # Conical taper: scale X and Y based on Z
        # Z goes from -1 to 1. Map it so top is wider, bottom tapers.
        z = v.co.z
        # Scale factor: linear interpolation between top (wider) and bottom (tapered)
        scale_factor = 0.5 + (z + 1.0) * 0.4 # Top (z=1) scale ~ 1.3, Bottom (z=-1) scale ~ 0.5
        v.co.x *= scale_factor
        v.co.y *= scale_factor
        # Stretch vertically
        v.co.z *= 1.4
        
    bm.to_mesh(fruit.data)
    bm.free()
    
    # Smooth shading
    for poly in fruit.data.polygons:
        poly.use_smooth = True
    fruit.data.materials.append(fruit_mat)

    # --- SEEDS AND DIMPLES ---
    # We will use BMesh again to create slight dimples and place seed objects
    bm = bmesh.new()
    bm.from_mesh(fruit.data)
    
    seed_count = 250
    seeds_placed = 0
    
    # To distribute seeds evenly, we sample the vertices of the fruit
    sample_verts = list(bm.verts)
    random.shuffle(sample_verts)
    
    for v in sample_verts:
        if seeds_placed >= seed_count:
            break
        
        # 1. Create a dimple: Push nearby vertices inward slightly
        center = v.co.copy()
        radius = 0.15
        for nearby in bm.verts:
            dist = (nearby.co - center).length
            if dist < radius:
                # Move towards the origin of the fruit
                direction = (nearby.co - Vector((0,0,0))).normalized()
                push_amount = (radius - dist) * 0.1
                nearby.co -= direction * push_amount

        # 2. Create a seed mesh
        bpy.ops.mesh.primitive_uv_sphere_add(segments=8, ring_count=4, radius=0.03)
        seed = bpy.context.active_object
        seed.name = f"Seed_{seeds_placed}"
        
        # Scale seed to be elongated (ellipsoid)
        seed.scale = (0.6, 0.3, 1.0)
        
        # Position and rotate seed to face away from center
        seed.location = center + (center.normalized() * 0.02)
        rot_quat = seed.location.to_track_quat('Z', 'Z')
        seed.rotation_mode = 'QUATERNION'
        seed.rotation_quaternion = rot_quat
        
        seed.data.materials.append(seed_mat)
        # Parent to fruit for easier movement if needed, but prompt says origin
        seed.parent = fruit
        
        seeds_placed += 1

    bm.to_mesh(fruit.data)
    bm.free()

    # --- CALYX (LEAVES) ---
    leaf_count = 7
    for i in range(leaf_count):
        bpy.ops.mesh.primitive_plane_add(size=1.0, location=(0, 0, 1.3))
        leaf = bpy.context.active_object
        leaf.name = f"Sepal_{i}"
        
        # Shape the leaf
        bm = bmesh.new()
        bm.from_mesh(leaf.data)
        
        # Move vertices to create a pointed, broad leaf shape
        for v in bm.verts:
            # Map square plane to organic leaf shape
            x, y = v.co.x, v.co.y
            dist = (x**2 + y**2)**0.5
            if dist > 0.5:
                # Taper the edges
                v.co.x *= (1.0 - dist * 0.5)
                v.co.y *= (1.0 - dist * 0.5)
            # Add crinkles/noise
            v.co.z += random.uniform(-0.05, 0.05)
            
        bm.to_mesh(leaf.data)
        bm.free()
        
        # Position the leaf: rotate and move it outward
        angle = (2 * math.pi / leaf_count) * i
        leaf.rotation_euler = (math.radians(30 + random.uniform(-10, 10)), 0, angle)
        
        # Move to top of fruit, offset from center
        offset = 0.2
        leaf.location.x = math.cos(angle) * offset
        leaf.location.y = math.sin(angle) * offset
        leaf.location.z = 1.35
        
        # Bend the leaf outwards slightly
        # Using a simple scale and rotation to simulate the "spread"
        leaf.scale = (0.6, 0.8, 1.0)
        
        leaf.data.materials.append(leaf_mat)

    # --- STEM ---
    bpy.ops.mesh.primitive_cylinder_add(radius=0.05, depth=0.4, location=(0, 0, 1.5))
    stem = bpy.context.active_object
    stem.name = "Stem"
    # Taper the stem slightly
    bm = bmesh.new()
    bm.from_mesh(stem.data)
    for v in bm.verts:
        if v.co.z > 0: # Top of stem
            v.co.x *= 0.7
            v.co.y *= 0.7
    bm.to_mesh(stem.data)
    bm.free()
    stem.data.materials.append(leaf_mat)

if __name__ == "__main__":
    clear_scene()
    create_strawberry()
