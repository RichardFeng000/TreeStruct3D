import bpy

def clear_scene():
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color=(1, 1, 1, 1)):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    node_principled = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_principled.inputs['Base Color'].default_value = color
    node_principled.inputs['Roughness'].default_value = 0.05
    # IOR for ceramic is roughly 1.5 - 1.6
    if 'IOR' in node_principled.inputs:
        node_principled.inputs['IOR'].default_value = 1.5
    
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    mat.node_tree.links.new(node_principled.outputs['BSDF'], node_output.inputs['Surface'])
    return mat

def create_bowl():
    # Profile points: (radius, height)
    # Pedestal base -> basin bottom -> basin rim
    profile = [
        (0.0, 0.0),          # Center bottom
        (0.15, 0.0),         # Bottom edge
        (0.16, 0.2),         # Pedestal rise
        (0.25, 0.3),         # Bowl widest part (bottom of basin)
        (0.3, 0.6),          # Basin rim
    ]
    
    # We'll use a simpler bmesh approach to avoid the 'collapsed center' issue in create_revolved_mesh
    bm = bmesh.new()
    segments = 32
    rings = []
    
    for px, pz in profile:
        ring = []
        if px == 0:
            v = bm.verts.new(Vector((0, 0, pz)))
            ring.append(v)
        else:
            for i in range(segments):
                angle = (2 * math.pi * i) / segments
                ring.append(bm.verts.new(Vector((px * math.cos(angle), px * math.sin(angle), pz))))
        rings.append(ring)

    for r in range(len(rings) - 1):
        curr_r = rings[r]
        next_r = rings[r+1]
        
        if len(curr_r) == 1: # Center point
            for i in range(segments):
                next_i = (i + 1) % segments
                bm.faces.new([curr_r[0], next_r[next_i], next_r[i]])
        elif len(next_r) == 1: # Collapsing to center point
            for i in range(segments):
                prev_i = (i - 1) % segments
                bm.faces.new([curr_r[i], curr_r[prev_i], next_r[0]])
        else: # Ring to ring
            for i in range(segments):
                next_i = (i + 1) % segments
                bm.faces.new([curr_r[i], curr_r[next_i], next_r[next_i], next_r[i]])

    # Add a thickness to the rim for realism
    # We'll just use a subdivision surface and bevel later on the object
    bm.to_mesh(bpy.data.meshes.new("BowlMesh"))
    obj = bpy.data.objects.new("ToiletBowl", bpy.data.meshes.get("BowlMesh"))
    bpy.context.collection.objects.link(obj)
    
    subdiv = obj.modifiers.new(name="Subdiv", type='SUBSURF')
    subdiv.levels = 2
    
    return obj

def create_tank():
    bm = bmesh.new()
    bmesh.ops.create_cube(bm, size=1.0)
    
    # Dimensions: Width ~ 0.45, Depth ~ 0.25, Height ~ 0.4
    # Scale and move to be behind the bowl
    for v in bm.verts:
        v.co.x *= 0.22 # Half width
        v.co.y *= 0.13 # Half depth
        v.co.z *= 0.2  # Half height
        v.co.z += 0.6   # Move up to top of bowl
        v.co.y -= 0.3   # Move back relative to center

    bm.to_mesh(bpy.data.meshes.new("TankMesh"))
    obj = bpy.data.objects.new("ToiletTank", bpy.data.meshes.get("TankMesh"))
    bpy.context.collection.objects.link(obj)
    
    # Bevel for porcelain softness
    bevel = obj.modifiers.new(name="Bevel", type='BEVEL')
    bevel.width = 0.04
    bevel.segments = 5
    
    subdiv = obj.modifiers.new(name="Subdiv", type='SUBSURF')
    subdiv.levels = 1
    
    return obj

def create_seat():
    # Create a flat ring (torus-like)
    bm = bmesh.new()
    segments = 32
    outer_r = 0.31
    inner_r = 0.22
    height = 0.6
    thickness = 0.04

    # Outer ring
    v_outer = []
    for i in range(segments):
        angle = (2 * math.pi * i) / segments
        v_outer.append(bm.verts.new(Vector((outer_r * math.cos(angle), outer_r * math.sin(angle), height))))

    # Inner ring
    v_inner = []
    for i in range(segments):
        angle = (2 * math.pi * i) / segments
        v_inner.append(bm.verts.new(Vector((inner_r * math.cos(angle), inner_r * math.sin(angle), height))))

    # Bridge rings for the top face
    for i in range(segments):
        next_i = (i + 1) % segments
        bm.faces.new([v_outer[i], v_outer[next_i], v_inner[next_i], v_inner[i]])

    # Extrude for thickness
    bm.verts.ensure_lookup_table()
    faces = list(bm.faces)
    res = bmesh.ops.extrude_face_region(bm, geom=faces)
    for v in res['geom']:
        v.co.z += thickness

    # To make the seat look "round" and not flat, we'll apply a slight Y scale to the final object
    bm.to_mesh(bpy.data.meshes.new("SeatMesh"))
    obj = bpy.data.objects.new("ToiletSeat", bpy.data.meshes.get("SeatMesh"))
    bpy.context.collection.objects.link(obj)
    
    # Make the seat oval
    obj.scale[1] = 1.25
    
    bev = obj.modifiers.new(name="Bevel", type='BEVEL')
    bev.width = 0.01
    
    return obj

def create_lid():
    # Oval seat lid tilted upright
    bm = bmesh.new()
    bmesh.ops.create_cylinder(bm, segments=32, radius=0.32, depth=0.04)
    
    # Make it an oval by scaling later or manually now
    for v in bm.verts:
        v.co.y *= 1.25

    bm.to_mesh(bpy.data.meshes.new("LidMesh"))
    obj = bpy.data.objects.new("ToiletLid", bpy.data.meshes.get("LidMesh"))
    bpy.context.collection.objects.link(obj)
    
    # Pivot point is at the back of the seat: Y ~ -0.2, Z ~ 0.6
    # First move geometry so origin is at the pivot
    for v in obj.data.vertices:
        v.co.y += 0.2
        v.co.z -= 0.6

    obj.location = (0, -0.2, 0.6)
    # Tilt upright (roughly 80 degrees)
    obj.rotation_euler[0] = math.radians(-80)
    
    subdiv = obj.modifiers.new(name="Subdiv", type='SUBSURF')
    subdiv.levels = 1
    
    return obj

def create_handle():
    # Small flush handle on the side of the tank
    bm = bmesh.new()
    bmesh.ops.create_cube(bm, size=0.05)
    for v in bm.verts:
        v.co.x *= 0.3 # Length along X (outward from tank)
        v.co.y *= 0.4 
        v.co.z *= 0.2
        
    bm.to_mesh(bpy.data.meshes.new("HandleMesh"))
    obj = bpy.data.objects.new("FlushHandle", bpy.data.meshes.get("HandleMesh"))
    bpy.context.collection.objects.link(obj)
    
    # Position on the side of the tank: X=0.22, Y=-0.3 (tank center), Z=0.8
    obj.location = (0.24, -0.3, 0.8)
    
    return obj

def main():
    clear_scene()
    
    ceramic_mat = create_material("CeramicWhite")
    
    bowl = create_bowl()
    tank = create_tank()
    seat = create_seat()
    lid = create_lid()
    handle = create_handle()
    
    parts = [bowl, tank, seat, lid, handle]
    for p in parts:
        if not p.data.materials:
            p.data.materials.append(ceramic_mat)
        else:
            p.data.materials[0] = ceramic_mat

if __name__ == "__main__":
    main()
