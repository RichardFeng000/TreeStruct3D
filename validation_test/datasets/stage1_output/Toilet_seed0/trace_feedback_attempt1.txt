def clear_scene():
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color=(1, 1, 1, 1)):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    node_principled = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_principled.inputs['Base Color'].default_value = color
    node_principled.inputs['Roughness'].default_value = 0.1
    # In Blender 4.0+ / 5.0, Specular is handled differently or via IOR/Roughness
    # We just set a clean white look.
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    mat.node_tree.links.new(node_principled.outputs['BSDF'], node_output.inputs['Surface'])
    return mat

def create_bowl():
    # Create the main bowl and base as one piece
    bm = bmesh.new()
    
    # Base (Pedestal)
    bmesh.ops.create_circle(bm, segments=32, radius=0.15, location=(0, 0, 0))
    # Extrude up to the transition
    face = bm.faces[0]
    res = bmesh.ops.extrude_face_region(bm, geom=[face])
    verts = res['geom']
    for v in verts:
        v.co.z += 0.2
    
    # Expand out to form the bowl base
    bmesh.ops.scale(bm, vec=(1.5, 1.5, 1.0), verts=[v for v in bm.verts if v.co.z > 0.19])
    
    # Create the main basin (approximate sphere/ellipsoid)
    # We'll use a separate mesh for the interior/exterior logic or just model it as a shell
    # To keep it clean, let's create a cylinder and deform it
    bmesh.ops.create_circle(bm, segments=32, radius=0.22, location=(0, 0, 0.2))
    face = bm.faces[-1]
    res = bmesh.ops.extrude_face_region(bm, geom=[face])
    verts = res['geom']
    for v in verts:
        v.co.z += 0.3
    
    # Scale the top of the bowl to be slightly wider and round the bottom
    # Since we want a "smooth porcelain" look, we'll use modifiers later
    # Let's refine the basin shape
    for v in bm.verts:
        if v.co.z > 0.2:
            # Create an oval/round bowl shape
            dist_sq = v.co.x**2 + v.co.y**2
            v.co.z += math.sqrt(max(0, 0.25 - dist_sq)) * 0.5
            
    bm.to_mesh(bpy.data.meshes.new("BowlMesh"))
    obj = bpy.data.objects.new("ToiletBowl", bpy.data.meshes.get("BowlMesh"))
    bpy.context.collection.objects.link(obj)
    
    # Apply subdivision for smoothness
    subdiv = obj.modifiers.new(name="Subdiv", type='SUBSURF')
    subdiv.levels = 2
    subdiv.render_levels = 2
    
    return obj

def create_tank():
    # Rectangular tank with rounded corners
    bm = bmesh.new()
    bmesh.ops.create_cube(bm, size=1.0)
    
    # Scale to tank dimensions: Width ~ 0.4, Depth ~ 0.2, Height ~ 0.35
    # Center it so the bottom is at bowl height
    scale = Vector((0.22, 0.12, 0.18))
    for v in bm.verts:
        v.co *= scale
        v.co.z += 0.4  # Lift up
        v.co.y -= 0.25 # Push back relative to bowl center
        
    bm.to_mesh(bpy.data.meshes.new("TankMesh"))
    obj = bpy.data.objects.new("ToiletTank", bpy.data.meshes.get("TankMesh"))
    bpy.context.collection.objects.link(obj)
    
    # Bevel for rounded edges
    bevel = obj.modifiers.new(name="Bevel", type='BEVEL')
    bevel.width = 0.03
    bevel.segments = 5
    
    subdiv = obj.modifiers.new(name="Subdiv", type='SUBSURF')
    subdiv.levels = 1
    
    return obj

    bpy.context.collection.objects.link(obj)

def create_lid():

    bm = bmesh.new()

        v.co.y *= 1.2

    obj = bpy.data.objects.new("ToiletLid", bpy.data.meshes.get("LidMesh"))
    bpy.context.collection.objects.link(obj)
    
    # Position and tilt
    obj.location = (0, -0.15, 0.6) # Pivot point approximation
    obj.rotation_euler[0] = math.radians(-80) # Tilt upright
    
    subdiv = obj.modifiers.new(name="Subdiv", type='SUBSURF')
    subdiv.levels = 1
    
    return obj

def create_handle():
    # Small flush handle on the side of the tank

    return obj
