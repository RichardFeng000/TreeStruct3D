import bpy
import bmesh
import math
import random
from mathutils import Vector

def clear_scene():
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
    return mat

def create_leaf(name, angle, scale=1.0, material=None):
    """Creates a broad, plicate (ribbed) Veratrum leaf."""
    mesh = bpy.data.meshes.new(name)
    obj = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(obj)

    bm = bmesh.new()
    
    # Dimensions for 'broad' leaves
    segments_length = 30
    segments_width = 24 # Higher resolution for pleats
    leaf_length = 4.0 * scale
    leaf_max_width = 1.2 * scale

    verts = []
    for i in range(segments_length + 1):
        t = i / segments_length
        # Tapered length profile: start narrow, widen quickly, then taper to a point
        w_factor = math.sin(math.pi * (t**0.7)) if t > 0 else 0
        
        # Arching curve of the leaf
        z_bend = -0.5 * (t**2) * scale
        y_offset = t * leaf_length
        
        for j in range(segments_width + 1):
            u = (j / segments_width) - 0.5
            # Longitudinal ribbing: the 'plicate' fold effect
            # Veratrum leaves are folded like a fan along their length
            rib_depth = 0.12 * math.sin(u * math.pi * 6) * scale
            
            x_pos = u * leaf_max_width * w_factor
            v = bm.verts.new(Vector((x_pos, y_offset, z_bend + rib_depth)))
            verts.append(v)

    for i in range(segments_length):
        for j in range(segments_width):
            bm.faces.new((
                verts[i * (segments_width + 1) + j],
                verts[(i + 1) * (segments_width + 1) + j],
                verts[(i + 1) * (segments_width + 1) + j + 1],
                verts[i * (segments_width + 1) + j + 1]
            ))

    bm.to_mesh(mesh)
    bm.free()

    obj.rotation_euler[2] = math.radians(angle)
    # Tilt leaves outward slightly
    obj.rotation_euler[0] = math.radians(-15)
    
    if material:
        obj.data.materials.append(material)
    return obj

def create_stalk(material):
    """Creates a long arching flower stalk originating from origin."""
    curve_data = bpy.data.curves.new('StalkCurve', type='CURVE')
    curve_data.dimensions = '3D'
    curve_data.fill_mode = 'FULL'
    curve_data.bevel_depth = 0.07
    curve_data.bevel_resolution = 4

    polyline = curve_data.splines.new('BEZIER')
    # Points ensuring the stalk starts at origin and arches outward
    points = [
        (0, 0, 0),
        (0.2, 0.5, 2.5),
        (0.8, 1.5, 4.5),
        (1.5, 3.0, 5.5)
    ]
    
    polyline.bezier_points.add(len(points) - 1)
    for i, coord in enumerate(points):
        p = polyline.bezier_points[i]
        p.co = Vector(coord)
        # Smooth handles for a natural arch
        if i == 0:
            p.handle_right = p.co + Vector((0, 0.2, 1.5))
        elif i == len(points)-1:
            p.handle_left = p.co - Vector((0.4, 0.8, 1.0))
        else:
            p.handle_left = p.co - Vector((0.2, 0.5, 1.0))
            p.handle_right = p.co + Vector((0.2, 0.5, 1.0))

    obj = bpy.data.objects.new('Stalk', curve_data)
    bpy.context.collection.objects.link(obj)
    obj.data.materials.append(material)
    return obj

def create_floret(position, material):
    """Creates a small flower cluster."""
    bm = bmesh.new()
    bmesh.ops.create_icosphere(bm, subdivisions=1, radius=0.08, matrix=bpy.types.Object.matrix_world) # dummy
    # Simplified using primitive op for speed and stability in script
    bpy.ops.mesh.primitive_ico_sphere_add(radius=0.07, location=position, subdivisions=1)
    floret = bpy.context.active_object
    floret.data.materials.append(material)
    return floret
