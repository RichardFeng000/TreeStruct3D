def generate_branches(start_pos, direction, radius, length_budget, depth, max_depth, segments=10):
    branches = []
    pos = Vector(start_pos)
    dir_vec = Vector(direction).normalized()
    r = radius
    path = [(Vector(pos), r)]
    seg_len = length_budget / segments

    for i in range(segments):
        p_scale = 0.10 + 0.04 * depth
        if random.random() < 0.08:
            p_scale *= 2.5
        perturb = Vector((
            random.uniform(-p_scale, p_scale),
            random.uniform(-p_scale, p_scale),
            random.uniform(-0.04, 0.12),
        ))
        dir_vec = (dir_vec + perturb * 0.5).normalized()
        pos = pos + dir_vec * seg_len
        t = (i + 1) / segments
        r = radius * (1.0 - 0.35 * t)
        path.append((Vector(pos), r))

    branches.append(path)

    if depth < max_depth and r > 0.035:
        n_sub = random.choice([2, 2, 2, 3])
        for b in range(n_sub):
            if depth == 0:
                angle_off = random.uniform(0.5, 1.0)
            else:
                angle_off = random.uniform(0.3, 0.75)
            axis = Vector((random.uniform(-1, 1), random.uniform(-1, 1), random.uniform(-0.2, 0.5))).normalized()
            new_dir = dir_vec.copy()
            rot = Matrix.Rotation(angle_off * (1 if b % 2 == 0 else -1), 4, axis)
            new_dir = (rot @ new_dir).normalized()
            new_dir.z += random.uniform(0.05, 0.2)
            new_dir = new_dir.normalized()
            new_len = length_budget * random.uniform(0.6, 0.8)
            new_r = r * random.uniform(0.62, 0.78)
            sub = generate_branches(pos, new_dir, new_r, new_len, depth + 1, max_depth, segments)
            branches.extend(sub)

    return branches

def create_tube(bm, path, segments=16):
    if len(path) < 2:
        return
    rings = []
    for i, (point, radius) in enumerate(path):
        if i == 0:
            tangent = (path[1][0] - point).normalized()
        elif i == len(path) - 1:
            tangent = (point - path[i - 1][0]).normalized()
        else:
            tangent = (path[i + 1][0] - path[i - 1][0]).normalized()
        up = Vector((0, 0, 1))
        if abs(tangent.dot(up)) > 0.95:
            up = Vector((1, 0, 0))
        right = tangent.cross(up).normalized()
        up = right.cross(tangent).normalized()
        ring = []
        for j in range(segments):
            ang = 2 * math.pi * j / segments
            offset = (right * math.cos(ang) + up * math.sin(ang)) * radius
            v = bm.verts.new(point + offset)
            ring.append(v)
        rings.append(ring)

    c_start = bm.verts.new(path[0][0])
    for j in range(segments):
        bm.faces.new([c_start, rings[0][(j + 1) % segments], rings[0][j]])

    for i in range(len(rings) - 1):
        for j in range(segments):
            bm.faces.new([
                rings[i][j],
                rings[i][(j + 1) % segments],
                rings[i + 1][(j + 1) % segments],
                rings[i + 1][j]
            ])

    c_end = bm.verts.new(path[-1][0])
    for j in range(segments):
        bm.faces.new([rings[-1][j], rings[-1][(j + 1) % segments], c_end])


def create_base(bm, radius=0.55, height=0.2, segments=20):
    rings = []
    num_rings = 4
    for i in range(num_rings + 1):
        t = i / num_rings
        r = radius * (1.0 - t * 0.6)
        z = height * t
        ring = []
        for j in range(segments):
            ang = 2 * math.pi * j / segments
            noise_r = r * (1.0 + 0.12 * math.sin(ang * 3 + t * 5) + 0.06 * math.sin(ang * 7 + t * 3))
            x = noise_r * math.cos(ang)
            y = noise_r * math.sin(ang)
            v = bm.verts.new((x, y, z))
            ring.append(v)
        rings.append(ring)

def add_dome(bm, center, normal, radius, height, segments=8, num_rings=2):
    up = normal.normalized()
    ref = Vector((1, 0, 0))
    if abs(up.dot(ref)) > 0.95:
        ref = Vector((0, 1, 0))
    right = up.cross(ref).normalized()
    forward = right.cross(up).normalized()
    center = center - up * 0.003

    all_rings = []
    for r_i in range(num_rings + 1):
        t = r_i / num_rings
        rr = radius * math.cos(t * math.pi * 0.5)
        rh = height * math.sin(t * math.pi * 0.5)
        ring = []
        if rr < 0.001:
            v = bm.verts.new(center + up * rh)
            ring.append(v)
        else:
            for j in range(segments):
                ang = 2 * math.pi * j / segments
                offset = (right * math.cos(ang) + forward * math.sin(ang)) * rr
                v = bm.verts.new(center + offset + up * rh)
                ring.append(v)
        all_rings.append(ring)

    for i in range(len(all_rings) - 1):
        r1 = all_rings[i]
        r2 = all_rings[i + 1]
        if len(r2) == 1:
            for j in range(len(r1)):
                bm.faces.new([r1[j], r1[(j + 1) % len(r1)], r2[0]])
        elif len(r1) == 1:
            for j in range(len(r2)):
                bm.faces.new([r1[0], r2[(j + 1) % len(r2)], r2[j]])
        else:
            for j in range(len(r1)):
                bm.faces.new([
                    r1[j],
