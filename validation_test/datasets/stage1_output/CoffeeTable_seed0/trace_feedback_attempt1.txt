def create_rounded_slab(name, length, width, thickness, radius, z_pos):
    """Creates a rectangular slab with rounded corners."""
    # Create mesh and object

    obj = bpy.data.objects.new(name, mesh)

def create_pedestal_leg(name, x_pos, height, shaft_radius, base_radius):
    """Creates a pedestal leg with a flared circular base."""
    mesh = bpy.data.meshes.new(name)
    obj = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(obj)

    bm = bmesh.new()
    
    # Create cylinder (shaft)
    bmesh.ops.create_cone(bm, 
                          cap_ends=True, 
                          segments=32, 
                          radius1=shaft_radius, 
                          radius2=shaft_radius, 
                          depth=height)

    # The cylinder is created along Z axis from -height/2 to height/2
    # Identify the bottom face (lowest z value)
    bottom_face = None
    min_z = float('inf')
    for f in bm.faces:
        center_z = sum(v.co.z for v in f.verts) / len(f.verts)
        if center_z < min_z:
            min_z = center_z
            bottom_face = f

    # Scale the bottom face to create the flare
    # We extrude the bottom face slightly then scale it, or just move vertices
    # Let's use a small extrusion for better geometry
    res = bmesh.ops.extrude_face(bm, face=[bottom_face])
    verts_to_move = res['geom'].verts
    faces_to_move = res['geom'].faces
    
    # Move the extruded face down slightly and scale it out
    for v in verts_to_move:
        v.co.z -= 0.02 # Tiny offset for the base plate thickness
        v.co.x *= (base_radius / shaft_radius)
        v.co.y *= (base_radius / shaft_radius)

    # To make the flare smooth, we can create a transition between 
    # the bottom of the shaft and the outer edge of the base.
    # We'll do this by adding more segments to the cone or simply beveling.
    # Since it's a pedestal, let's refine the shape:
    # Let's restart the leg logic slightly for a "flared" look rather than a flat plate.
    
    bm = bmesh.new()
    # Use create_cone as a way to make a truncated cone (frustum) for the base
    # Base flare part (Bottom)
    bmesh.ops.create_cone(bm, 
                          cap_ends=True, 
                          segments=32, 
                          radius1=base_radius, 
                          radius2=shaft_radius, 
                          depth=0.05) # Flare height
    
    # Shaft part (Top)
    # We can't easily stack cones in BMesh without manual vertex merging.
    # Let's just manually manipulate vertices of a cylinder.
    bm = bmesh.new()
    bmesh.ops.create_cone(bm, cap_ends=True, segments=32, radius1=shaft_radius, radius2=shaft_radius, depth=height)
    
    # Move bottom face to flare
    bottom_face = None
    min_z = float('inf')
    for f in bm.faces:
        cz = sum(v.co.z for v in f.verts)/len(f.verts)
        if cz < min_z: 
            min_z = cz
            bottom_face = f
            
    # Instead of a flat plate, let's make the bottom part of the cylinder conical
    # Select vertices at the bottom and scale them out
    for v in bottom_face.verts:
        v.co.x *= (base_radius / shaft_radius)
        v.co.y *= (base_radius / shaft_radius)
        
    # Bevel the flared edge for smoothness
    edges_to_bevel = [e for e in bm.edges if any(v.co.z < -height/2 + 0.01 for v in e.verts)]
    bmesh.ops.bevel(bm, geom=edges_to_bevel, offset=0.01, segments=3, affect='EDGES')

    bm.to_mesh(mesh)
    bm.free()
    
    obj.location = (x_pos, 0, 0)
    return obj
