def clear_scene():
    """Clears all objects from the current scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Creates a simple diffuse material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    
    # Create Principled BSDF and Output nodes
    node_bsdf = nodes.new(type='ShaderNodeBsdfPrincipled')
    node_output = nodes.new(type='ShaderNodeOutputMaterial')
    
    node_bsdf.inputs['Base Color'].default_value = color
    node_bsdf.inputs['Roughness'].default_value = 0.2  # Slightly shiny
    node_bsdf.inputs['Metallic'].default_value = 0.8   # Metallic look
    
    mat.node_tree.links.new(node_bsdf.outputs['BSDF'], node_output.inputs['Surface'])
    return mat

    # Parameters
    handle_length = 18.0
    handle_width = 0.8
    handle_thickness = 0.3

    bm = bmesh.new()

    # --- Construct the Bowl ---
    # Create a flattened sphere for the bowl base
    # We use a grid-like approach to have better control over the concave shape
    segments = 16
    for i in range(segments + 1):
        phi = math.pi * i / segments
        for j in range(segments + 1):
            theta = 2 * math.pi * j / segments
            
            x = bowl_radius_x * math.sin(phi) * math.cos(theta)
            y = bowl_radius_y * math.sin(phi) * math.sin(theta)
            z = (bowl_radius_x * 0.5) * math.cos(phi) # Flattened z
            
            # Create the "scoop" effect: push top vertices down
            if z > 0:
                # Interpolate depth based on distance from center of bowl surface
                dist_from_center = (x**2 / bowl_radius_x**2 + y**2 / bowl_radius_y**2)**0.5
                z -= bowl_depth * (1.0 - dist_from_center)
            
            bm.verts.new((x, y, z))
