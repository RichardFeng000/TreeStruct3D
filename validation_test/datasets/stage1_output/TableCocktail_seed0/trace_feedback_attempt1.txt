import bpy
import bmesh
import math

def clear_scene():
    """Removes all default objects from the scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_table_top():
    """Creates a circular, slightly beveled table top."""
    radius = 0.28  # 56cm diameter
    thickness = 0.04
    height = 1.1   # Total height to the top surface
    
    # Create cylinder for the top
    bpy.ops.mesh.primitive_cylinder_add(
        vertices=64, 
        radius=radius, 
        depth=thickness, 
        location=(0, 0, height - (thickness / 2))
    )
    top = bpy.context.active_object
    top.name = "TableTop"

    # Add a small bevel to the top and bottom edges for realism
    # We can do this by entering edit mode and selecting edges or using modifiers
    bev = top.modifiers.new(name="Bevel", type='BEVEL')
    bev.width = 0.01
    bev.segments = 3
    
    return top

def create_pedestal():
    """Creates the tapered stem and flared base using a BMesh profile."""
    # Profile points (Radius, Z)
    # We define the silhouette of the pedestal from bottom to top
    profile = [
        (0.35, 0.0),      # Bottom outer edge
        (0.28, 0.02),     # Slight lift for rounded base effect
        (0.15, 0.15),     # Flare transition
        (0.07, 0.6),      # Mid stem taper
        (0.05, 1.06),     # Top of stem connection
    ]
    
    # Add center point for filling the base and top
    center_bottom = (0.0, 0.0)
    center_top = (0.0, 1.06)

    mesh = bpy.data.meshes.new("PedestalMesh")
    obj = bpy.data.objects.new("Pedestal", mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()
    
    # Create the vertices for the profile (on XZ plane)
    profile_verts = []
    for r, z in profile:
        profile_verts.append(bm.verts.new((r, 0, z)))
    
    # Add center points to avoid a hollow tube
    v_bottom_center = bm.verts.new((0, 0, 0))
    v_top_center = bm.verts.new((0, 0, 1.06))
    
    # Connect profile vertices into edges
    for i in range(len(profile_verts) - 1):
        bm.edges.new((profile_verts[i], profile_verts[i+1]))
        
    # Create a "Screw" effect manually by rotating the profile
    segments = 64
    angle_step = (2 * math.pi) / segments
    
    # Generate all vertices for the rotation
    rotated_verts = []
    for s in range(segments):
        angle = s * angle_step
        cos_a = math.cos(angle)
        sin_a = math.sin(angle)
        
        layer = []
        # Rotate profile points
        for r, z in profile:
            layer.append(bm.verts.new((r * cos_a, r * sin_a, z)))
        # Add the center points (they don't move with rotation, but we need 
        # them to link faces for a solid mesh)
        # Note: We only need one set of centers for the whole object
        rotated_verts.append(layer)

    # To avoid overlapping vertices at the center, we create shared center vertices
    shared_bottom_center = bm.verts.new((0, 0, 0))
    shared_top_center = bm.verts.new((0, 0, 1.06))
    
    # Create faces between segments
    for s in range(segments):
        s_next = (s + 1) % segments
        
        curr_layer = rotated_verts[s]
        next_layer = rotated_verts[s_next]
        
        # Connect the profile rings
        for i in range(len(curr_layer) - 1):
            bm.faces.new((curr_layer[i], curr_layer[i+1], next_layer[i+1], next_layer[i]))
            
        # Bottom cap: Connect base ring to center
        bm.faces.new((shared_bottom_center, next_layer[0], curr_layer[0]))
        
        # Top cap: Connect top ring to center
        bm.faces.new((shared_top_center, curr_layer[-1], next_layer[-1]))

    # Fill the bottom base disc and top stem face properly
    # Base Bottom
    for s in range(segments):
        s_next = (s + 1) % segments
        bm.faces.new((shared_bottom_center, rotated_verts[s_next][0], rotated_verts[s][0]))
        
    # Top Stem Cap
    for s in range(segments):
        s_next = (s + 1) % segments
        bm.faces.new((shared_top_center, rotated_verts[s][len(profile)-1], rotated_verts[s_next][len(profile)-1]))

    bm.to_mesh(mesh)
    bm.free()
    
    # Smooth shading
    obj.data.polygons.foreach_set("use_smooth", [True] * len(obj.data.polygons))
    
    # Add Subdivision Surface for a high-fidelity organic feel
    subdiv = obj.modifiers.new(name="Subsurf", type='SUBSURF')
    subdiv.levels = 2
    subdiv.render_levels = 2

    return obj

def main():
    clear_scene()
    
    # Generate components
    create_table_top()
    create_pedestal()
    
    # Final polish: ensure objects are centered and origin is at bottom
    # The geometry created above already centers on X,Y and starts from Z=0

if __name__ == "__main__":
    main()
