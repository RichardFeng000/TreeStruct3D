def clear_scene():
    """Clears the default Blender scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color, metallic=0.0, roughness=0.5):
    """Creates a simple Principled BSDF material."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
        bsdf.inputs['Metallic'].default_value = metallic
        bsdf.inputs['Roughness'].default_value = roughness
    return mat

def create_monitor():
    # --- Parameters ---
    screen_w, screen_h, screen_d = 24.0, 14.0, 0.5
    bezel_thickness = 0.6
    base_w, base_d, base_h = 8.0, 5.0, 0.3
    arm_r, arm_h = 0.4, 4.0

    # Materials
    mat_bezel = create_material("BezelMat", (0.8, 0.75, 0.6, 1.0), metallic=0.9, roughness=0.2) # Tan-Silver
    mat_screen = create_material("ScreenMat", (0.3, 0.3, 0.3, 1.0), metallic=0.1, roughness=0.4)  # Gray Cloud-ish
    mat_base = create_material("BaseMat", (0.1, 0.1, 0.1, 1.0), metallic=0.5, roughness=0.6)     # Dark Base

    # --- 1. Monitor Chassis (Bezel + Screen Body) ---
    # Create the main outer body of the monitor
    bpy.ops.mesh.primitive_cube_add(size=1.0)
    monitor_obj = bpy.context.active_object
    monitor_obj.name = "Monitor_Chassis"
    monitor_obj.scale = (screen_w / 2, screen_h / 2, screen_d / 2)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)

    # Use BMesh to create the recessed screen area
    bm = bmesh.new()
    bm.from_mesh(monitor_obj.data)
    
    # Find the front face (positive Z in local space if we didn't rotate, but cube is X,Y,Z)
    # Let's assume Z is depth (front/back), Y is height, X is width
    # Re-scaling: scale was applied, so dims are screen_w, screen_h, screen_d.
    # Actually let's orient it: X=width, Z=height, Y=depth
    monitor_obj.rotation_euler = (0, 0, 0) # Reset to standard
    # Re-scale for correct orientation: X=Width, Y=Depth, Z=Height
    bpy.ops.object.select_all(action='DESELECT')
    bpy.context.view_layer.objects.active = monitor_obj
    monitor_obj.scale = (screen_w / 2, screen_d / 2, screen_h / 2)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)

    bm = bmesh.new()
    bm.from_mesh(monitor_obj.data)
    
    # Find the front face (Positive Y axis)
    front_face = None
    for f in bm.faces:
        if f.normal.y > 0.9:
            front_face = f
            break
    
    if front_face:
        # Inset for bezel
        res = bmesh.ops.inset_individual(bm, faces=[front_face], width=bezel_thickness)
        screen_face = res['faces'][0]
        # Extrude inwards to create the display area
        bmesh.ops.translate(bm, verts=screen_face.verts, vec=(0, -0.1, 0))
    
    bm.to_mesh(monitor_obj.data)
    bm.free()

    # Assign Materials via face indices (simulated)
    # Since we have a recessed area, the screen face is now separate from bezel faces
    # We will iterate through faces and check distance to center for material assignment
    for f in monitor_obj.data.polygons:
        center = f.center
        # If it's the recessed face (y is smaller than outer edge)
        if center.y < (screen_d / 2):
            f.material_index = 1 # ScreenMat
        else:
            f.material_index = 0 # BezelMat

    monitor_obj.data.materials.append(mat_bezel)
    monitor_obj.data.materials.append(mat_screen)

    # Add bevel for realism
    bevel_mod = monitor_obj.modifiers.new(name="Bevel", type='BEVEL')
    bevel_mod.width = 0.1
    bevel_mod.segments = 3

    # --- 2. Pedestal Arm ---
    bpy.ops.mesh.primitive_cylinder_add(radius=arm_r, depth=arm_h, location=(0, -screen_d/4, -screen_h/2 + arm_h/2))
    arm_obj = bpy.context.active_object
    arm_obj.name = "Pedestal_Arm"
    arm_obj.data.materials.append(mat_base)

    # --- 3. Base Stand ---
    bpy.ops.mesh.primitive_cube_add(size=1.0, location=(0, 0, -screen_h/2))
    base_obj = bpy.context.active_object
    base_obj.name = "Base_Stand"
    base_obj.scale = (base_w / 2, base_d / 2, base_h / 2)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    base_obj.data.materials.append(mat_base)

    # Bevel the base
    base_bevel = base_obj.modifiers.new(name="Bevel", type='BEVEL')
    base_bevel.width = 0.15
    base_bevel.segments = 3

    # Final positioning and centering
    # Offset everything so the base is at Z=0
    offset_z = screen_h/2
    monitor_obj.location.z = offset_z
    arm_obj.location.z = offset_z + (-screen_h/2 + arm_h/2)
    base_obj.location.z = offset_z + (-screen_h/2)

def main():
