def clear_scene():
    """Clears all objects from the current scene."""
    if bpy.context.object and bpy.context.object.mode != 'OBJECT':
        bpy.ops.object.mode_set(mode='OBJECT')
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Creates a simple Principled BSDF material with a given color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    
    # Create principled BSDF and output nodes
    shader = nodes.new('ShaderNodeBsdfPrincipled')
    shader.inputs['Base Color'].default_value = color
    output = nodes.new('ShaderNodeOutputMaterial')
    
    links = mat.node_tree.links
    links.new(shader.outputs['BSDF'], output.inputs['Surface'])
    return mat

def create_box(name, size, location, material):
    """Creates a box mesh with given dimensions and assigns a material."""
    mesh = bpy.data.meshes.new(name)
    obj = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()
    # Create cube centered at origin
    bmesh.ops.create_cube(bm, size=1.0)
    # Scale to match requested dimensions
    # Note: create_cube size is the total edge length of the unit cube? 
    # Actually, bmesh.ops.create_cube creates a cube from -0.5 to 0.5 if size is 1.0.
    # We scale manually by multiplying vertices.
    for v in bm.verts:
        v.co.x *= size[0]
        v.co.y *= size[1]
        v.co.z *= size[2]
    
    bm.to_mesh(mesh)
    bm.free()
    
    obj.location = location
    obj.data.materials.append(material)
    return obj

def create_cylinder_between_points(name, p1, p2, radius, material):
    """Creates a cylinder that spans between two 3D points."""
    # Calculate direction and length
    direction = p2 - p1
    length = direction.length
    rot_quat = direction.to_track_quat('Z', 'Y')
    
    mesh = bpy.data.meshes.new(name)
    obj = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()
    # Create a cylinder aligned with Z axis
    bmesh.ops.create_cylinder(
        bm, 
        cap_ends=True, 
        segments=16, 
        radius=radius, 
        depth=length
    )
    
    # Shift the mesh so it's centered on length/2 (since create_cylinder centers at origin)
    for v in bm.verts:
        v.co.z += length / 2.0
        
    bm.to_mesh(mesh)
    bm.free()
    
    obj.rotation_mode = 'QUATERNION'
    obj.rotation_quaternion = rot_quat
    # Position it at the start point p1
    obj.location = p1
    obj.data.materials.append(material)
    return obj
