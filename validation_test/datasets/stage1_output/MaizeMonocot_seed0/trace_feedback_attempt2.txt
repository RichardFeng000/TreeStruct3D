def create_stalk(height=4.0, radius=0.1):
    """Creates a segmented corn stalk with nodes."""
    bm = bmesh.new()
    segments = 15
    rings = 8
    
    # Create vertices for the stalk
    for i in range(segments + 1):
        z = (i / segments) * height
        # Add slight bulge at nodes (sinusoidal variance)
        node_scale = 1.0 + 0.15 * math.cos(i * math.pi) if i < segments else 1.0
        # Taper slightly towards the top
        curr_radius = radius * node_scale * (1.0 - (z / (height * 4)))
        
        for j in range(rings):
            angle = (j / rings) * 2 * math.pi
            x = math.cos(angle) * curr_radius
            y = math.sin(angle) * curr_radius
            bm.verts.new((x, y, z))
    
    bm.verts.ensure_lookup_table()
    
    # Create faces
    for i in range(segments):
        for j in range(rings):
            v1 = bm.verts[i * rings + j]
            v2 = bm.verts[i * rings + (j + 1) % rings]
            v3 = bm.verts[(i + 1) * rings + (j + 1) % rings]
            v4 = bm.verts[(i + 1) * rings + j]
            bm.faces.new((v1, v2, v3, v4))
            
    mesh = bpy.data.meshes.new("StalkMesh")
    bm.to_mesh(mesh)
    bm.free()
    
    obj = bpy.data.objects.new("CornStalk", mesh)
    bpy.context.collection.objects.link(obj)
    return obj

def create_leaf(length=2.5, width=0.4):
    """Creates a single arching leaf with thickness."""
    bm = bmesh.new()
    res_l = 16
    res_w = 6
    
    verts = []
    # Base geometry (a plane)
    for i in range(res_l + 1):
        u = i / res_l # 0 to 1
        x = u * length
        # Arching curve: parabolic drop and slight lift
        z = -0.3 * (u**2) * length + (0.1 * u)
        y_offset = 0.2 * math.sin(u * math.pi)
        
        for j in range(res_w + 1):
            v = j / res_w # 0 to 1
            # Taper width: wide at base, narrow at tip
            w = (v - 0.5) * width * (1.0 - u * 0.8)
            verts.append(bm.verts.new((x, y_offset + w, z)))
            
    bm.verts.ensure_lookup_table()
    
    # Create faces for the surface
    faces = []
    for i in range(res_l):
        for j in range(res_w):
            idx = i * (res_w + 1) + j
            f = bm.faces.new((
                verts[idx], 
                verts[idx + 1], 
                verts[(i+1)*(res_w+1) + j + 1], 
                verts[(i+1)*(res_w+1) + j]
            ))
            faces.append(f)
    
    # Fix forThickness: Extrude the faces and move them
    extrude_result = bmesh.ops.extrude_face_region(bm, geom=faces)
    extruded_verts = [v for v in extrude_result['geom'] if isinstance(v, bmesh.types.BMVert)]
    for v in extruded_verts:
        # Offset along Z to simulate thickness
        v.co += Vector((0, 0, -0.01))
    
    mesh = bpy.data.meshes.new("LeafMesh")
    bm.to_mesh(mesh)
    bm.free()
    
    obj = bpy.data.objects.new("Leaf", mesh)
    bpy.context.collection.objects.link(obj)
    return obj

def create_ear(position=(0, 0, 0)):
    """Creates an ear of corn consisting of a cob and kernels."""
    bm = bmesh.new()
    
    # Cob geometry (Cylinder)
    cob_radius = 0.12
    cob_height = 0.6
    bmesh.ops.create_cylinder(bm, segments=12, radius=cob_radius, depth=cob_height)
    
    # Kernel geometry: create small boxes around the cob
    kernel_size = 0.025
    rows = 14 # longitudinal
    cols = 16 # circumferential
    
    for r in range(rows):
        z_off = -cob_height/2 + (r / (rows-1)) * cob_height
        for c in range(cols):
            angle = (c / cols) * 2 * math.pi
            kx = math.cos(angle) * (cob_radius + kernel_size/2)
            ky = math.sin(angle) * (cob_radius + kernel_size/2)
            
            # Create a small cube for the kernel
            s = kernel_size / 2
            v1 = bm.verts.new((kx-s, ky-s, z_off-s))
            v2 = bm.verts.new((kx+s, ky-s, z_off-s))
            v3 = bm.verts.new((kx+s, ky+s, z_off-s))
            v4 = bm.verts.new((kx-s, ky+s, z_off-s))
            v5 = bm.verts.new((kx-s, ky-s, z_off+s))
            v6 = bm.verts.new((kx+s, ky-s, z_off+s))
            v7 = bm.verts.new((kx+s, ky+s, z_off+s))
            v8 = bm.verts.new((kx-s, ky+s, z_off+s))
            bm.faces.new((v1, v2, v6, v5))
            bm.faces.new((v2, v3, v7, v6))
            bm.faces.new((v3, v4, v8, v7))
            bm.faces.new((v4, v1, v5, v8))
            bm.faces.new((v5, v6, v7, v8))
            bm.faces.new((v1, v4, v3, v2))

    mesh = bpy.data.meshes.new("EarMesh")
    bm.to_mesh(mesh)
    bm.free()
    obj = bpy.data.objects.new("CornEar", mesh)
    bpy.context.collection.objects.link(obj)
    obj.location = position
    return obj

def create_husk(position, rotation):
    """Creates a husk leaf wrapping the ear."""
    bm = bmesh.new()
    res = 10
    for i in range(res + 1):
        u = i / res
        x = u * 0.7 # Length along corn axis
        z = -0.1 * (u**2) * 0.7
        for j in range(res + 1):
            v = j / res
            w = (v - 0.5) * 0.3 * (1.0 - u * 0.4) # Tapered width
            bm.verts.new((x, w, z))
            
    bm.verts.ensure_lookup_table()
    for i in range(res):
        for j in range(res):
            idx = i * (res + 1) + j
            bm.faces.new((
                bm.verts[idx], 
                bm.verts[idx+1], 
                bm.verts[(i+1)*(res+1)+j+1], 
                bm.verts[(i+1)*(res+1)+j]
            ))
    
    mesh = bpy.data.meshes.new("HuskMesh")
    bm.to_mesh(mesh)
    bm.free()
    obj = bpy.data.objects.new("Husk", mesh)
    bpy.context.collection.objects.link(obj)
    obj.location = position
    obj.rotation_euler = rotation
    return obj

def assemble_maize():
    clear_scene()
    
    stalk_h = 4.0
    stalk = create_stalk(height=stalk_h)
    
    # Create leaves in a spiraled pattern along the stalk
    num_leaves = 12
    for i in range(num_leaves):
        z = (i / num_leaves) * (stalk_h * 0.85)
        angle = (i / num_leaves) * 2 * math.pi * 2.5 # Golden angle-ish spiral
        
        leaf = create_leaf()
        leaf.location = (0, 0, z)
        # Rotation to make it sprout from stalk
        leaf.rotation_euler = (0, 0, angle)
        # Tilt outwards and downwards
        leaf.rotation_euler.x = math.radians(random.uniform(-40, -70))
        leaf.rotation_euler.y = math.radians(random.uniform(-10, 10))
    
    # Create Ears of Corn
    num_ears = 2
    for i in range(num_ears):
        z_ear = 1.5 + (i * 0.7)
        angle_ear = (random.random() * 2 * math.pi)
        pos_ear = (math.cos(angle_ear)*0.1, math.sin(angle_ear)*0.1, z_ear)
        
        # Rotate ear to be slightly angled upwards
        ear = create_ear(position=pos_ear)
        rot_euler = (math.radians(35), 0, angle_ear + math.pi/2)
        ear.rotation_euler = rot_euler
        
        # Add husks wrapping the ear
        for h in range(4):
            husk = create_husk(position=pos_ear, rotation=rot_euler)
            husk.rotation_euler.z += (h * 0.8)
            husk.scale = (1.0, 1.0, 1.0)

    # Create Tassel at the very top
    create_tassel(top_pos=(0, 0, stalk_h))
