import bpy
import bmesh
import math
from mathutils import Vector, Matrix

def clear_scene():
    """Removes all default objects from the scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Creates a simple material with a specific RGBA color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
    return mat

def create_stem(material):
    """Creates a thin low-poly cylindrical stem."""
    # Create mesh and object
    mesh = bpy.data.meshes.new("Stem")
    obj = bpy.data.objects.new("Stem", mesh)
    bpy.context.collection.objects.link(obj)

    bm = bmesh.new()
    
    # Parametric cylinder: radius 0.02, height 1.0, vertices=8 for low poly look
    segments = 8
    radius = 0.03
    height = 1.0
    
    # Create bottom circle
    verts_bottom = []
    for i in range(segments):
        angle = (2 * math.pi * i) / segments
        x = math.cos(angle) * radius
        y = math.sin(angle) * radius
        verts_bottom.append(bm.verts.new((x, y, 0)))
        
    # Create top circle
    verts_top = []
    for i in range(segments):
        angle = (2 * math.pi * i) / segments
        x = math.cos(angle) * radius
        y = math.sin(angle) * radius
        verts_top.append(bm.verts.new((x, y, height)))
        
    # Bridge the circles with faces
    for i in range(segments):
        v1 = verts_bottom[i]
        v2 = verts_bottom[(i + 1) % segments]
        v3 = verts_top[(i + 1) % segments]
        v4 = verts_top[i]
        bm.faces.new((v1, v2, v3, v4))
        
    # Close the caps (low poly triangle fans)
    bm.faces.new(verts_bottom)
    bm.faces.new(verts_top)
    
    bm.to_mesh(mesh)
    bm.free()
    
    obj.data.materials.append(material)
    return obj

def create_leaf(name, material):
    """Creates a single low-poly oval leaflet."""
    mesh = bpy.data.meshes.new(name)
    obj = bpy.data.objects.new(name, mesh)
    bpy.context.collection.objects.link(obj)
    
    bm = bmesh.new()
    
    # We create a low poly 'flattened' sphere/ellipsoid for the leaf
    # Start with an IcoSphere-like structure but manually scaled
    # Using 4 segments around, 3 rings along
    rings = 4
    segments = 6
    
    verts = []
    for i in range(rings):
        phi = (math.pi * i) / (rings - 1)
        ring_verts = []
        # Vary radius for a rounded leaf shape
        r = math.sin(phi) * 0.2 
        for j in range(segments):
            theta = (2 * math.pi * j) / segments
            x = math.cos(theta) * r
            y = math.sin(theta) * r
            z = phi # This will be the length of the leaf
            ring_verts.append(bm.verts.new((x, y, z)))
        verts.append(ring_verts)
        
    for i in range(rings - 1):
        for j in range(segments):
            v1 = verts[i][j]
            v2 = verts[i][(j + 1) % segments]
            v3 = verts[i+1][(j + 1) % segments]
            v4 = verts[i+1][j]
            bm.faces.new((v1, v2, v3, v4))
            
    # Cap the ends
    bm.faces.new(verts[0])
    bm.faces.new(verts[-1])
    
    # Flatten the leaf on one axis (make it a disc-like petal)
    # We'll do this via object scaling later to maintain clean bmesh logic here
    
    bm.to_mesh(mesh)
    bm.free()
    
    obj.data.materials.append(material)
    return obj

def main():
    clear_scene()
    
    # Dark Green color (R, G, B, A)
    dark_green = (0.05, 0.25, 0.05, 1.0)
    mat = create_material("CloverGreen", dark_green)
    
    # Stem
    stem = create_stem(mat)
    stem_height = 1.0
    
    # Leaf configuration
    num_leaves = 3
    for i in range(num_leaves):
        leaf = create_leaf(f"Leaf_{i}", mat)
        
        # Scale leaf to be flat and oval
        leaf.scale = (0.8, 0.3, 0.4) # X=width, Y=thickness, Z=length
        
        # Position at top of stem
        angle = (2 * math.pi * i) / num_leaves
        
        # The leaf was created along the Z axis, so we need to rotate it
        # 1. Rotate to point outward from center
        # 2. Tilt slightly upward
        
        # Rotation for radial arrangement
        rot_quat = Matrix.Rotation(angle, 4, 'Z') @ \
                   Matrix.Rotation(math.radians(60), 4, 'Y') # Lean outwards
        
        leaf.rotation_euler = (0, 0, 0) # Reset
        leaf.rotation_mode = 'QUATERNION'
        leaf.rotation_quaternion = (rot_quat @ Matrix.Identity()).to_quaternion()
        
        # Offset position: stem top + slight radial shift
        # Because the leaf is scaled and rotated, we move it to the tip
        offset_dist = 0.05
        leaf.location = (
            math.cos(angle) * offset_dist,
            math.sin(angle) * offset_dist,
            stem_height
        )

if __name__ == "__main__":
