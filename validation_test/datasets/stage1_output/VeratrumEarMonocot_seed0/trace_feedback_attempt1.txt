import bpy
import bmesh
import math
import random
from mathutils import Vector, Matrix

def clear_scene():
    """Clears the default Blender scene."""
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()

def create_material(name, color):
    """Creates a simple principled BSDF material with a specific color."""
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    bsdf = nodes.get("Principled BSDF")
    if bsdf:
        bsdf.inputs['Base Color'].default_value = color
    return mat

def create_bract_mesh():
    """Generates a single leaf-like bract mesh."""
    bm = bmesh.new()
    
    # Parameters for the leaf shape
    length = 1.2
    width = 0.3
    segments = 8
    
    # Create vertices for a tapered leaf shape
    # We'll create two strips of vertices (left and right) and bridge them
    verts_l = []
    verts_r = []
    
    for i in range(segments + 1):
        t = i / segments
        # Parabolic-like width profile: peaks early, then tapers to a point
        w = width * math.sin(math.pi * t * 0.7) if t < 0.8 else width * (1.0 - t) * 0.5
        if w < 0: w = 0
        
        # Add a slight curve to the leaf longitudinally
        z = t * length
        x_offset = 0.1 * (t**2) # Slight bend
        
        verts_l.append(bm.verts.new((x_offset - w, 0, z)))
        verts_r.append(bm.verts.new((x_offset + w, 0, z)))

    # Create faces between the strips
    for i in range(segments):
        bm.faces.new((verts_l[i], verts_l[i+1], verts_r[i+1], verts_r[i]))

    # Give it some thickness (extrude slightly)
    bmesh.ops.extrude_face_region(bm, geom=bm.faces[:])
    for v in bm.verts:
        if v.select: # New vertices from extrusion
            v.co.y += 0.02

    # Final touch: make the tip very sharp and the base slightly rounded
    bm.to_mesh("bract_temp")
    bm.free()
    return "bract_temp"

def assemble_veratrum():
    clear_scene()

    # Create materials for gradient effect
    dark_green = create_material("DarkGreen", (0.02, 0.15, 0.02, 1.0))
    mid_green = create_material("MidGreen", (0.05, 0.3, 0.05, 1.0))
    light_green = create_material("LightGreen", (0.15, 0.45, 0.15, 1.0))

    # Central Stem
    stem_radius = 0.15
    stem_height = 6.0
    bpy.ops.mesh.primitive_cylinder_add(radius=stem_radius, depth=stem_height, location=(0, 0, stem_height / 2))
    stem = bpy.context.active_object
    stem.name = "Stem"
    stem.data.materials.append(dark_green)

    # Bract setup
    bract_mesh_name = create_bract_mesh()
    temp_obj = bpy.data.objects.new("TempBract", bpy.data.meshes[bract_mesh_name])
    bpy.context.collection.objects.link(temp_obj)

    # Distribution parameters
    num_bracts = 120
    phi = math.pi * (3 - math.sqrt(5)) # Golden angle in radians (~137.5 degrees)
    z_spacing = stem_height / num_bracts
    
    all_bract_objs = []

    for i in range(num_bracts):
        # Calculate position and rotation
        angle = i * phi
        z = i * z_spacing
        
        # Create a duplicate of the bract
        bract_obj = temp_obj.copy()
        bract_obj.data = temp_obj.data.copy()
        bpy.context.collection.objects.link(bract_obj)
        
        # Position
        bract_obj.location = (math.cos(angle) * stem_radius, math.sin(angle) * stem_radius, z)
        
        # Rotation: Point outwards from center and angle upwards slightly
        rot_quat = Vector((0, 0, angle)).to_track_quat('Y', 'Z')
        bract_obj.rotation_mode = 'QUATERNION'
        bract_obj.rotation_quaternion = rot_quat
        
        # Add a slight tilt so they overlap like scales
        bract_obj.rotation_euler = (0.4, 0, 0) # Tilt up
        
        # Scale based on height: slightly smaller at the top
        scale_factor = 1.0 - (i / num_bracts) * 0.3
        bract_obj.scale = (scale_factor, scale_factor, scale_factor)

        # Assign material based on Z position for gradient effect
        if i < num_bracts * 0.3:
            mat = dark_green
        elif i < num_bracts * 0.7:
            mat = mid_green
        else:
            mat = light_green
        bract_obj.data.materials.append(mat)
        
        all_bract_objs.append(bract_obj)

    # Cleanup temp object
    bpy.context.collection.objects.remove(temp_obj)
    bpy.data.meshes.remove(bpy.data.meshes[bract_mesh_name])

    # Join all bracts and stem into one object for a clean output
    bpy.ops.object.select_all(action='DESELECT')
    stem.select_set(True)
    for obj in all_bract_objs:
        obj.select_set(True)
    
    bpy.context.view_layer.objects.active = stem
    bpy.ops.object.join()
    
    # Final Polish: Add a Subdivision Surface modifier to the whole thing for organic feel
    subdiv = stem.modifiers.new(name="Subdiv", type='SUBSURF')
    subdiv.levels = 1
    subdiv.render_levels = 2

if __name__ == "__main__":
    assemble_veratrum()
