def create_limb(name, pos, rot, scale=(0.3, 0.3, 1.5)):
    bpy.ops.mesh.primitive_cylinder_add(vertices=12, radius=0.3, depth=1.5, location=pos, rotation=rot)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = scale
    return obj

def build_creature():
    # 1. Create Torso
    bpy.ops.mesh.primitive_uv_sphere_add(radius=1, location=(0, 0, 1))
    body = bpy.context.active_object
    body.name = "Body"
    body.scale = (2.2, 0.8, 0.9)

    # 2. Create Head
    bpy.ops.mesh.primitive_uv_sphere_add(radius=0.6, location=(2.5, 0, 1.4))
    head = bpy.context.active_object
    head.name = "Head"
    head.scale = (1.1, 1.0, 0.9)

    # Muzzle
    bpy.ops.mesh.primitive_uv_sphere_add(radius=0.3, location=(3.0, 0, 1.2))
    muzzle = bpy.context.active_object
    muzzle.scale = (1.2, 0.8, 0.7)

    # 3. Create Legs
    legs_pos = [
        (1.2, 0.5, 0.4),   # Front Right
        (1.2, -0.5, 0.4),  # Front Left
        (-1.2, 0.5, 0.4),  # Back Right
        (-1.2, -0.5, 0.4)  # Back Left
    ]
    legs = []
    for i, pos in enumerate(legs_pos):
        bpy.ops.mesh.primitive_cylinder_add(vertices=12, radius=0.25, depth=1.2, location=pos)
        leg = bpy.context.active_object
        leg.name = f"Leg_{i}"
        # Crouch pose: rotate slightly
        if i < 2: # Front legs
            leg.rotation_euler[1] = math.radians(-15)
        else: # Back legs
            leg.rotation_euler[1] = math.radians(10)
        legs.append(leg)

    # 4. Create Tail
    bpy.ops.mesh.primitive_cylinder_add(vertices=8, radius=0.15, depth=2.0, location=(-2.7, 0, 1))
    tail = bpy.context.active_object
    tail.name = "Tail"
    tail.rotation_euler[1] = math.radians(-30)
    tail.scale = (0.5, 0.5, 1.2)

    # Join all parts into one mesh for the fur system and material
    bpy.ops.object.select_all(action='DESELECT')
    body.select_set(True)
    head.select_set(True)
    muzzle.select_set(True)
    tail.select_set(True)
    for leg in legs:
        leg.select_set(True)
    
    bpy.context.view_layer.objects.active = body
    bpy.ops.object.join()

    # Apply Subdivision Surface for organic look
    subsurf = body.modifiers.new(name="Subdiv", type='SUBSURF')
    subsurf.levels = 2
    subsurf.render_levels = 3

    return body

def apply_fur(obj):
    # Create Particle System
    part = obj.modifiers.new(name="Fur", type='PARTICLE_SYSTEM')
    psys = obj.particle_systems[0]
    settings = psys.settings
    
    settings.type = 'HAIR'
    settings.render_type = 'PATH'
    settings.count = 50000 # Dense fur
    settings.hair_length = 0.15
    
    # Roughness for "fluffy" look
    settings.roughness_1 = 0.2
    settings.roughness_2 = 0.1
    
    # Child particles for extreme density without crashing Blender
    settings.child_type = 'INTERPOLATED'
    settings.children_count = 10

def main():
    clear_scene()
    
    creature = build_creature()
    
    # Material: Orange-Red, Black stripes, White belly
    orange = (0.8, 0.3, 0.1, 1)
    black = (0.02, 0.02, 0.02, 1)
    tiger_mat = create_material("TigerFur", orange, black, is_tiger=True)
    creature.data.materials.append(tiger_mat)
    
    apply_fur(creature)
    
    # Final positioning to ensure it's at origin and slightly crouching
    creature.location = (0, 0, 0)

if __name__ == "__main__":
    main()
